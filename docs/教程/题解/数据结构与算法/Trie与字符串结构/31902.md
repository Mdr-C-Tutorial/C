# 31902：基于 Trie 的整数词典与节点回收

## 题目

在仅包含 ASCII 小写字母的 Trie 上实现整数词典 `LowerDictionary`，支持键值插入或更新（`put`）、按键读取（`get`）以及按键删除并回收尾部无用结点（`remove`）。

## 解析

每个结点保存 `children[26]`、布尔标记 `has_value` 与关联的 64 位无符号整数 `value`：

1. **强异常安全的插入与更新 (`put`)**：先校验全部字符属于 `'a'` 到 `'z'`，再沿已有分支下探。若完整走完 `length` 步，则直接在终点结点上更新 `value`，并根据旧 `has_value` 设置 `*out_replaced`；此时无需分配任何新结点。若在深度 `split` 处遇到空分支，则先在树外独立分配剩余的 `length - split` 个结点并串联成单链，分配全部成功后再一次性挂接到 `cursor` 下。这样一旦中途内存不足，只需释放树外临时链，词典原有结构丝毫不变。
2. **不污染输出对象的查找 (`get`)**：完整校验输入合法后沿树下探。若中途遇到空指针或终点 `has_value` 为假，按照题目要求保持 `*out_value` 不变并成功返回 `true`；仅当命中有效键时才写入 `*out_value`。
3. **自底向上的尾部结点回收 (`remove`)**：删除键时递归（或沿路径回溯）到达终点结点。若终点不存在该键，置 `*out_removed = false` 并原样返回；否则清除终点的 `has_value` 标记。在回溯过程中，若某个子结点满足 `!child->has_value` 且全部 26 个子指针均为空，说明该子结点不再承载任何现存键或更长键的前缀，父结点立即调用 `free` 释放该子结点并将对应槽位置回 `NULL`。根结点本身始终保留以代表空前缀起点。

## 答案

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

enum {
    DICT_ALPHA = 26
};

typedef struct DictNode {
    struct DictNode *children[DICT_ALPHA];
    uint64_t value;
    bool has_value;
} DictNode;

struct LowerDictionary {
    DictNode *root;
};

typedef struct LowerDictionary LowerDictionary;

static bool valid_key_bytes(
    const unsigned char *word,
    size_t length
) {
    if (length > 0 && word == NULL) {
        return false;
    }
    for (size_t i = 0; i < length; ++i) {
        if (word[i] < 'a' || word[i] > 'z') {
            return false;
        }
    }
    return true;
}

static bool node_has_children(const DictNode *node) {
    for (size_t i = 0; i < DICT_ALPHA; ++i) {
        if (node->children[i] != NULL) {
            return true;
        }
    }
    return false;
}

static void destroy_dict_nodes(DictNode *node) {
    if (node == NULL) {
        return;
    }
    for (size_t i = 0; i < DICT_ALPHA; ++i) {
        destroy_dict_nodes(node->children[i]);
    }
    free(node);
}

LowerDictionary *lower_dictionary_create(void) {
    LowerDictionary *dictionary = malloc(sizeof *dictionary);
    if (dictionary == NULL) {
        return NULL;
    }
    dictionary->root = calloc(1, sizeof *dictionary->root);
    if (dictionary->root == NULL) {
        free(dictionary);
        return NULL;
    }
    return dictionary;
}

void lower_dictionary_destroy(LowerDictionary *dictionary) {
    if (dictionary == NULL) {
        return;
    }
    destroy_dict_nodes(dictionary->root);
    free(dictionary);
}

bool lower_dictionary_put(
    LowerDictionary *dictionary,
    const unsigned char *word,
    size_t length,
    uint64_t value,
    bool *out_replaced
) {
    if (dictionary == NULL || out_replaced == NULL ||
        !valid_key_bytes(word, length)) {
        return false;
    }

    DictNode *cursor = dictionary->root;
    size_t split = 0;
    while (split < length) {
        size_t index = (size_t)(word[split] - 'a');
        if (cursor->children[index] == NULL) {
            break;
        }
        cursor = cursor->children[index];
        ++split;
    }

    if (split == length) {
        bool replaced = cursor->has_value;
        cursor->value = value;
        cursor->has_value = true;
        *out_replaced = replaced;
        return true;
    }

    DictNode *chain_head = NULL;
    DictNode *chain_tail = NULL;
    for (size_t i = split; i < length; ++i) {
        DictNode *node = calloc(1, sizeof *node);
        if (node == NULL) {
            destroy_dict_nodes(chain_head);
            return false;
        }
        if (chain_head == NULL) {
            chain_head = node;
        } else {
            size_t prev_index = (size_t)(word[i - 1] - 'a');
            chain_tail->children[prev_index] = node;
        }
        chain_tail = node;
    }

    size_t first_index = (size_t)(word[split] - 'a');
    cursor->children[first_index] = chain_head;
    chain_tail->value = value;
    chain_tail->has_value = true;
    *out_replaced = false;
    return true;
}

bool lower_dictionary_get(
    const LowerDictionary *dictionary,
    const unsigned char *word,
    size_t length,
    uint64_t *out_value
) {
    if (dictionary == NULL || out_value == NULL ||
        !valid_key_bytes(word, length)) {
        return false;
    }

    const DictNode *cursor = dictionary->root;
    for (size_t i = 0; i < length; ++i) {
        size_t index = (size_t)(word[i] - 'a');
        if (cursor->children[index] == NULL) {
            return true;
        }
        cursor = cursor->children[index];
    }
    if (cursor->has_value) {
        *out_value = cursor->value;
    }
    return true;
}

static bool remove_recursive(
    DictNode *node,
    const unsigned char *word,
    size_t length,
    size_t depth,
    bool *out_removed
) {
    if (depth == length) {
        if (!node->has_value) {
            *out_removed = false;
            return false;
        }
        node->has_value = false;
        node->value = 0;
        *out_removed = true;
        return !node_has_children(node);
    }

    size_t index = (size_t)(word[depth] - 'a');
    DictNode *child = node->children[index];
    if (child == NULL) {
        *out_removed = false;
        return false;
    }

    bool should_free_child = remove_recursive(
        child, word, length, depth + 1, out_removed
    );
    if (should_free_child) {
        free(child);
        node->children[index] = NULL;
        return !node->has_value && !node_has_children(node);
    }
    return false;
}

bool lower_dictionary_remove(
    LowerDictionary *dictionary,
    const unsigned char *word,
    size_t length,
    bool *out_removed
) {
    if (dictionary == NULL || out_removed == NULL ||
        !valid_key_bytes(word, length)) {
        return false;
    }
    bool removed = false;
    (void)remove_recursive(
        dictionary->root, word, length, 0, &removed
    );
    *out_removed = removed;
    return true;
}
```

查找、插入与删除均只沿长度为 $L = |word|$ 的唯一路径访问结点，每个结点上的子指针检查耗时 $O(|\Sigma|) = O(1)$，因此单次操作的时间复杂度严格为 $O(|word|)$，尾部无用结点在删除回溯时即时释放，不留悬空路径。

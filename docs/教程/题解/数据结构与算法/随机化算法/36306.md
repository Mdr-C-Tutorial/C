# 36306 - 跳跃表（Skip List）随机层数生成与对数查找/插入

## 题目

实现一个基于概率均衡的**跳跃表（Skip List）**数据结构，支持按 64 位整型键值进行有序存储，并实现 $O(\log n)$ 期望时间复杂度的**键值查找与节点插入**。

函数签名与数据结构定义如下：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef uint64_t (*RandomBits)(void *context);

typedef struct SkipNode SkipNode;
struct SkipNode {
    int64_t key;
    uint64_t val;
    SkipNode *forward[];
};

typedef struct {
    SkipNode *header;
    size_t level;
    size_t length;
    RandomBits random_bits;
    void *context;
} SkipList;

SkipList *skiplist_create(RandomBits random_bits, void *context);
bool skiplist_insert(SkipList *list, int64_t key, uint64_t val);
bool skiplist_search(const SkipList *list, int64_t key, uint64_t *out_val);
void skiplist_destroy(SkipList *list);
```

要求：
1. 最大层数限制为 `MAX_LEVEL = 16`。
2. 节点的随机层数生成必须符合**几何分布**：每次以概率 $p = 1/2$ 决定是否晋升至更高一层。
3. 插入操作：若键已存在，更新对应的值；若键不存在，创建新节点，维护各层的前驱指针链接。
4. 资源释放必须完整，严禁任何内存泄漏。

## 解题思路

1. **跳跃表的层级跳跃原理**：
   普通有序链表的查找只能按顺序线性扫描，耗时 $O(n)$。
   跳跃表通过在链表之上建立多级“快速索引道（Express Lanes）”：
   - 0 层包含所有元素；
   - 1 层包含约 $1/2$ 的元素；
   - 2 层包含约 $1/4$ 的元素；
   - $\dots$
   查找时从最高层开始向右跳跃，遇到比目标键大的节点时向下一层跌落，呈现如同二叉平衡树的搜索轨迹，平均耗时仅为 $O(\log n)$。

2. **几何分布随机层数生成**：
   跳跃表通过**抛硬币（Coin Tossing）**维持平衡，完全规避了红黑树极其繁琐复杂的旋转调整：
   - 初始层数 `lvl = 1`；
   - 循环抛硬币：若随机位为 1 且 `lvl < MAX_LEVEL`，层数递增 1；若为 0 则停止循环。
   节点拥有层数 $k$ 的概率严格等于 $(1/2)^{k-1}$。

3. **插入与前驱更新数组（`update`）**：
   在插入键值对时：
   维护一个大小为 `MAX_LEVEL` 的指针数组 `update`。
   从当前最高层自顶向下查找目标键的前驱节点，记录每一层在“跌落”时所在的节点指针至 `update[i]`。
   - 若在 0 层找到已存在的键，原地更新 `val`；
   - 若不存在，随机生成新节点层数 `new_lvl`，分配含柔性数组的节点内存，依次将每一层的链表指针插入：
     `node->forward[i] = update[i]->forward[i];`
     `update[i]->forward[i] = node;`

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

#define SKIPLIST_MAX_LEVEL 16

typedef uint64_t (*RandomBits)(void *context);

typedef struct SkipNode SkipNode;
struct SkipNode {
    int64_t key;
    uint64_t val;
    SkipNode *forward[];
};

typedef struct {
    SkipNode *header;
    size_t level;
    size_t length;
    RandomBits random_bits;
    void *context;
} SkipList;

static SkipNode *create_node(size_t level, int64_t key, uint64_t val) {
    SkipNode *node = malloc(sizeof(SkipNode) + level * sizeof(SkipNode *));
    if (node == NULL) {
        return NULL;
    }
    node->key = key;
    node->val = val;
    for (size_t i = 0; i < level; ++i) {
        node->forward[i] = NULL;
    }
    return node;
}

static size_t random_level(RandomBits random_bits, void *context) {
    size_t lvl = 1;
    while (lvl < SKIPLIST_MAX_LEVEL) {
        uint64_t bit = random_bits(context) & 1;
        if (bit == 1) {
            lvl++;
        } else {
            break;
        }
    }
    return lvl;
}

SkipList *skiplist_create(RandomBits random_bits, void *context) {
    if (random_bits == NULL) {
        return NULL;
    }

    SkipList *list = malloc(sizeof *list);
    if (list == NULL) {
        return NULL;
    }

    list->header = create_node(SKIPLIST_MAX_LEVEL, 0, 0);
    if (list->header == NULL) {
        free(list);
        return NULL;
    }

    list->level = 1;
    list->length = 0;
    list->random_bits = random_bits;
    list->context = context;
    return list;
}

bool skiplist_search(const SkipList *list, int64_t key, uint64_t *out_val) {
    if (list == NULL || out_val == NULL) {
        return false;
    }

    const SkipNode *curr = list->header;
    for (ptrdiff_t i = (ptrdiff_t)list->level - 1; i >= 0; --i) {
        while (curr->forward[i] != NULL && curr->forward[i]->key < key) {
            curr = curr->forward[i];
        }
    }

    curr = curr->forward[0];
    if (curr != NULL && curr->key == key) {
        *out_val = curr->val;
        return true;
    }

    return false;
}

bool skiplist_insert(SkipList *list, int64_t key, uint64_t val) {
    if (list == NULL) {
        return false;
    }

    SkipNode *update[SKIPLIST_MAX_LEVEL];
    SkipNode *curr = list->header;

    for (ptrdiff_t i = (ptrdiff_t)list->level - 1; i >= 0; --i) {
        while (curr->forward[i] != NULL && curr->forward[i]->key < key) {
            curr = curr->forward[i];
        }
        update[i] = curr;
    }

    curr = curr->forward[0];
    if (curr != NULL && curr->key == key) {
        curr->val = val; // 键已存在，直接覆盖更新
        return true;
    }

    size_t new_lvl = random_level(list->random_bits, list->context);
    if (new_lvl > list->level) {
        for (size_t i = list->level; i < new_lvl; ++i) {
            update[i] = list->header;
        }
        list->level = new_lvl;
    }

    SkipNode *new_node = create_node(new_lvl, key, val);
    if (new_node == NULL) {
        return false;
    }

    for (size_t i = 0; i < new_lvl; ++i) {
        new_node->forward[i] = update[i]->forward[i];
        update[i]->forward[i] = new_node;
    }

    list->length++;
    return true;
}

void skiplist_destroy(SkipList *list) {
    if (list == NULL) {
        return;
    }
    SkipNode *curr = list->header->forward[0];
    while (curr != NULL) {
        SkipNode *next = curr->forward[0];
        free(curr);
        curr = next;
    }
    free(list->header);
    free(list);
}

static uint64_t test_prng(void *ctx) {
    uint64_t *s = (uint64_t *)ctx;
    *s = (*s * 6364136223846793005ULL) + 1442695040888963407ULL;
    return *s;
}

int main(void) {
    uint64_t seed = 8888ULL;
    SkipList *sl = skiplist_create(test_prng, &seed);
    assert(sl != NULL);

    uint64_t val = 0;

    // 查找空表
    assert(!skiplist_search(sl, 10, &val));

    // 插入若干乱序键值对
    assert(skiplist_insert(sl, 30, 300));
    assert(skiplist_insert(sl, 10, 100));
    assert(skiplist_insert(sl, 20, 200));
    assert(skiplist_insert(sl, 50, 500));
    assert(skiplist_insert(sl, 40, 400));
    assert(sl->length == 5);

    // 查找验证
    assert(skiplist_search(sl, 10, &val) && val == 100);
    assert(skiplist_search(sl, 20, &val) && val == 200);
    assert(skiplist_search(sl, 30, &val) && val == 300);
    assert(skiplist_search(sl, 40, &val) && val == 400);
    assert(skiplist_search(sl, 50, &val) && val == 500);

    // 不存在的键
    assert(!skiplist_search(sl, 99, &val));

    // 更新已存在的键
    assert(skiplist_insert(sl, 20, 9999));
    assert(skiplist_search(sl, 20, &val) && val == 9999);
    assert(sl->length == 5);

    skiplist_destroy(sl);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：
  - **查找**：期望 $O(\log n)$，最坏情况（概率趋近于 0）退化为 $O(n)$；
  - **插入**：期望 $O(\log n)$；
  - **删除**：期望 $O(\log n)$。
- **空间复杂度**：期望 $O(n)$。每个节点的期望层数为 $\sum_{i=1}^\infty i (1/2)^i = 2$，平均每个节点仅需 2 个前向指针。

## 延伸思考 / 模式总结

跳跃表被 William Pugh 发明为“平衡树的一种简单替代品”。在多线程并发场景下（如 Redis 内部的 Sorted Set、LevelDB 的 MemTable），跳跃表的优势尤为显著：因为跳跃表的插入操作只需要局部修改前后节点的指针，无需像红黑树那样进行可能蔓延至整棵树的全局染色和再平衡旋转，并发锁粒度极其细小。

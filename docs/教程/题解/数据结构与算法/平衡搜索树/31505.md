# 31505 - 红黑树插入与三种修复情形

## 题目描述

实现红黑树（Red-Black Tree）的完整插入算法 `rb_insert`，采用哨兵叶结点 `nil` 简化边界条件，并在新结点插入后通过颜色翻转与树旋转消除红红冲突，完整实现三种修复情形。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct RbTree RbTree;

RbTree *rb_create(void);
void rb_destroy(RbTree *tree);
bool rb_insert(
    RbTree *tree,
    int64_t key,
    bool *out_inserted
);
bool rb_contains(const RbTree *tree, int64_t key);
bool rb_validate(const RbTree *tree);
```

要求：

1. 新结点以**红色**插入树底。若父结点为黑色，插入直接完成；若父结点为红色，触发修复循环。
2. 完整处理叔结点为红色（情形 1）以及叔结点为黑色（内侧情形 2、外侧情形 3）的再平衡与重染色逻辑及其对称版本。
3. 循环结束后强制将根结点重置为黑色。
4. 使用全局统一分配的黑色哨兵结点 `nil` 替代传统的 `NULL` 指针，彻底消除繁琐的判空特判。
5. 每次插入后均需满足红黑树的全部 5 条定义性质。

## 解题思路

### 1. 为什么新结点必须染成红色？

若新结点染为黑色，则该分支的黑高必然比其他路径多 1，瞬间破坏全局性质 5（黑高一致）。黑高失衡非常难以局部修复。
而将新结点染为红色，**绝不会改变任何路径上的黑高**，唯一可能被打破的是性质 4（红父无红子）。这使得修复只需聚焦于局部的颜色冲突消除。

### 2. 插入修复的三种情形分步推演

设当前插入的红结点为 $z$，其父结点 $p = z.\text{parent}$ 也是红色，祖父结点为 $g = p.\text{parent}$（$g$ 必然为黑色）。以 $p$ 为 $g$ 的左孩子为例（右孩子对称）：

设叔结点为 $y = g.\text{right}$：

- **情形 1：叔结点 $y$ 为红色**
  - **操作**：将父结点 $p$ 和叔结点 $y$ 染为黑色，将祖父结点 $g$ 染为红色。
  - **效果**：将多余的红色从下一层向上“挤出”到祖父 $g$ 处。令当前指针 $z = g$，继续向上循环回溯检查 $g$ 是否与其父结点再次产生红红冲突。
- **情形 2：叔结点 $y$ 为黑色，且 $z$ 为右孩子（内侧折线，LR 型）**
  - **操作**：令 $z = p$，对 $p$ 执行一次**左旋**。
  - **效果**：将折线形态拉直，平滑转化为情形 3，并不改变任何黑高。
- **情形 3：叔结点 $y$ 为黑色，且 $z$ 为左孩子（外侧直线，LL 型）**
  - **操作**：将父结点 $p$ 染为黑色，将祖父结点 $g$ 染为红色，随后对祖父 $g$ 执行一次**右旋**。
  - **效果**：旋转后新子树根为黑色 $p$，其左孩子为红色 $z$，右孩子为红色 $g$。此时红红冲突彻底消除，整棵子树的黑高与插入前完全一致，循环成功终止！

最后，无论根结点在此过程中是否被染成红色，均无条件将 `root->color` 设为 `RB_BLACK`。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef enum {
    RB_RED = 0,
    RB_BLACK = 1
} RbColor;

typedef struct RbNode RbNode;

struct RbNode {
    int64_t key;
    RbColor color;
    RbNode *left;
    RbNode *right;
    RbNode *parent;
};

typedef struct RbTree {
    RbNode *root;
    RbNode *nil;
    size_t size;
} RbTree;

RbTree *rb_create(void) {
    RbTree *tree = (RbTree *)malloc(sizeof(RbTree));
    if (tree == NULL) {
        return NULL;
    }
    tree->nil = (RbNode *)malloc(sizeof(RbNode));
    if (tree->nil == NULL) {
        free(tree);
        return NULL;
    }
    tree->nil->color = RB_BLACK;
    tree->nil->left = tree->nil;
    tree->nil->right = tree->nil;
    tree->nil->parent = tree->nil;
    tree->nil->key = 0;

    tree->root = tree->nil;
    tree->size = 0;
    return tree;
}

static void free_nodes(RbTree *tree, RbNode *n) {
    if (n == tree->nil) {
        return;
    }
    free_nodes(tree, n->left);
    free_nodes(tree, n->right);
    free(n);
}

void rb_destroy(RbTree *tree) {
    if (tree == NULL) {
        return;
    }
    free_nodes(tree, tree->root);
    free(tree->nil);
    free(tree);
}

static void left_rotate(RbTree *tree, RbNode *x) {
    RbNode *y = x->right;
    x->right = y->left;
    if (y->left != tree->nil) {
        y->left->parent = x;
    }
    y->parent = x->parent;
    if (x->parent == tree->nil) {
        tree->root = y;
    } else if (x == x->parent->left) {
        x->parent->left = y;
    } else {
        x->parent->right = y;
    }
    y->left = x;
    x->parent = y;
}

static void right_rotate(RbTree *tree, RbNode *y) {
    RbNode *x = y->left;
    y->left = x->right;
    if (x->right != tree->nil) {
        x->right->parent = y;
    }
    x->parent = y->parent;
    if (y->parent == tree->nil) {
        tree->root = x;
    } else if (y == y->parent->left) {
        y->parent->left = x;
    } else {
        y->parent->right = x;
    }
    x->right = y;
    y->parent = x;
}

static void rb_insert_fixup(RbTree *tree, RbNode *z) {
    while (z->parent->color == RB_RED) {
        if (z->parent == z->parent->parent->left) {
            RbNode *y = z->parent->parent->right;
            if (y->color == RB_RED) {
                /* 情形 1：叔结点为红色 -> 翻转颜色，向上回溯 */
                z->parent->color = RB_BLACK;
                y->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                z = z->parent->parent;
            } else {
                /* 情形 2：叔结点为黑色且 z 为右孩子 -> 左旋化为情形 3 */
                if (z == z->parent->right) {
                    z = z->parent;
                    left_rotate(tree, z);
                }
                /* 情形 3：叔结点为黑色且 z 为左孩子 -> 变色并右旋消除冲突 */
                z->parent->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                right_rotate(tree, z->parent->parent);
            }
        } else {
            /* 对称分支：父结点为祖父结点的右孩子 */
            RbNode *y = z->parent->parent->left;
            if (y->color == RB_RED) {
                z->parent->color = RB_BLACK;
                y->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                z = z->parent->parent;
            } else {
                if (z == z->parent->left) {
                    z = z->parent;
                    right_rotate(tree, z);
                }
                z->parent->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                left_rotate(tree, z->parent->parent);
            }
        }
    }
    tree->root->color = RB_BLACK;
}

bool rb_insert(RbTree *tree, int64_t key, bool *out_inserted) {
    if (tree == NULL || out_inserted == NULL || tree->size == SIZE_MAX) {
        return false;
    }
    RbNode *y = tree->nil;
    RbNode *x = tree->root;
    while (x != tree->nil) {
        y = x;
        if (key == x->key) {
            *out_inserted = false;
            return true;
        }
        x = (key < x->key) ? x->left : x->right;
    }

    RbNode *z = (RbNode *)malloc(sizeof(RbNode));
    if (z == NULL) {
        return false;
    }
    z->key = key;
    z->color = RB_RED;
    z->left = tree->nil;
    z->right = tree->nil;
    z->parent = y;

    if (y == tree->nil) {
        tree->root = z;
    } else if (z->key < y->key) {
        y->left = z;
    } else {
        y->right = z;
    }

    rb_insert_fixup(tree, z);
    tree->size++;
    *out_inserted = true;
    return true;
}

bool rb_contains(const RbTree *tree, int64_t key) {
    if (tree == NULL) {
        return false;
    }
    const RbNode *x = tree->root;
    while (x != tree->nil) {
        if (key == x->key) {
            return true;
        }
        x = (key < x->key) ? x->left : x->right;
    }
    return false;
}

static bool validate_helper(
    const RbTree *tree,
    const RbNode *n,
    const RbNode *p,
    bool has_low,
    int64_t low,
    bool has_high,
    int64_t high,
    size_t *out_bh
) {
    if (n == tree->nil) {
        *out_bh = 0;
        return true;
    }
    if ((has_low && n->key <= low) || (has_high && n->key >= high)) {
        return false;
    }
    if (n->color == RB_RED && p != NULL && p->color == RB_RED) {
        return false;
    }
    size_t lbh = 0, rbh = 0;
    if (!validate_helper(tree, n->left, n, has_low, low, true, n->key, &lbh) ||
        !validate_helper(tree, n->right, n, true, n->key, has_high, high, &rbh)) {
        return false;
    }
    if (lbh != rbh) {
        return false;
    }
    *out_bh = lbh + (n->color == RB_BLACK ? 1 : 0);
    return true;
}

bool rb_validate(const RbTree *tree) {
    if (tree == NULL) {
        return false;
    }
    if (tree->root == tree->nil) {
        return tree->size == 0;
    }
    if (tree->root->color != RB_BLACK) {
        return false;
    }
    size_t bh = 0;
    return validate_helper(tree, tree->root, NULL, false, 0, false, 0, &bh);
}

int main(void) {
    RbTree *tree = rb_create();
    assert(tree != NULL);
    assert(rb_validate(tree));

    bool ins = false;
    /* 顺序插入 100 个递增整数，极端考验旋转与变色性能 */
    for (int64_t i = 1; i <= 100; ++i) {
        assert(rb_insert(tree, i, &ins) && ins);
        assert(rb_validate(tree));
    }
    assert(tree->size == 100);

    /* 重复插入测试 */
    assert(rb_insert(tree, 50, &ins) && !ins);
    assert(tree->size == 100);
    assert(rb_validate(tree));

    /* 成员查找测试 */
    for (int64_t i = 1; i <= 100; ++i) {
        assert(rb_contains(tree, i));
    }
    assert(!rb_contains(tree, 0));
    assert(!rb_contains(tree, 101));

    rb_destroy(tree);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：插入搜索耗时 $O(\log n)$；修复循环中，情形 1 虽向上爬升至多 $O(\log n)$ 层，但仅涉及变色；情形 2 与情形 3 执行至多 2 次旋转后便必定终止。故单次插入时间复杂度严格为 $O(\log n)$，且旋转次数至多为 2。
- **空间复杂度**：由于采用显式父指针迭代循环，无需递归调用帧，额外空间复杂度为严格的 $O(1)$。

## 延伸思考 / 模式总结

红黑树的插入修复过程展示了“通过局部的颜色提升将问题向顶层转移，直到遇到非冲突状态或通过常数次局部几何旋转将其吸收”的高级自平衡哲学。相比 AVL 树，红黑树单次插入的旋转上限严格为 2 次，极大地减少了对内存拓扑的破坏与写屏障开销。

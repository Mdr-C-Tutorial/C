# 31506 - 红黑树删除与双黑节点消除

## 题目描述

实现红黑树结点的动态删除操作 `rb_remove`，处理被删结点为黑色时所引发的黑高亏损（“双黑”状态，Double Black），并完整消除双黑结点的四种情形。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct RbTree RbTree;

bool rb_remove(
    RbTree *tree,
    int64_t key,
    bool *out_removed
);
```

要求：

1. 待删结点拥有两个孩子时，用中序后继替代，并将问题转换为至多只有一个孩子结点的物理删除。
2. 当实际被删除的结点为黑色时，会导致其子树的黑高减少 1，将“额外的一层黑色”赋予替代结点 $x$，形成“双黑”结点。
3. 完整实现双黑消除的四种情形：
   - **情形 1**：兄弟结点 $w$ 为红色；
   - **情形 2**：兄弟结点 $w$ 为黑色，且其两个孩子均为黑色；
   - **情形 3**：兄弟结点 $w$ 为黑色，其近端孩子为红色，远端孩子为黑色；
   - **情形 4**：兄弟结点 $w$ 为黑色，其远端孩子为红色。
4. 修复完成后将 $x$ 重新染为单一黑色，且全树必须在每次删除后均通过红黑树合法性验证。

## 解题思路

### 1. 双黑概念的引入

若被删结点的原始颜色为红色，树的黑高并未发生任何改变，直接移除即可。
但若被删结点的原始颜色为**黑色**，经过该结点的所有简单路径上的黑高都减少了 1。为了在逻辑上维持性质 5（黑高一致），我们假设接替该位置的结点 $x$ 携带了**两重黑色**（Double Black）：
- 若 $x$ 原本是红色的，只需将其染成普通黑色，即可自然补偿丢失的黑高，修复完成；
- 若 $x$ 原本就是黑色的（包括黑色的哨兵结点 `nil`），则 $x$ 成为真正的双黑结点，必须通过颜色转移或旋转消除多余的黑色。

### 2. 双黑消除四种情形推演（以 $x$ 为左孩子为例）

设 $x$ 的兄弟结点为 $w = x.\text{parent}.\text{right}$：

- **情形 1：兄弟 $w$ 为红色**
  - **推演**：$w$ 必须有两个黑色孩子，且父结点 $p$ 必为黑色。
  - **操作**：将 $w$ 染黑，$p$ 染红，对 $p$ 执行**左旋**，更新新兄弟 $w = p.\text{right}$。
  - **转化**：成功将情形转换为兄弟为黑色的情形 2、3 或 4。
- **情形 2：兄弟 $w$ 为黑色，且 $w$ 的两个孩子全为黑色**
  - **推演**：兄弟子树无法提供红结点借调。
  - **操作**：将 $w$ 染为红色，将双黑属性向父结点提升一层：$x = x.\text{parent}$。
  - **转化**：若新 $x$ 为红色，直接退出循环并在末尾染黑即补偿成功；若新 $x$ 原本为黑，则继续向上消除双黑。
- **情形 3：兄弟 $w$ 为黑色，近端孩子（$w.\text{left}$）为红色，远端孩子（$w.\text{right}$）为黑色**
  - **推演**：红结点位于内侧，需调整到外侧。
  - **操作**：将 $w.\text{left}$ 染黑，$w$ 染红，对 $w$ 执行**右旋**，更新 $w = p.\text{right}$。
  - **转化**：平滑转换为情形 4。
- **情形 4：兄弟 $w$ 为黑色，远端孩子（$w.\text{right}$）为红色**
  - **推演**：外侧红结点可以直接用于补偿黑高。
  - **操作**：将 $w$ 染为父结点当前的颜色，将父结点 $p$ 染黑，将 $w.\text{right}$ 染黑，最后对 $p$ 执行一次**左旋**，置 $x = \text{root}$。
  - **转化**：双黑彻底消除，循环直接终止！

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef enum {
    RB_RED = 0,
    RB_BLACK = 1
} RbColor;

typedef struct RbNode RbNode;

struct RbNode {
    int64_t key;
    RbColor color;
    RbNode *left;
    RbNode *right;
    RbNode *parent;
};

typedef struct RbTree {
    RbNode *root;
    RbNode *nil;
    size_t size;
} RbTree;

RbTree *rb_create(void) {
    RbTree *tree = (RbTree *)malloc(sizeof(RbTree));
    if (tree == NULL) {
        return NULL;
    }
    tree->nil = (RbNode *)malloc(sizeof(RbNode));
    if (tree->nil == NULL) {
        free(tree);
        return NULL;
    }
    tree->nil->color = RB_BLACK;
    tree->nil->left = tree->nil;
    tree->nil->right = tree->nil;
    tree->nil->parent = tree->nil;
    tree->nil->key = 0;

    tree->root = tree->nil;
    tree->size = 0;
    return tree;
}

static void free_nodes(RbTree *tree, RbNode *n) {
    if (n == tree->nil) {
        return;
    }
    free_nodes(tree, n->left);
    free_nodes(tree, n->right);
    free(n);
}

void rb_destroy(RbTree *tree) {
    if (tree == NULL) {
        return;
    }
    free_nodes(tree, tree->root);
    free(tree->nil);
    free(tree);
}

static void left_rotate(RbTree *tree, RbNode *x) {
    RbNode *y = x->right;
    x->right = y->left;
    if (y->left != tree->nil) {
        y->left->parent = x;
    }
    y->parent = x->parent;
    if (x->parent == tree->nil) {
        tree->root = y;
    } else if (x == x->parent->left) {
        x->parent->left = y;
    } else {
        x->parent->right = y;
    }
    y->left = x;
    x->parent = y;
}

static void right_rotate(RbTree *tree, RbNode *y) {
    RbNode *x = y->left;
    y->left = x->right;
    if (x->right != tree->nil) {
        x->right->parent = y;
    }
    x->parent = y->parent;
    if (y->parent == tree->nil) {
        tree->root = x;
    } else if (y == y->parent->left) {
        y->parent->left = x;
    } else {
        y->parent->right = x;
    }
    x->right = y;
    y->parent = x;
}

static void rb_insert_fixup(RbTree *tree, RbNode *z) {
    while (z->parent->color == RB_RED) {
        if (z->parent == z->parent->parent->left) {
            RbNode *y = z->parent->parent->right;
            if (y->color == RB_RED) {
                z->parent->color = RB_BLACK;
                y->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                z = z->parent->parent;
            } else {
                if (z == z->parent->right) {
                    z = z->parent;
                    left_rotate(tree, z);
                }
                z->parent->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                right_rotate(tree, z->parent->parent);
            }
        } else {
            RbNode *y = z->parent->parent->left;
            if (y->color == RB_RED) {
                z->parent->color = RB_BLACK;
                y->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                z = z->parent->parent;
            } else {
                if (z == z->parent->left) {
                    z = z->parent;
                    right_rotate(tree, z);
                }
                z->parent->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                left_rotate(tree, z->parent->parent);
            }
        }
    }
    tree->root->color = RB_BLACK;
}

bool rb_insert(RbTree *tree, int64_t key, bool *out_inserted) {
    if (tree == NULL || out_inserted == NULL || tree->size == SIZE_MAX) {
        return false;
    }
    RbNode *y = tree->nil;
    RbNode *x = tree->root;
    while (x != tree->nil) {
        y = x;
        if (key == x->key) {
            *out_inserted = false;
            return true;
        }
        x = (key < x->key) ? x->left : x->right;
    }

    RbNode *z = (RbNode *)malloc(sizeof(RbNode));
    if (z == NULL) {
        return false;
    }
    z->key = key;
    z->color = RB_RED;
    z->left = tree->nil;
    z->right = tree->nil;
    z->parent = y;

    if (y == tree->nil) {
        tree->root = z;
    } else if (z->key < y->key) {
        y->left = z;
    } else {
        y->right = z;
    }

    rb_insert_fixup(tree, z);
    tree->size++;
    *out_inserted = true;
    return true;
}

static void rb_transplant(RbTree *tree, RbNode *u, RbNode *v) {
    if (u->parent == tree->nil) {
        tree->root = v;
    } else if (u == u->parent->left) {
        u->parent->left = v;
    } else {
        u->parent->right = v;
    }
    v->parent = u->parent;
}

static RbNode *tree_minimum(RbTree *tree, RbNode *x) {
    while (x->left != tree->nil) {
        x = x->left;
    }
    return x;
}

static void rb_delete_fixup(RbTree *tree, RbNode *x) {
    while (x != tree->root && x->color == RB_BLACK) {
        if (x == x->parent->left) {
            RbNode *w = x->parent->right;
            if (w->color == RB_RED) {
                /* 情形 1：兄弟结点为红 */
                w->color = RB_BLACK;
                x->parent->color = RB_RED;
                left_rotate(tree, x->parent);
                w = x->parent->right;
            }
            if (w->left->color == RB_BLACK && w->right->color == RB_BLACK) {
                /* 情形 2：兄弟结点为黑且其两个孩子都为黑 */
                w->color = RB_RED;
                x = x->parent;
            } else {
                if (w->right->color == RB_BLACK) {
                    /* 情形 3：兄弟结点为黑，近端孩子红、远端孩子黑 */
                    w->left->color = RB_BLACK;
                    w->color = RB_RED;
                    right_rotate(tree, w);
                    w = x->parent->right;
                }
                /* 情形 4：兄弟结点为黑，远端孩子红 */
                w->color = x->parent->color;
                x->parent->color = RB_BLACK;
                w->right->color = RB_BLACK;
                left_rotate(tree, x->parent);
                x = tree->root;
            }
        } else {
            /* 对称分支 */
            RbNode *w = x->parent->left;
            if (w->color == RB_RED) {
                w->color = RB_BLACK;
                x->parent->color = RB_RED;
                right_rotate(tree, x->parent);
                w = x->parent->left;
            }
            if (w->right->color == RB_BLACK && w->left->color == RB_BLACK) {
                w->color = RB_RED;
                x = x->parent;
            } else {
                if (w->left->color == RB_BLACK) {
                    w->right->color = RB_BLACK;
                    w->color = RB_RED;
                    left_rotate(tree, w);
                    w = x->parent->left;
                }
                w->color = x->parent->color;
                x->parent->color = RB_BLACK;
                w->left->color = RB_BLACK;
                right_rotate(tree, x->parent);
                x = tree->root;
            }
        }
    }
    x->color = RB_BLACK;
}

bool rb_remove(RbTree *tree, int64_t key, bool *out_removed) {
    if (tree == NULL || out_removed == NULL) {
        return false;
    }
    RbNode *z = tree->root;
    while (z != tree->nil) {
        if (key == z->key) {
            break;
        }
        z = (key < z->key) ? z->left : z->right;
    }
    if (z == tree->nil) {
        *out_removed = false;
        return true;
    }

    RbNode *y = z;
    RbNode *x = NULL;
    RbColor y_original_color = y->color;

    if (z->left == tree->nil) {
        x = z->right;
        rb_transplant(tree, z, z->right);
    } else if (z->right == tree->nil) {
        x = z->left;
        rb_transplant(tree, z, z->left);
    } else {
        y = tree_minimum(tree, z->right);
        y_original_color = y->color;
        x = y->right;
        if (y->parent == z) {
            x->parent = y;
        } else {
            rb_transplant(tree, y, y->right);
            y->right = z->right;
            y->right->parent = y;
        }
        rb_transplant(tree, z, y);
        y->left = z->left;
        y->left->parent = y;
        y->color = z->color;
    }

    if (y_original_color == RB_BLACK) {
        rb_delete_fixup(tree, x);
    }

    free(z);
    tree->size--;
    *out_removed = true;
    return true;
}

static bool validate_helper(
    const RbTree *tree,
    const RbNode *n,
    const RbNode *p,
    bool has_low,
    int64_t low,
    bool has_high,
    int64_t high,
    size_t *out_bh
) {
    if (n == tree->nil) {
        *out_bh = 0;
        return true;
    }
    if ((has_low && n->key <= low) || (has_high && n->key >= high)) {
        return false;
    }
    if (n->color == RB_RED && p != NULL && p->color == RB_RED) {
        return false;
    }
    size_t lbh = 0, rbh = 0;
    if (!validate_helper(tree, n->left, n, has_low, low, true, n->key, &lbh) ||
        !validate_helper(tree, n->right, n, true, n->key, has_high, high, &rbh)) {
        return false;
    }
    if (lbh != rbh) {
        return false;
    }
    *out_bh = lbh + (n->color == RB_BLACK ? 1 : 0);
    return true;
}

bool rb_validate(const RbTree *tree) {
    if (tree == NULL) {
        return false;
    }
    if (tree->root == tree->nil) {
        return tree->size == 0;
    }
    if (tree->root->color != RB_BLACK) {
        return false;
    }
    size_t bh = 0;
    return validate_helper(tree, tree->root, NULL, false, 0, false, 0, &bh);
}

int main(void) {
    RbTree *tree = rb_create();
    assert(tree != NULL);

    bool ins = false;
    for (int64_t i = 1; i <= 60; ++i) {
        assert(rb_insert(tree, i, &ins) && ins);
        assert(rb_validate(tree));
    }
    assert(tree->size == 60);

    bool rem = false;
    /* 阶段 1：交替删除偶数结点，触发大量双黑消除情形 */
    for (int64_t i = 2; i <= 60; i += 2) {
        assert(rb_remove(tree, i, &rem) && rem);
        assert(rb_validate(tree));
    }
    assert(tree->size == 30);

    /* 阶段 2：删除奇数结点直到树清空 */
    for (int64_t i = 1; i <= 60; i += 2) {
        assert(rb_remove(tree, i, &rem) && rem);
        assert(rb_validate(tree));
    }
    assert(tree->size == 0);
    assert(tree->root == tree->nil);

    rb_destroy(tree);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：删除搜索定位耗时 $O(\log n)$；双黑修复循环中，情形 2 向上回溯至多 $O(\log n)$ 层（仅涉及常数次指针移动与变色），而情形 1、3、4 累计最多执行 3 次旋转便彻底终止。故单次删除的总时间复杂度为严格的 $O(\log n)$，且旋转次数至多为 3。
- **空间复杂度**：由于依靠父指针迭代自底向上遍历，额外空间复杂度为严格的 $O(1)$。

## 延伸思考 / 模式总结

红黑树删除算法是自平衡二叉搜索树中最精妙的设计之一。它将看似复杂的路径黑高亏损抽象为单一的“双黑标记”，并通过四种穷尽的兄弟构型将其逐步向外侧推导吸收。至多 3 次旋转的数学上限，使红黑树在现代高性能操作系统核心与存储索引引擎中长盛不衰。

# 31503 - AVL 树节点删除与回溯再平衡传递

## 题目描述

在 AVL 树中实现结点的删除操作 `avl_remove`，正确处理无子结点、单子结点与双子结点情形，并保证在删除引起的局部树高缩减沿祖先路径向上回溯时，持续传递并完成可能的多重再平衡旋转。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct AvlTree AvlTree;

bool avl_remove(
    AvlTree *tree,
    int64_t key,
    bool *out_removed
);
```

要求：

1. 待删结点的结构处理：无孩子时直接释放；单孩子时由唯一孩子取代位置；双孩子时寻找右子树中序后继（最小值）替换其键，再递归在右子树中删除该后继。
2. **回溯再平衡的向上传递性**：与插入不同，删除引起的子树高度降低可能触发多次再平衡，甚至一直向上波及至根结点。要求在回溯函数栈退栈时，对每一个祖先均调用 `rebalance`。
3. 当某结点的失衡类型满足 $\text{bf}(z) > 1$ 且其左孩子的 $\text{bf}(z.\text{left}) = 0$ 时，单次右旋即可修复该结点，必须正确覆盖此类等高子树边界。
4. 正确维护 `tree->size`，并在每次删除后均满足 AVL 树全部不变式。

## 解题思路

### 1. 为什么删除可能需要 $O(\log n)$ 次旋转

在 AVL 树中插入新结点时，最深失衡结点的再平衡旋转会使该局部子树的高度恢复为插入前的数值，因而在更高层祖先看来，子树高度并未发生改变，旋转立即终止。

然而在删除操作中，**对某个祖先执行旋转再平衡后，该子树的高度不仅没有恢复，反而可能会进一步减少 1**！这就导致该子树的父结点可能接着失衡，再平衡旋转需要沿着整条祖先回溯路径逐层向上检查并执行，最坏情况下可能发生 $O(\log n)$ 次树旋转。

### 2. 子结点平衡因子为 0 的特殊情形

在标准插入再平衡中，由于插入总是使某一侧高度增加 1，因此较高孩子的平衡因子必然为 $+1$ 或 $-1$。
但在删除中，较矮一侧的缺失会导致较高一侧失衡。此时较高孩子的平衡因子可能恰好为 $0$：
- 设 $\text{bf}(z) = +2$，且其左孩子满足 $\text{bf}(z.\text{left}) = 0$。
- 此时执行一次单右旋即可令整棵子树合法（旋转后 $x$ 的平衡因子为 $-1$，$y$ 的平衡因子为 $+1$），无需也不应执行左旋预处理。
- 因此判定条件必须放宽为：
  - 左失衡（$\text{bf} > 1$）：若 $\text{bf}(z.\text{left}) < 0$ 执行 LR 双旋，否则（$\ge 0$）执行 LL 单旋；
  - 右失衡（$\text{bf} < -1$）：若 $\text{bf}(z.\text{right}) > 0$ 执行 RL 双旋，否则（$\le 0$）执行 RR 单旋。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct AvlTree AvlTree;
typedef struct AvlNode AvlNode;

struct AvlNode {
    int64_t key;
    size_t height;
    AvlNode *left;
    AvlNode *right;
};

struct AvlTree {
    AvlNode *root;
    size_t size;
};

static size_t node_height(const AvlNode *node) {
    return node == NULL ? 0 : node->height;
}

static size_t max_size(size_t a, size_t b) {
    return a > b ? a : b;
}

static void update_height(AvlNode *node) {
    size_t hl = node_height(node->left);
    size_t hr = node_height(node->right);
    node->height = 1 + max_size(hl, hr);
}

static int64_t balance_factor(const AvlNode *node) {
    if (node == NULL) {
        return 0;
    }
    return (int64_t)node_height(node->left) - (int64_t)node_height(node->right);
}

static AvlNode *rotate_right(AvlNode *y) {
    AvlNode *x = y->left;
    AvlNode *t2 = x->right;
    x->right = y;
    y->left = t2;
    update_height(y);
    update_height(x);
    return x;
}

static AvlNode *rotate_left(AvlNode *x) {
    AvlNode *y = x->right;
    AvlNode *t2 = y->left;
    y->left = x;
    x->right = t2;
    update_height(x);
    update_height(y);
    return y;
}

static AvlNode *rebalance(AvlNode *node) {
    update_height(node);
    int64_t bf = balance_factor(node);

    if (bf > 1) {
        if (balance_factor(node->left) < 0) {
            node->left = rotate_left(node->left);
        }
        return rotate_right(node);
    }
    if (bf < -1) {
        if (balance_factor(node->right) > 0) {
            node->right = rotate_right(node->right);
        }
        return rotate_left(node);
    }
    return node;
}

AvlTree *avl_create(void) {
    AvlTree *tree = (AvlTree *)malloc(sizeof *tree);
    if (tree == NULL) {
        return NULL;
    }
    tree->root = NULL;
    tree->size = 0;
    return tree;
}

static void destroy_nodes(AvlNode *node) {
    if (node == NULL) {
        return;
    }
    destroy_nodes(node->left);
    destroy_nodes(node->right);
    free(node);
}

void avl_destroy(AvlTree *tree) {
    if (tree == NULL) {
        return;
    }
    destroy_nodes(tree->root);
    free(tree);
}

static AvlNode *insert_node(
    AvlNode *node,
    int64_t key,
    bool *out_inserted,
    bool *out_ok
) {
    if (node == NULL) {
        AvlNode *new_node = (AvlNode *)malloc(sizeof *new_node);
        if (new_node == NULL) {
            *out_ok = false;
            return NULL;
        }
        new_node->key = key;
        new_node->height = 1;
        new_node->left = NULL;
        new_node->right = NULL;
        *out_inserted = true;
        *out_ok = true;
        return new_node;
    }
    if (key == node->key) {
        *out_inserted = false;
        *out_ok = true;
        return node;
    }
    if (key < node->key) {
        node->left = insert_node(node->left, key, out_inserted, out_ok);
    } else {
        node->right = insert_node(node->right, key, out_inserted, out_ok);
    }
    if (!*out_ok) {
        return node;
    }
    return rebalance(node);
}

bool avl_insert(AvlTree *tree, int64_t key, bool *out_inserted) {
    if (tree == NULL || out_inserted == NULL || tree->size == SIZE_MAX) {
        return false;
    }
    bool inserted = false;
    bool ok = false;
    AvlNode *new_root = insert_node(tree->root, key, &inserted, &ok);
    if (!ok) {
        return false;
    }
    tree->root = new_root;
    if (inserted) {
        tree->size++;
    }
    *out_inserted = inserted;
    return true;
}

static AvlNode *remove_node_rec(AvlNode *root, int64_t key, bool *out_removed) {
    if (root == NULL) {
        *out_removed = false;
        return NULL;
    }
    if (key < root->key) {
        root->left = remove_node_rec(root->left, key, out_removed);
    } else if (key > root->key) {
        root->right = remove_node_rec(root->right, key, out_removed);
    } else {
        *out_removed = true;
        /* 无孩子或仅有单孩子情形 */
        if (root->left == NULL) {
            AvlNode *right_child = root->right;
            free(root);
            return right_child;
        }
        if (root->right == NULL) {
            AvlNode *left_child = root->left;
            free(root);
            return left_child;
        }
        /* 拥有左右双孩子：寻找右子树中序后继 */
        AvlNode *succ = root->right;
        while (succ->left != NULL) {
            succ = succ->left;
        }
        root->key = succ->key;
        bool succ_rem = false;
        root->right = remove_node_rec(root->right, succ->key, &succ_rem);
    }
    /* 退栈回溯，修复失衡 */
    return rebalance(root);
}

bool avl_remove(AvlTree *tree, int64_t key, bool *out_removed) {
    if (tree == NULL || out_removed == NULL) {
        return false;
    }
    bool removed = false;
    tree->root = remove_node_rec(tree->root, key, &removed);
    if (removed) {
        tree->size--;
    }
    *out_removed = removed;
    return true;
}

static bool validate_node(
    const AvlNode *node,
    bool has_lower,
    int64_t lower,
    bool has_upper,
    int64_t upper,
    size_t *out_height,
    size_t *out_count
) {
    if (node == NULL) {
        *out_height = 0;
        *out_count = 0;
        return true;
    }
    if ((has_lower && node->key <= lower) ||
        (has_upper && node->key >= upper)) {
        return false;
    }
    size_t hl = 0, hr = 0, cl = 0, cr = 0;
    if (!validate_node(node->left, has_lower, lower, true, node->key, &hl, &cl) ||
        !validate_node(node->right, true, node->key, has_upper, upper, &hr, &cr)) {
        return false;
    }
    if ((hl > hr && hl - hr > 1) || (hr > hl && hr - hl > 1)) {
        return false;
    }
    size_t expected_h = 1 + max_size(hl, hr);
    if (node->height != expected_h) {
        return false;
    }
    *out_height = expected_h;
    *out_count = cl + cr + 1;
    return true;
}

bool avl_validate(const AvlTree *tree) {
    if (tree == NULL) {
        return false;
    }
    size_t h = 0, c = 0;
    return validate_node(tree->root, false, 0, false, 0, &h, &c) && c == tree->size;
}

int main(void) {
    AvlTree *tree = avl_create();
    assert(tree != NULL);

    bool ins = false;
    for (int64_t i = 1; i <= 50; ++i) {
        assert(avl_insert(tree, i, &ins) && ins);
        assert(avl_validate(tree));
    }
    assert(tree->size == 50);

    bool rem = false;
    /* 测试删除叶结点、内部单孩子结点与双孩子根结点 */
    assert(avl_remove(tree, 50, &rem) && rem);
    assert(avl_validate(tree));

    assert(avl_remove(tree, 25, &rem) && rem);
    assert(avl_validate(tree));

    assert(avl_remove(tree, 1, &rem) && rem);
    assert(avl_validate(tree));

    /* 测试删除不存在的键 */
    assert(avl_remove(tree, 999, &rem) && !rem);
    assert(avl_validate(tree));

    /* 连续删除全部剩余结点 */
    for (int64_t i = 2; i <= 49; ++i) {
        if (i != 25) {
            assert(avl_remove(tree, i, &rem) && rem);
            assert(avl_validate(tree));
        }
    }
    assert(tree->size == 0);
    assert(tree->root == NULL);

    avl_destroy(tree);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：删除搜索定位耗时 $O(\log n)$；双孩子后继查找耗时至多 $O(\log n)$；沿回溯路径执行再平衡，每一层旋转耗时 $O(1)$，共计 $O(\log n)$ 层，因此单次删除的最坏时间复杂度依然为严格的 $O(\log n)$。
- **空间复杂度**：递归调用栈深度受树高制约，额外空间复杂度为 $O(\log n)$。

## 延伸思考 / 模式总结

AVL 树的删除操作是自平衡二叉搜索树中“连锁反应传播”的典型范例。对比红黑树的删除修复（红黑树删除至多触发 3 次旋转即可终止，其余均为变色），AVL 树在删除时旋转操作次数更多（最坏情况可达 $O(\log n)$ 次旋转）。这也解释了为什么在写操作极其密集的工业场景（如 Linux 内核调度器或标准库 `std::map`）中，更倾向于选择旋转代价常数更小的红黑树。

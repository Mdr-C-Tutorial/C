# 31502 - AVL 树插入与四种失衡再平衡

## 题目描述

实现自平衡二叉搜索树（AVL 树）的动态插入操作，在递归回溯路径上正确检测并修复四种失衡情形（LL、RR、LR、RL），并提供全树不变式验证器。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct AvlTree AvlTree;

AvlTree *avl_create(void);
void avl_destroy(AvlTree *tree);
bool avl_insert(
    AvlTree *tree,
    int64_t key,
    bool *out_inserted
);
bool avl_contains(const AvlTree *tree, int64_t key);
bool avl_validate(const AvlTree *tree);
```

要求：

1. 树中每个结点的平衡因子必须时刻落在 `[-1, 1]` 闭区间内；若插入引起失衡，必须通过单旋或双旋在回溯时修复。
2. 重复键插入时成功返回并设置 `*out_inserted = false`，树中不增加重复结点。
3. 内存分配失败或形参无效时，树保持原有拓扑不变，并安全返回 `false`。
4. `avl_validate` 必须递归校验全树每一处的搜索树顺序、结点记录高度的真实性，以及每个结点的平衡因子，不能仅校验根结点。
5. 插入与查找的时间复杂度为 $O(\log n)$，验证时间复杂度为 $O(n)$；`avl_destroy(NULL)` 安全。

## 解题思路

### 1. 四种失衡模式的判定与修复

当新叶子结点插入后，沿着递归调用链向上回溯。设首个被打破平衡的失衡祖先结点为 $z$：

| 失衡类型 | 判定条件 | 核心特征 | 修复手段 |
|---|---|---|---|
| **LL 型** | $\text{bf}(z) > 1$ 且 $\text{bf}(z.\text{left}) \ge 0$ | 插入发生在左孩子的左子树 | 对 $z$ 执行一次**右旋** |
| **RR 型** | $\text{bf}(z) < -1$ 且 $\text{bf}(z.\text{right}) \le 0$ | 插入发生在右孩子的右子树 | 对 $z$ 执行一次**左旋** |
| **LR 型** | $\text{bf}(z) > 1$ 且 $\text{bf}(z.\text{left}) < 0$ | 插入发生在左孩子的右子树 | 先对 $z.\text{left}$ **左旋**，再对 $z$ **右旋** |
| **RL 型** | $\text{bf}(z) < -1$ 且 $\text{bf}(z.\text{right}) > 0$ | 插入发生在右孩子的左子树 | 先对 $z.\text{right}$ **右旋**，再对 $z$ **左旋** |

在插入操作中，一旦在回溯途中对失衡祖先执行了一次单旋或双旋，该子树的高度便完全恢复到插入前的水平，**不会向上级祖先继续传递失衡**。因此单次插入最多触发常数次旋转。

### 2. 验证器的多重归纳设计

`avl_validate` 通过后序递归同时计算并校验三个属性：
1. **二叉搜索树开区间单调性**：利用祖先传递的开区间 `(lower, upper)` 校验当前键是否合法，严防整数溢出；
2. **高度计算一致性**：校验结点缓存的 `node->height` 是否严格等于 `1 + max(left_h, right_h)`；
3. **平衡因子合法性**：校验 $|left\_h - right\_h| \le 1$；
4. **结点计数守恒**：统计子树非空结点总和，最终确认其与 `tree->size` 严格相等。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct AvlTree AvlTree;
typedef struct AvlNode AvlNode;

struct AvlNode {
    int64_t key;
    size_t height;
    AvlNode *left;
    AvlNode *right;
};

struct AvlTree {
    AvlNode *root;
    size_t size;
};

static size_t node_height(const AvlNode *node) {
    return node == NULL ? 0 : node->height;
}

static size_t max_size(size_t a, size_t b) {
    return a > b ? a : b;
}

static void update_height(AvlNode *node) {
    size_t hl = node_height(node->left);
    size_t hr = node_height(node->right);
    node->height = 1 + max_size(hl, hr);
}

static int64_t balance_factor(const AvlNode *node) {
    if (node == NULL) {
        return 0;
    }
    return (int64_t)node_height(node->left) - (int64_t)node_height(node->right);
}

static AvlNode *rotate_right(AvlNode *y) {
    AvlNode *x = y->left;
    AvlNode *t2 = x->right;
    x->right = y;
    y->left = t2;
    update_height(y);
    update_height(x);
    return x;
}

static AvlNode *rotate_left(AvlNode *x) {
    AvlNode *y = x->right;
    AvlNode *t2 = y->left;
    y->left = x;
    x->right = t2;
    update_height(x);
    update_height(y);
    return y;
}

static AvlNode *rebalance(AvlNode *node) {
    update_height(node);
    int64_t bf = balance_factor(node);

    /* LL 型或 LR 型失衡 */
    if (bf > 1) {
        if (balance_factor(node->left) < 0) {
            /* LR 型：先对左孩子左旋，转换为 LL 型 */
            node->left = rotate_left(node->left);
        }
        /* 执行右旋 */
        return rotate_right(node);
    }
    /* RR 型或 RL 型失衡 */
    if (bf < -1) {
        if (balance_factor(node->right) > 0) {
            /* RL 型：先对右孩子右旋，转换为 RR 型 */
            node->right = rotate_right(node->right);
        }
        /* 执行左旋 */
        return rotate_left(node);
    }
    return node;
}

AvlTree *avl_create(void) {
    AvlTree *tree = (AvlTree *)malloc(sizeof *tree);
    if (tree == NULL) {
        return NULL;
    }
    tree->root = NULL;
    tree->size = 0;
    return tree;
}

static void destroy_nodes(AvlNode *node) {
    if (node == NULL) {
        return;
    }
    destroy_nodes(node->left);
    destroy_nodes(node->right);
    free(node);
}

void avl_destroy(AvlTree *tree) {
    if (tree == NULL) {
        return;
    }
    destroy_nodes(tree->root);
    free(tree);
}

static AvlNode *insert_node(
    AvlNode *node,
    int64_t key,
    bool *out_inserted,
    bool *out_ok
) {
    if (node == NULL) {
        AvlNode *new_node = (AvlNode *)malloc(sizeof *new_node);
        if (new_node == NULL) {
            *out_ok = false;
            return NULL;
        }
        new_node->key = key;
        new_node->height = 1;
        new_node->left = NULL;
        new_node->right = NULL;
        *out_inserted = true;
        *out_ok = true;
        return new_node;
    }
    if (key == node->key) {
        *out_inserted = false;
        *out_ok = true;
        return node;
    }
    if (key < node->key) {
        node->left = insert_node(node->left, key, out_inserted, out_ok);
    } else {
        node->right = insert_node(node->right, key, out_inserted, out_ok);
    }
    if (!*out_ok) {
        return node;
    }
    return rebalance(node);
}

bool avl_insert(
    AvlTree *tree,
    int64_t key,
    bool *out_inserted
) {
    if (tree == NULL || out_inserted == NULL || tree->size == SIZE_MAX) {
        return false;
    }
    bool inserted = false;
    bool ok = false;
    AvlNode *new_root = insert_node(tree->root, key, &inserted, &ok);
    if (!ok) {
        return false;
    }
    tree->root = new_root;
    if (inserted) {
        tree->size++;
    }
    *out_inserted = inserted;
    return true;
}

bool avl_contains(const AvlTree *tree, int64_t key) {
    if (tree == NULL) {
        return false;
    }
    const AvlNode *curr = tree->root;
    while (curr != NULL) {
        if (key == curr->key) {
            return true;
        }
        curr = key < curr->key ? curr->left : curr->right;
    }
    return false;
}

static bool validate_node(
    const AvlNode *node,
    bool has_lower,
    int64_t lower,
    bool has_upper,
    int64_t upper,
    size_t *out_height,
    size_t *out_count
) {
    if (node == NULL) {
        *out_height = 0;
        *out_count = 0;
        return true;
    }
    if ((has_lower && node->key <= lower) ||
        (has_upper && node->key >= upper)) {
        return false;
    }
    size_t hl = 0, hr = 0, cl = 0, cr = 0;
    if (!validate_node(node->left, has_lower, lower, true, node->key, &hl, &cl) ||
        !validate_node(node->right, true, node->key, has_upper, upper, &hr, &cr)) {
        return false;
    }
    if ((hl > hr && hl - hr > 1) || (hr > hl && hr - hl > 1)) {
        return false;
    }
    size_t expected_h = 1 + max_size(hl, hr);
    if (node->height != expected_h) {
        return false;
    }
    *out_height = expected_h;
    *out_count = cl + cr + 1;
    return true;
}

bool avl_validate(const AvlTree *tree) {
    if (tree == NULL) {
        return false;
    }
    size_t h = 0, c = 0;
    return validate_node(tree->root, false, 0, false, 0, &h, &c) && c == tree->size;
}

int main(void) {
    avl_destroy(NULL);

    AvlTree *tree = avl_create();
    assert(tree != NULL);
    assert(avl_validate(tree));

    bool ins = false;
    /* 构造涵盖 RR、RL、LL、LR 四种失衡的插入序列:
       10, 20, 30 (触发根结点 10 处的 RR 旋转)
       25, 27 (触发 RL 旋转)
       5, 4 (触发 LL 旋转)
       7, 6 (触发 LR 旋转)
    */
    int64_t seq[] = {10, 20, 30, 25, 27, 5, 4, 7, 6};
    for (size_t i = 0; i < sizeof(seq)/sizeof(seq[0]); ++i) {
        assert(avl_insert(tree, seq[i], &ins) && ins);
        assert(avl_validate(tree));
    }
    assert(tree->size == 9);

    /* 重复插入测试 */
    assert(avl_insert(tree, 25, &ins) && !ins);
    assert(tree->size == 9);
    assert(avl_validate(tree));

    /* 成员查找测试 */
    for (size_t i = 0; i < sizeof(seq)/sizeof(seq[0]); ++i) {
        assert(avl_contains(tree, seq[i]));
    }
    assert(!avl_contains(tree, 100));

    avl_destroy(tree);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：插入沿一条根叶路径下探，比较次数不超过树高 $O(\log n)$；回溯途中最多执行一次再平衡（1 或 2 次单旋），单次插入总耗时为严格的 $O(\log n)$。查找耗时为 $O(\log n)$。全树验证需遍历所有结点，时间复杂度为 $O(n)$。
- **空间复杂度**：递归栈深度受树高限制，额外空间复杂度为 $O(\log n)$。

## 延伸思考 / 模式总结

AVL 树通过强力维护每一个结点的绝对高度差 $\le 1$，保证了树形态的高度平衡性（树高严格小于 $1.44 \log_2 n$）。相比于红黑树，AVL 树具有更低的最大高度，因此在**读多写少（只读查找频次远高于插入）**的场景下，能提供更极致的查找性能。

# 31502：AVL 树的下界查询与区间剪枝查询

## 题目

在 `31501` 的 `AvlTree` 上增加下界查询与有序闭区间查询：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

bool avl_lower_bound(
    const AvlTree *tree,
    int64_t target,
    int64_t *out_key,
    bool *out_found
);
bool avl_range_query(
    const AvlTree *tree,
    int64_t low,
    int64_t high,
    int64_t *out_keys,
    size_t capacity,
    size_t *out_count
);
```

要求：

1. `avl_lower_bound` 查找树中满足 `key >= target` 的最小键：若存在则写入 `*out_key`、设置 `*out_found = true` 并返回 `true`；若不存在满足条件的键，成功返回 `true`、设置 `*out_found = false`，且不得修改 `*out_key`。
2. `avl_range_query` 按严格升序导出闭区间 `[low, high]` 内的全部键：遍历时必须利用二叉搜索树有序性剪枝不与 `[low, high]` 相交的子树（仅当 `node->key > low` 时递归左子树，仅当 `node->key < high` 时递归右子树）。
3. 当 `low > high`、指针参数无效、或命中键数超过 `capacity`（以及命中键数大于 `0` 但 `out_keys == NULL`）时返回 `false`，且不得修改输出数组与输出对象。
4. `avl_lower_bound` 的时间复杂度为 $O(\log n)$，`avl_range_query` 的时间复杂度为 $O(\log n + k)$，其中 $k$ 为落在 `[low, high]` 内的键数量。

## 解析

AVL 树在保持平衡的同时严格满足二叉搜索树（BST）的全局有序性：

1. **下界查询 (`avl_lower_bound`)**：从根结点出发维护候选指针 `candidate = NULL`。每当当前结点满足 `node->key >= target` 时，它就是一个合法的下界候选，而更小但仍不小于 `target` 的键只可能出现在其左子树中，因此记录 `candidate = node` 并转向左孩子；反之，若 `node->key < target`，当前结点及其整个左子树均严格小于 `target`，直接转向右孩子。整条搜索路径长度不超过树高 $O(\log n)$。
2. **闭区间剪枝查询 (`avl_range_query`)**：对子树根 `node` 执行受限中序遍历：
   - 仅当 `node->key > low` 时，左子树中才可能存在不小于 `low` 的键，此时递归访问左子树；若 `node->key <= low`，左子树全部键严格小于 `low`，直接整棵剪枝；
   - 若 `low <= node->key && node->key <= high`，当前键属于结果集；
   - 仅当 `node->key < high` 时，右子树中才可能存在不大于 `high` 的键，此时递归访问右子树；若 `node->key >= high`，右子树全部键严格大于 `high`，直接整棵剪枝。

为了保证当输出缓冲区 `capacity` 不足时不污染调用方的 `out_keys` 数组，同时避免在只读查询中动态分配堆内存，可以先执行一趟带剪枝的区间计数，确认 `matched <= capacity` 后再执行第二趟带剪枝的升序写入。两趟访问的结点集合完全相同，被访问的结点要么位于 `[low, high]` 内部（共 $k$ 个），要么位于两条边界搜索路径上（至多 $2h = O(\log n)$ 个），总时间复杂度严格为 $O(\log n + k)$。

## 答案

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct AvlTree AvlTree;
typedef struct AvlNode AvlNode;

struct AvlNode {
    int64_t key;
    size_t height;
    AvlNode *left;
    AvlNode *right;
};

struct AvlTree {
    AvlNode *root;
    size_t size;
};

static size_t node_height(const AvlNode *node) {
    return node == NULL ? 0 : node->height;
}

static size_t maximum_size(size_t left, size_t right) {
    return left > right ? left : right;
}

static void update_height(AvlNode *node) {
    size_t child_height = maximum_size(
        node_height(node->left), node_height(node->right)
    );
    node->height = child_height == SIZE_MAX
        ? SIZE_MAX
        : child_height + 1;
}

static AvlNode *rotate_right(AvlNode *root) {
    AvlNode *pivot = root->left;
    root->left = pivot->right;
    pivot->right = root;
    update_height(root);
    update_height(pivot);
    return pivot;
}

static AvlNode *rotate_left(AvlNode *root) {
    AvlNode *pivot = root->right;
    root->right = pivot->left;
    pivot->left = root;
    update_height(root);
    update_height(pivot);
    return pivot;
}

static AvlNode *rebalance(AvlNode *node) {
    update_height(node);
    size_t left_height = node_height(node->left);
    size_t right_height = node_height(node->right);

    if (left_height > right_height &&
        left_height - right_height > 1) {
        if (node_height(node->left->right) >
            node_height(node->left->left)) {
            node->left = rotate_left(node->left);
        }
        return rotate_right(node);
    }
    if (right_height > left_height &&
        right_height - left_height > 1) {
        if (node_height(node->right->left) >
            node_height(node->right->right)) {
            node->right = rotate_right(node->right);
        }
        return rotate_left(node);
    }
    return node;
}

static bool insert_node(
    AvlNode **link,
    int64_t key,
    bool *out_inserted
) {
    if (*link == NULL) {
        AvlNode *node = malloc(sizeof *node);
        if (node == NULL) {
            return false;
        }
        node->key = key;
        node->height = 1;
        node->left = NULL;
        node->right = NULL;
        *link = node;
        *out_inserted = true;
        return true;
    }
    if (key == (*link)->key) {
        *out_inserted = false;
        return true;
    }

    AvlNode **next = key < (*link)->key
        ? &(*link)->left
        : &(*link)->right;
    if (!insert_node(next, key, out_inserted)) {
        return false;
    }
    if (*out_inserted) {
        *link = rebalance(*link);
    }
    return true;
}

AvlTree *avl_create(void) {
    AvlTree *tree = malloc(sizeof *tree);
    if (tree == NULL) {
        return NULL;
    }
    tree->root = NULL;
    tree->size = 0;
    return tree;
}

static void destroy_nodes(AvlNode *node) {
    if (node == NULL) {
        return;
    }
    destroy_nodes(node->left);
    destroy_nodes(node->right);
    free(node);
}

void avl_destroy(AvlTree *tree) {
    if (tree == NULL) {
        return;
    }
    destroy_nodes(tree->root);
    free(tree);
}

bool avl_contains(const AvlTree *tree, int64_t key) {
    if (tree == NULL) {
        return false;
    }
    const AvlNode *node = tree->root;
    while (node != NULL) {
        if (key == node->key) {
            return true;
        }
        node = key < node->key ? node->left : node->right;
    }
    return false;
}

bool avl_insert(
    AvlTree *tree,
    int64_t key,
    bool *out_inserted
) {
    if (tree == NULL || out_inserted == NULL) {
        return false;
    }
    if (tree->size == SIZE_MAX && !avl_contains(tree, key)) {
        return false;
    }
    bool inserted = false;
    if (!insert_node(&tree->root, key, &inserted)) {
        return false;
    }
    if (inserted) {
        ++tree->size;
    }
    *out_inserted = inserted;
    return true;
}

bool avl_lower_bound(
    const AvlTree *tree,
    int64_t target,
    int64_t *out_key,
    bool *out_found
) {
    if (tree == NULL || out_key == NULL || out_found == NULL) {
        return false;
    }
    const AvlNode *cursor = tree->root;
    const AvlNode *candidate = NULL;
    while (cursor != NULL) {
        if (cursor->key >= target) {
            candidate = cursor;
            cursor = cursor->left;
        } else {
            cursor = cursor->right;
        }
    }
    if (candidate == NULL) {
        *out_found = false;
        return true;
    }
    *out_key = candidate->key;
    *out_found = true;
    return true;
}

static void count_in_range(
    const AvlNode *node,
    int64_t low,
    int64_t high,
    size_t *counter
) {
    if (node == NULL) {
        return;
    }
    if (node->key > low) {
        count_in_range(node->left, low, high, counter);
    }
    if (node->key >= low && node->key <= high) {
        ++(*counter);
    }
    if (node->key < high) {
        count_in_range(node->right, low, high, counter);
    }
}

static void collect_in_range(
    const AvlNode *node,
    int64_t low,
    int64_t high,
    int64_t *out_keys,
    size_t *cursor
) {
    if (node == NULL) {
        return;
    }
    if (node->key > low) {
        collect_in_range(node->left, low, high, out_keys, cursor);
    }
    if (node->key >= low && node->key <= high) {
        out_keys[(*cursor)++] = node->key;
    }
    if (node->key < high) {
        collect_in_range(node->right, low, high, out_keys, cursor);
    }
}

bool avl_range_query(
    const AvlTree *tree,
    int64_t low,
    int64_t high,
    int64_t *out_keys,
    size_t capacity,
    size_t *out_count
) {
    if (tree == NULL || out_count == NULL || low > high) {
        return false;
    }
    size_t matched = 0;
    count_in_range(tree->root, low, high, &matched);
    if (matched > capacity || (matched > 0 && out_keys == NULL)) {
        return false;
    }
    size_t cursor = 0;
    collect_in_range(tree->root, low, high, out_keys, &cursor);
    *out_count = cursor;
    return true;
}
```

下界查询沿单一根叶路径下降，时间复杂度为 $O(\log n)$、额外空间复杂度为 $O(1)$；闭区间查询剪除所有位于 `[low, high]` 之外的子树，时间复杂度为 $O(\log n + k)$，递归调用栈深度不超过树高 $O(\log n)$。

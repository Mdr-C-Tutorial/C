# 31508 - 树堆随机优先级与旋转堆维护

## 题目描述

实现结合了二叉搜索树（BST）与二叉堆（Heap）特性的随机化平衡数据结构——树堆（Treap），支持插入、删除、查找以及全树双重不变式验证。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct Treap Treap;

Treap *treap_create(uint64_t seed);
void treap_destroy(Treap *tree);
bool treap_insert(Treap *tree, int64_t key, bool *out_inserted);
bool treap_remove(Treap *tree, int64_t key, bool *out_removed);
bool treap_contains(const Treap *tree, int64_t key);
bool treap_validate(const Treap *tree);
```

要求：

1. 每个结点保存用户键 `key` 与内部生成的伪随机优先级 `priority`。
2. **BST 不变式**：对于任意结点，其左子树所有结点的 `key` 严格小于当前结点的 `key`，右子树所有结点的 `key` 严格大于当前结点的 `key`。
3. **大顶堆（或小顶堆）不变式**：对于任意非空结点，其 `priority` 严格大于（或小于）其所有子结点的 `priority`。
4. 插入新结点时，先按 BST 规则挂接为叶子，若违反堆性质，通过自底向上的树旋转（左旋或右旋）将该结点提升，直到堆性质恢复。
5. 删除结点时，通过旋转将待删结点不断向下方“沉降”（每次旋转让具有较高优先级的孩子上升），直到待删结点变为叶子结点后再释放删除。
6. 期望时间复杂度为 $O(\log n)$，验证器需同时检查 BST 有序性与堆有序性。

## 解题思路

### 1. 为什么随机优先级能保证平衡？

若一棵二叉搜索树是将 $n$ 个键按照某种完全随机的排列依次插入而形成的，由概率论可知，其平均树高严格为 $O(\log n)$。
Treap 的惊人结论（Aragon & Seidel，1989）在于：**只要所有结点的键各不相同，且优先级也各不相同，那么满足“键为 BST、优先级为大顶堆”的二叉树拓扑结构是全宇宙唯一确定的！**
当优先级是由高质量伪随机数发生器独立均匀生成时，Treap 的最终物理拓扑在数学上等价于“将所有结点按照优先级从大到小的随机次序插入普通 BST”。因此，无论外部用户按照何种恶意的升序或降序插入键，Treap 的期望高度始终被锚定在 $O(\log n)$，彻底免疫最坏退化！

### 2. 插入与删除的旋转沉浮

- **插入 (`treap_insert`)**：
  沿 BST 路径递归插入叶子结点并赋予随机优先级。回溯时：
  - 若左孩子优先级大于当前结点：执行一次**右旋**提升左孩子；
  - 若右孩子优先级大于当前结点：执行一次**左旋**提升右孩子；
  旋转由于保持了中序有序性，因而 BST 性质绝不破坏，同时使堆性质得以修复。
- **删除 (`treap_remove`)**：
  搜索定位到待删结点。若该结点拥有两个孩子，比较其左右孩子的优先级：
  - 若左孩子优先级更高：执行一次**右旋**，将待删结点下沉到右子树，继续递归在右侧删除；
  - 若右孩子优先级更高：执行一次**左旋**，将待删结点下沉到左子树，继续递归在左侧删除；
  直到待删结点成为叶子或仅有单孩子时，直接将其释放，无需复杂的后继键值覆盖。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct TreapNode TreapNode;

struct TreapNode {
    int64_t key;
    uint32_t priority;
    TreapNode *left;
    TreapNode *right;
};

typedef struct Treap {
    TreapNode *root;
    uint64_t rng_state;
    size_t size;
} Treap;

static uint32_t treap_rand(Treap *t) {
    /* 64 位 xorshift 伪随机数发生器 */
    t->rng_state ^= t->rng_state >> 12;
    t->rng_state ^= t->rng_state << 25;
    t->rng_state ^= t->rng_state >> 27;
    return (uint32_t)((t->rng_state * UINT64_C(0x2545F4914F6CDD1D)) >> 32);
}

Treap *treap_create(uint64_t seed) {
    Treap *t = (Treap *)malloc(sizeof(Treap));
    if (t == NULL) {
        return NULL;
    }
    t->root = NULL;
    t->rng_state = seed != 0 ? seed : 0x853c49e6748fea9bULL;
    t->size = 0;
    return t;
}

static void free_nodes(TreapNode *n) {
    if (n == NULL) {
        return;
    }
    free_nodes(n->left);
    free_nodes(n->right);
    free(n);
}

void treap_destroy(Treap *t) {
    if (t == NULL) {
        return;
    }
    free_nodes(t->root);
    free(t);
}

static TreapNode *rotate_right(TreapNode *y) {
    TreapNode *x = y->left;
    y->left = x->right;
    x->right = y;
    return x;
}

static TreapNode *rotate_left(TreapNode *x) {
    TreapNode *y = x->right;
    x->right = y->left;
    y->left = x;
    return y;
}

static TreapNode *treap_insert_rec(
    Treap *t,
    TreapNode *node,
    int64_t key,
    bool *out_inserted,
    bool *out_ok
) {
    if (node == NULL) {
        TreapNode *n = (TreapNode *)malloc(sizeof(TreapNode));
        if (n == NULL) {
            *out_ok = false;
            return NULL;
        }
        n->key = key;
        n->priority = treap_rand(t);
        n->left = NULL;
        n->right = NULL;
        *out_inserted = true;
        *out_ok = true;
        return n;
    }
    if (key == node->key) {
        *out_inserted = false;
        *out_ok = true;
        return node;
    }
    if (key < node->key) {
        node->left = treap_insert_rec(t, node->left, key, out_inserted, out_ok);
        /* 若左孩子优先级违反大顶堆，执行右旋提升 */
        if (*out_ok && node->left->priority > node->priority) {
            node = rotate_right(node);
        }
    } else {
        node->right = treap_insert_rec(t, node->right, key, out_inserted, out_ok);
        /* 若右孩子优先级违反大顶堆，执行左旋提升 */
        if (*out_ok && node->right->priority > node->priority) {
            node = rotate_left(node);
        }
    }
    return node;
}

bool treap_insert(Treap *t, int64_t key, bool *out_inserted) {
    if (t == NULL || out_inserted == NULL) {
        return false;
    }
    bool inserted = false;
    bool ok = false;
    TreapNode *new_root = treap_insert_rec(t, t->root, key, &inserted, &ok);
    if (!ok) {
        return false;
    }
    t->root = new_root;
    if (inserted) {
        t->size++;
    }
    *out_inserted = inserted;
    return true;
}

static TreapNode *treap_remove_rec(TreapNode *node, int64_t key, bool *out_removed) {
    if (node == NULL) {
        *out_removed = false;
        return NULL;
    }
    if (key < node->key) {
        node->left = treap_remove_rec(node->left, key, out_removed);
    } else if (key > node->key) {
        node->right = treap_remove_rec(node->right, key, out_removed);
    } else {
        *out_removed = true;
        /* 叶结点直接释放 */
        if (node->left == NULL && node->right == NULL) {
            free(node);
            return NULL;
        }
        if (node->left == NULL) {
            TreapNode *r = node->right;
            free(node);
            return r;
        }
        if (node->right == NULL) {
            TreapNode *l = node->left;
            free(node);
            return l;
        }
        /* 两个孩子均存在：让优先级更高的一侧孩子通过旋转上升，待删结点下沉 */
        if (node->left->priority > node->right->priority) {
            node = rotate_right(node);
            node->right = treap_remove_rec(node->right, key, out_removed);
        } else {
            node = rotate_left(node);
            node->left = treap_remove_rec(node->left, key, out_removed);
        }
    }
    return node;
}

bool treap_remove(Treap *t, int64_t key, bool *out_removed) {
    if (t == NULL || out_removed == NULL) {
        return false;
    }
    bool removed = false;
    t->root = treap_remove_rec(t->root, key, &removed);
    if (removed) {
        t->size--;
    }
    *out_removed = removed;
    return true;
}

bool treap_contains(const Treap *t, int64_t key) {
    if (t == NULL) {
        return false;
    }
    const TreapNode *curr = t->root;
    while (curr != NULL) {
        if (key == curr->key) {
            return true;
        }
        curr = (key < curr->key) ? curr->left : curr->right;
    }
    return false;
}

static bool validate_treap_rec(
    const TreapNode *node,
    bool has_low,
    int64_t low,
    bool has_high,
    int64_t high
) {
    if (node == NULL) {
        return true;
    }
    /* 检验 BST 键有序性 */
    if ((has_low && node->key <= low) || (has_high && node->key >= high)) {
        return false;
    }
    /* 检验大顶堆优先级不变式 */
    if (node->left != NULL && node->left->priority > node->priority) {
        return false;
    }
    if (node->right != NULL && node->right->priority > node->priority) {
        return false;
    }

    return validate_treap_rec(node->left, has_low, low, true, node->key) &&
           validate_treap_rec(node->right, true, node->key, has_high, high);
}

bool treap_validate(const Treap *t) {
    if (t == NULL) {
        return false;
    }
    return validate_treap_rec(t->root, false, 0, false, 0);
}

int main(void) {
    Treap *t = treap_create(42);
    assert(t != NULL);
    assert(treap_validate(t));

    bool ins = false;
    /* 顺序插入 50 个单调键，验证随机优先级能否抵御单链退化 */
    for (int64_t i = 1; i <= 50; ++i) {
        assert(treap_insert(t, i, &ins) && ins);
        assert(treap_validate(t));
    }
    assert(t->size == 50);

    /* 重复插入测试 */
    assert(treap_insert(t, 25, &ins) && !ins);
    assert(t->size == 50);
    assert(treap_validate(t));

    /* 成员包含测试 */
    for (int64_t i = 1; i <= 50; ++i) {
        assert(treap_contains(t, i));
    }
    assert(!treap_contains(t, 100));

    /* 动态删除测试 */
    bool rem = false;
    for (int64_t i = 1; i <= 25; ++i) {
        assert(treap_remove(t, i, &rem) && rem);
        assert(treap_validate(t));
        assert(!treap_contains(t, i));
    }
    assert(t->size == 25);

    /* 清空剩余结点 */
    for (int64_t i = 26; i <= 50; ++i) {
        assert(treap_remove(t, i, &rem) && rem);
        assert(treap_validate(t));
    }
    assert(t->size == 0);
    assert(t->root == NULL);

    treap_destroy(t);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：得益于随机独立优先级，树的高度的期望值为 $O(\log n)$。查找、插入与删除操作均沿单条根叶路径下探或旋转，平均/期望时间复杂度均为严格的 $O(\log n)$。
- **空间复杂度**：递归栈深度期望为 $O(\log n)$，额外空间复杂度为 $O(\log n)$。

## 延伸思考 / 模式总结

Treap 是计算机科学中“将随机化算法引入确定性数据结构”的经典杰作。相比于 AVL 树（需精密计算平衡因子）和红黑树（需分类处理叔叔颜色与双黑结点），Treap 的逻辑代码量极少，却能提供等价的概率性能保证。在无旋 Treap（Merge/Split Treap）中，它还可以进一步扩展支持区间翻转与可持久化。

# 31508 - 伸展树自底向上伸展操作

## 题目

实现自调整二叉搜索树——伸展树（Splay Tree），支持结点的插入、查找、删除，以及核心的自底向上伸展操作（Splay），涵盖 Zig、Zig-Zig、Zig-Zag 三种旋转构型。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct SplayTree SplayTree;

SplayTree *splay_create(void);
void splay_destroy(SplayTree *tree);
bool splay_insert(SplayTree *tree, int64_t key);
bool splay_find(SplayTree *tree, int64_t key);
bool splay_remove(SplayTree *tree, int64_t key);
```

要求：

1. 结点结构中**无需存储**平衡因子或颜色标记，仅保存父指针与左右孩子指针。
2. 任何访问操作（查找、插入、删除）均将目标结点（或最后访问的最近邻结点）自底向上伸展（Splay）到根结点。
3. 严格遵循 Tarjan 伸展规则：
   - **Zig**：目标结点的父结点为根，执行一次单旋转；
   - **Zig-Zig**：目标结点与父结点具有同向倾斜（同为左孩子或同为右孩子），**必须先旋转祖父结点，再旋转父结点**；
   - **Zig-Zag**：目标结点与父结点反向倾斜（内侧折线），先旋转父结点，再旋转祖父结点。
4. 保证在任意访问序列下，单次操作的均摊时间复杂度为 $O(\log n)$。

## 解题思路

### 1. 为什么朴素单旋不能保证均摊 $O(\log n)$？

若在每次访问深层结点时，简单地连续执行朴素单旋将其提升到根，在一条退化为单链的树上按顺序访问结点时，单旋仅仅是将整条链反向翻转，无法实质性改变后续结点的长路径，会导致连续 $n$ 次访问的总时间恶化为 $\Theta(n^2)$。

### 2. Tarjan 双重伸展的奇妙之处（祖父优先）

Tarjan 与 Sleator 在 1985 年提出：在同侧同向倾斜（Zig-Zig）时，**必须先对祖父结点 $g$ 旋转，然后再对父结点 $p$ 旋转**！
通过势能分析法（Potential Method）证明：这种“祖父优先”的旋转会使整条原访问路径上的所有深层后代结点的深度大约减半（路径压缩效应）。每次耗费较大时间的深层访问，都会产生强烈的树形扁平化收益，从而在数学上严格保证了任意 $m$ 次操作的均摊复杂度为 $O(m \log n)$。

### 3. 操作与伸展结合

- **查找 (`splay_find`)**：搜索目标键。若命中，将目标结点伸展至根；若未命中，将查找路径上最后遇到的非空叶子结点伸展至根。
- **插入 (`splay_insert`)**：先执行查找，此时与待插键最接近的结点已被伸展到根。直接在根处切分子树，将新结点作为新根插入。
- **删除 (`splay_remove`)**：先执行查找将待删结点伸展到根，随后分离左右子树；在左子树中伸展最大键至左子树根，再将右子树直接挂接为其右孩子即可。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct SplayNode SplayNode;

struct SplayNode {
    int64_t key;
    SplayNode *left;
    SplayNode *right;
    SplayNode *parent;
};

typedef struct SplayTree {
    SplayNode *root;
    size_t size;
} SplayTree;

SplayTree *splay_create(void) {
    SplayTree *tree = (SplayTree *)malloc(sizeof(SplayTree));
    if (tree == NULL) {
        return NULL;
    }
    tree->root = NULL;
    tree->size = 0;
    return tree;
}

static void free_nodes(SplayNode *n) {
    if (n == NULL) {
        return;
    }
    free_nodes(n->left);
    free_nodes(n->right);
    free(n);
}

void splay_destroy(SplayTree *tree) {
    if (tree == NULL) {
        return;
    }
    free_nodes(tree->root);
    free(tree);
}

static void rotate_left(SplayTree *tree, SplayNode *x) {
    SplayNode *y = x->right;
    assert(y != NULL);
    x->right = y->left;
    if (y->left != NULL) {
        y->left->parent = x;
    }
    y->parent = x->parent;
    if (x->parent == NULL) {
        tree->root = y;
    } else if (x == x->parent->left) {
        x->parent->left = y;
    } else {
        x->parent->right = y;
    }
    y->left = x;
    x->parent = y;
}

static void rotate_right(SplayTree *tree, SplayNode *y) {
    SplayNode *x = y->left;
    assert(x != NULL);
    y->left = x->right;
    if (x->right != NULL) {
        x->right->parent = y;
    }
    x->parent = y->parent;
    if (y->parent == NULL) {
        tree->root = x;
    } else if (y == y->parent->left) {
        y->parent->left = x;
    } else {
        y->parent->right = x;
    }
    x->right = y;
    y->parent = x;
}

static void splay(SplayTree *tree, SplayNode *x) {
    if (x == NULL) {
        return;
    }
    while (x->parent != NULL) {
        SplayNode *p = x->parent;
        SplayNode *g = p->parent;
        if (g == NULL) {
            /* Zig 构型：单旋 */
            if (x == p->left) {
                rotate_right(tree, p);
            } else {
                rotate_left(tree, p);
            }
        } else if (x == p->left && p == g->left) {
            /* Zig-Zig 同向左：先转祖父，再转父 */
            rotate_right(tree, g);
            rotate_right(tree, p);
        } else if (x == p->right && p == g->right) {
            /* Zig-Zig 同向右：先转祖父，再转父 */
            rotate_left(tree, g);
            rotate_left(tree, p);
        } else if (x == p->right && p == g->left) {
            /* Zig-Zag 折线：先转父，再转祖父 */
            rotate_left(tree, p);
            rotate_right(tree, g);
        } else {
            /* Zig-Zag 折线：先转父，再转祖父 */
            rotate_right(tree, p);
            rotate_left(tree, g);
        }
    }
}

bool splay_find(SplayTree *tree, int64_t key) {
    if (tree == NULL || tree->root == NULL) {
        return false;
    }
    SplayNode *curr = tree->root;
    SplayNode *last = NULL;
    while (curr != NULL) {
        last = curr;
        if (key == curr->key) {
            splay(tree, curr);
            return true;
        }
        curr = (key < curr->key) ? curr->left : curr->right;
    }
    /* 未命中时将最后访问的最近邻叶子伸展至根 */
    splay(tree, last);
    return false;
}

bool splay_insert(SplayTree *tree, int64_t key) {
    if (tree == NULL) {
        return false;
    }
    if (tree->root == NULL) {
        SplayNode *node = (SplayNode *)malloc(sizeof(SplayNode));
        if (node == NULL) {
            return false;
        }
        node->key = key;
        node->left = NULL;
        node->right = NULL;
        node->parent = NULL;
        tree->root = node;
        tree->size = 1;
        return true;
    }

    if (splay_find(tree, key)) {
        return true;
    }

    SplayNode *root = tree->root;
    SplayNode *node = (SplayNode *)malloc(sizeof(SplayNode));
    if (node == NULL) {
        return false;
    }
    node->key = key;
    node->parent = NULL;

    if (key < root->key) {
        node->right = root;
        node->left = root->left;
        if (root->left != NULL) {
            root->left->parent = node;
        }
        root->left = NULL;
        root->parent = node;
    } else {
        node->left = root;
        node->right = root->right;
        if (root->right != NULL) {
            root->right->parent = node;
        }
        root->right = NULL;
        root->parent = node;
    }

    tree->root = node;
    tree->size++;
    return true;
}

bool splay_remove(SplayTree *tree, int64_t key) {
    if (tree == NULL || tree->root == NULL) {
        return false;
    }
    if (!splay_find(tree, key)) {
        return false;
    }

    /* 目标键现已位于根结点 */
    SplayNode *del = tree->root;
    if (del->left == NULL) {
        tree->root = del->right;
        if (tree->root != NULL) {
            tree->root->parent = NULL;
        }
    } else {
        SplayNode *left_subtree = del->left;
        left_subtree->parent = NULL;

        /* 在左子树中寻找最大结点并伸展至左子树根 */
        SplayNode *max_node = left_subtree;
        while (max_node->right != NULL) {
            max_node = max_node->right;
        }

        SplayTree left_tree = {.root = left_subtree, .size = 0};
        splay(&left_tree, max_node);
        left_subtree = left_tree.root;

        /* 此时 left_subtree 没有右孩子，直接挂接原右子树 */
        left_subtree->right = del->right;
        if (del->right != NULL) {
            del->right->parent = left_subtree;
        }
        tree->root = left_subtree;
    }

    free(del);
    tree->size--;
    return true;
}

static bool validate_bst(const SplayNode *n, bool has_low, int64_t low, bool has_high, int64_t high) {
    if (n == NULL) {
        return true;
    }
    if ((has_low && n->key <= low) || (has_high && n->key >= high)) {
        return false;
    }
    if (n->left != NULL && n->left->parent != n) {
        return false;
    }
    if (n->right != NULL && n->right->parent != n) {
        return false;
    }
    return validate_bst(n->left, has_low, low, true, n->key) &&
           validate_bst(n->right, true, n->key, has_high, high);
}

int main(void) {
    SplayTree *tree = splay_create();
    assert(tree != NULL);

    /* 连续插入 30 个递增整数 */
    for (int64_t i = 1; i <= 30; ++i) {
        assert(splay_insert(tree, i));
        assert(tree->root != NULL && tree->root->key == i);
        assert(validate_bst(tree->root, false, 0, false, 0));
    }
    assert(tree->size == 30);

    /* 访问深层结点 1 -> 触发连续 Zig-Zig 伸展至根 */
    assert(splay_find(tree, 1));
    assert(tree->root->key == 1);
    assert(validate_bst(tree->root, false, 0, false, 0));

    /* 访问中间结点 15 -> 伸展至根 */
    assert(splay_find(tree, 15));
    assert(tree->root->key == 15);
    assert(validate_bst(tree->root, false, 0, false, 0));

    /* 删除根结点 15 */
    assert(splay_remove(tree, 15));
    assert(validate_bst(tree->root, false, 0, false, 0));
    assert(!splay_find(tree, 15));
    assert(tree->size == 29);

    splay_destroy(tree);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：单次最坏操作耗时可能为 $O(n)$（例如退化单链上的单次访问）；但借助势能分析法可证，任意连续 $m$ 次操作的总时间复杂度严格为 $O(m \log n)$，即**均摊时间复杂度为严格的 $O(\log n)$**。
- **空间复杂度**：由于不需要存储高度、平衡因子或颜色字段，每个结点相比普通平衡树节省了额外的元数据开销，空间紧凑且额外空间为严格的 $O(1)$。

## 延伸思考 / 模式总结

伸展树具备显著的**局部性优化特征（Temporal Locality）**：最近刚被频繁访问的一小部分“热点数据”会自动聚拢在树根附近，对这些热点键的单次访问时间仅为 $O(1)$！这种无需显式元数据的自适应平衡结构，在网络路由表缓存、文本压缩编码以及只读热点查询中展现出独特的优势。

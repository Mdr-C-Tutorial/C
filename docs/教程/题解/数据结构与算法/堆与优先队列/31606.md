# 31606 - 对顶堆动态维护数据流中位数

## 题目描述

设计并实现一个支持在线动态插入数值、并在 $O(1)$ 时间内查询当前所有已读入数值中位数的数据结构 `MedianFinder`（对顶堆 / 双堆法）：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct MedianFinder MedianFinder;

MedianFinder *median_finder_create(void);
void median_finder_destroy(MedianFinder *mf);
bool median_finder_add(MedianFinder *mf, int64_t val);
bool median_finder_find(const MedianFinder *mf, double *out_median);
size_t median_finder_size(const MedianFinder *mf);
```

要求：

1. 内部组合使用一个大顶堆（维护较小的一半元素）和一个小顶堆（维护较大的一半元素）。
2. 动态维护平衡不变式：
   - 设总元素个数为 $N$；
   - 当 $N$ 为偶数时，两堆规模严格相等：$\text{size}_{\text{max}} = \text{size}_{\text{min}}$；
   - 当 $N$ 为奇数时，大顶堆比小顶堆恰多容纳一个元素：$\text{size}_{\text{max}} = \text{size}_{\text{min}} + 1$；
   - 大顶堆的最大元素恒不大于小顶堆的最小元素：$\max(\text{Left}) \le \min(\text{Right})$。
3. `median_finder_add` 在单次 $O(\log n)$ 时间内完成新元素插入并动态重平衡；若内存分配失败，保持原结构不变并返回 `false`。
4. `median_finder_find` 在 $O(1)$ 时间内计算中位数并写入 `*out_median`：若总数为奇数，中位数即为大顶堆堆顶；若总数为偶数，中位数两堆顶均值。空结构时返回 `false` 且不修改输出对象。

## 解题思路

### 1. 中位数的分界性质与对顶堆拓扑

对于一个已排序的序列，中位数恰好将全集划分为左右对称的两半：
- 左半部分的所有数值均小于等于右半部分；
- 左半部分的极大值与右半部分的极小值构成了中位数的判定依据。

若使用平衡二叉搜索树（如 AVL 或红黑树）维护，查找中位数需要通过子树规模做排名检索，开销为 $O(\log n)$ 且代码复杂度较高。
而**对顶堆（Dual Heap）**将左半部分封装为大顶堆、右半部分封装为小顶堆：
- 大顶堆的堆顶指向左半部的最大值；
- 小顶堆的堆顶指向右半部的最小值；
两个堆顶在数值轴上“面对面顶住”，故称“对顶堆”。查询中位数只需读取两个堆顶，耗时严格为 $O(1)$。

```text
       [大顶堆: 较小的一半]            [小顶堆: 较大的一半]
          堆底 ... 叶子                   叶子 ... 堆底
                \                              /
                 \                            /
                [ 堆顶 max ]  <=  [ 堆顶 min ]
                      \              /
                       \            /
                      中 位 数 判 定 处
```

### 2. 动态插入与平衡不变式维系

每次调用 `median_finder_add(mf, val)` 时分两步执行：

1. **定向路由**：
   - 若大顶堆为空，或新值 $val \le \text{peek}(\text{max\_heap})$，则 $val$ 归属于较小半区，推入大顶堆；
   - 否则 $val$ 归属于较大半区，推入小顶堆。
2. **堆规模动态再平衡**：
   - 若插入后大顶堆规模超过小顶堆多于 1 个（即 $\text{size}_{\text{max}} > \text{size}_{\text{min}} + 1$），则将大顶堆堆顶弹出并推入小顶堆；
   - 若插入后小顶堆规模反超大顶堆（即 $\text{size}_{\text{max}} < \text{size}_{\text{min}}$），则将小顶堆堆顶弹出并推入大顶堆。
经过至多一次单元素迁移，两堆规模之差重新满足 $0 \le \text{size}_{\text{max}} - \text{size}_{\text{min}} \le 1$。

## C 语言实现

```c
#include <assert.h>
#include <math.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

typedef struct IntHeap {
    int64_t *items;
    size_t size;
    size_t capacity;
    bool is_min;
} IntHeap;

static IntHeap *heap_create(bool is_min) {
    IntHeap *h = malloc(sizeof *h);
    if (h == NULL) {
        return NULL;
    }
    h->items = NULL;
    h->size = 0;
    h->capacity = 0;
    h->is_min = is_min;
    return h;
}

static void heap_destroy(IntHeap *h) {
    if (h == NULL) {
        return;
    }
    free(h->items);
    free(h);
}

static bool heap_reserve(IntHeap *h) {
    if (h->size < h->capacity) {
        return true;
    }
    size_t max_cap = SIZE_MAX / sizeof(int64_t);
    if (h->capacity >= max_cap) {
        return false;
    }
    size_t next = h->capacity == 0 ? 8 : (h->capacity > max_cap / 2 ? max_cap : h->capacity * 2);
    if (next <= h->size) {
        return false;
    }
    int64_t *new_items = realloc(h->items, next * sizeof(int64_t));
    if (new_items == NULL) {
        return false;
    }
    h->items = new_items;
    h->capacity = next;
    return true;
}

static bool cmp(bool is_min, int64_t a, int64_t b) {
    return is_min ? (a < b) : (a > b);
}

static void heap_sift_up(IntHeap *h, size_t i) {
    while (i > 0) {
        size_t p = (i - 1) / 2;
        if (!cmp(h->is_min, h->items[i], h->items[p])) {
            break;
        }
        int64_t tmp = h->items[i];
        h->items[i] = h->items[p];
        h->items[p] = tmp;
        i = p;
    }
}

static void heap_sift_down(IntHeap *h, size_t i) {
    while (i < h->size / 2) {
        size_t left = i * 2 + 1;
        size_t right = left + 1;
        size_t best = i;
        if (cmp(h->is_min, h->items[left], h->items[best])) {
            best = left;
        }
        if (right < h->size && cmp(h->is_min, h->items[right], h->items[best])) {
            best = right;
        }
        if (best == i) {
            break;
        }
        int64_t tmp = h->items[i];
        h->items[i] = h->items[best];
        h->items[best] = tmp;
        i = best;
    }
}

static bool heap_push(IntHeap *h, int64_t val) {
    if (!heap_reserve(h)) {
        return false;
    }
    h->items[h->size] = val;
    heap_sift_up(h, h->size);
    h->size++;
    return true;
}

static bool heap_pop(IntHeap *h, int64_t *out) {
    if (h->size == 0) {
        return false;
    }
    *out = h->items[0];
    h->size--;
    if (h->size > 0) {
        h->items[0] = h->items[h->size];
        heap_sift_down(h, 0);
    }
    return true;
}

typedef struct MedianFinder MedianFinder;

struct MedianFinder {
    IntHeap *max_heap;
    IntHeap *min_heap;
};

MedianFinder *median_finder_create(void) {
    MedianFinder *mf = malloc(sizeof *mf);
    if (mf == NULL) {
        return NULL;
    }
    mf->max_heap = heap_create(false);
    mf->min_heap = heap_create(true);
    if (mf->max_heap == NULL || mf->min_heap == NULL) {
        heap_destroy(mf->max_heap);
        heap_destroy(mf->min_heap);
        free(mf);
        return NULL;
    }
    return mf;
}

void median_finder_destroy(MedianFinder *mf) {
    if (mf == NULL) {
        return;
    }
    heap_destroy(mf->max_heap);
    heap_destroy(mf->min_heap);
    free(mf);
}

bool median_finder_add(MedianFinder *mf, int64_t val) {
    if (mf == NULL) {
        return false;
    }
    if (mf->max_heap->size == 0 || val <= mf->max_heap->items[0]) {
        if (!heap_push(mf->max_heap, val)) {
            return false;
        }
    } else {
        if (!heap_push(mf->min_heap, val)) {
            return false;
        }
    }

    if (mf->max_heap->size > mf->min_heap->size + 1) {
        int64_t top = 0;
        heap_pop(mf->max_heap, &top);
        if (!heap_push(mf->min_heap, top)) {
            heap_push(mf->max_heap, top);
            return false;
        }
    } else if (mf->max_heap->size < mf->min_heap->size) {
        int64_t top = 0;
        heap_pop(mf->min_heap, &top);
        if (!heap_push(mf->max_heap, top)) {
            heap_push(mf->min_heap, top);
            return false;
        }
    }
    return true;
}

bool median_finder_find(const MedianFinder *mf, double *out_median) {
    if (mf == NULL || out_median == NULL) {
        return false;
    }
    size_t total = mf->max_heap->size + mf->min_heap->size;
    if (total == 0) {
        return false;
    }
    if (total % 2 == 1) {
        *out_median = (double)mf->max_heap->items[0];
    } else {
        *out_median = ((double)mf->max_heap->items[0] + (double)mf->min_heap->items[0]) / 2.0;
    }
    return true;
}

size_t median_finder_size(const MedianFinder *mf) {
    if (mf == NULL) {
        return 0;
    }
    return mf->max_heap->size + mf->min_heap->size;
}

int main(void) {
    MedianFinder *mf = median_finder_create();
    assert(mf != NULL);
    double m = 0;
    assert(!median_finder_find(mf, &m));

    assert(median_finder_add(mf, 5));
    assert(median_finder_find(mf, &m) && fabs(m - 5.0) < 1e-9);

    assert(median_finder_add(mf, 15));
    assert(median_finder_find(mf, &m) && fabs(m - 10.0) < 1e-9);

    assert(median_finder_add(mf, 1));
    assert(median_finder_find(mf, &m) && fabs(m - 5.0) < 1e-9);

    assert(median_finder_add(mf, 3));
    assert(median_finder_find(mf, &m) && fabs(m - 4.0) < 1e-9);

    assert(median_finder_size(mf) == 4);
    median_finder_destroy(mf);

    return 0;
}
```

## 复杂度分析

- **时间复杂度**：
  - `median_finder_add`：单次插入至多涉及一次堆插入与一次跨堆转移，两个步骤均在高度为 $\log_2(n/2)$ 的堆中进行，时间复杂度严格为 $O(\log n)$。
  - `median_finder_find`：直接读取两个堆顶元素并做常数次浮点运算，时间复杂度严格为 $O(1)$。
  - `median_finder_size`：结构体成员相加，严格为 $O(1)$。
- **空间复杂度**：两个堆总共容纳全部 $n$ 个输入元素，额外空间复杂度为 $O(n)$。

## 延伸思考 / 模式总结

1. **分治与对顶模式的推广**：对顶堆不仅可用于中位数（即第 $50\%$ 分位数），通过调整两个堆的规模比例系数 $\alpha$，还可用于在线维护任意动态分位数（例如第 $90\%$ 分位数 P90、P99 延迟监控）。
2. **多堆协同的系统一致性**：在跨堆迁移元素时，若目标堆扩容失败，必须妥善回滚已弹出的元素，确保数据结构不丢失任何已有数据，体现了工业级 ADT 设计的事务一致性理念。

# 31607 - 索引优先队列与优先级降低操作

## 题目

普通二叉堆仅支持查看或弹出全局极值，无法在对数时间内随机访问或修改特定元素的优先级。实现**索引优先队列（Index Priority Queue，`IndexMinPQ`）**，为每个外部对象分配唯一的非负整数标识符 `id`，并支持在 $O(\log n)$ 时间内完成 `decrease-key`（降低优先级键值）操作：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct IndexMinPQ IndexMinPQ;

IndexMinPQ *index_min_pq_create(size_t max_index_count);
void index_min_pq_destroy(IndexMinPQ *pq);
bool index_min_pq_contains(const IndexMinPQ *pq, size_t id);
bool index_min_pq_push(IndexMinPQ *pq, size_t id, int64_t key);
bool index_min_pq_decrease_key(IndexMinPQ *pq, size_t id, int64_t new_key);
bool index_min_pq_pop_min(IndexMinPQ *pq, size_t *out_id, int64_t *out_key);
bool index_min_pq_peek_min(const IndexMinPQ *pq, size_t *out_id, int64_t *out_key);
size_t index_min_pq_size(const IndexMinPQ *pq);
```

要求：

1. 结构体能够容纳外部标识符范围在 $[0, \text{max\_index\_count} - 1]$ 内的对象。
2. 内部维护三组核心映射：
   - 堆数组 `pq`：记录当前位于堆位置 $k$ 的外部标识符 `id`；
   - 逆向映射数组 `qp`：记录外部标识符 `id` 当前位于堆中的位置 $k$；若未在堆中则标记为无效（`SIZE_MAX`）；
   - 键值数组 `keys`：记录外部标识符 `id` 当前绑定的优先级键值。
3. `index_min_pq_decrease_key` 检查 `id` 是否存在且新键值是否严格小于当前键值；若是，将键值更新并在 $O(\log n)$ 内执行上滤调整。
4. `index_min_pq_pop_min` 弹出当前键值最小的标识符及其键值，维护逆向索引的注销与堆序不变式。
5. 所有越界、重复推入、空堆弹出及非法指针操作必须做防御性校验，失败时不得修改任何内部状态。

## 解题思路

### 1. 为什么需要逆向映射数组 `qp`？

在图的最短路径算法（Dijkstra）或最小生成树算法（Prim）中，当一条更短的松弛路径出现时，我们需要更新已知顶点的距离下界（即 `decrease_key(v, new_dist)`）。
如果使用传统堆结构：
- 我们无法知道顶点 $v$ 当前究竟存储在堆数组的哪个下标位置；
- 必须耗费 $O(n)$ 时间进行全堆线性搜索定位，导致 Dijkstra 算法的时间复杂度退化为 $O(V^2)$。

索引优先队列通过引入**双向映射（Bijective Mapping）**彻底解决了这一定位痛点：
- `pq[k] = id`：从堆位置 $k$ 找到外部对象 `id`；
- `qp[id] = k`：从外部对象 `id` 瞬间以 $O(1)$ 寻址得到其在堆中的位置 $k$。
当需要降低 `id` 的键值时：
1. 通过 `qp[id]` 直接定位堆位置 $k$；
2. 更新 `keys[id] = new_key`；
3. 从堆位置 $k$ 发起上滤 `sift_up(k)`，仅需 $O(\log n)$ 步即可恢复小顶堆序。

```text
 外部标识符 id  -------- qp[id] -------> 堆内部下标 k
       ^                                    |
       |                                    |
       +---------------- pq[k] -------------+
```

### 2. 双向同步交换不变式

在常规堆中，交换两个堆元素只需交换数组中的键值：
```c
swap(&arr[i], &arr[j]);
```
但在索引优先队列中，堆数组 `pq` 存储的是 `id`，必须在交换 `pq[i]` 与 `pq[j]` 的同时，同步更新两者的逆向定位指针：
```c
void exch(IndexMinPQ *pq, size_t i, size_t j) {
    size_t id_i = pq->pq[i];
    size_t id_j = pq->pq[j];
    pq->pq[i] = id_j;
    pq->pq[j] = id_i;
    pq->qp[id_i] = j;
    pq->qp[id_j] = i;
}
```
这一操作使得双向映射不变式 `pq[qp[id]] == id` 在任何上滤与下滤过程中始终保持成立。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

typedef struct IndexMinPQ IndexMinPQ;

struct IndexMinPQ {
    size_t max_capacity;
    size_t size;
    size_t *pq;
    size_t *qp;
    int64_t *keys;
};

IndexMinPQ *index_min_pq_create(size_t max_index_count) {
    if (max_index_count == 0 || max_index_count > SIZE_MAX - 1) {
        return NULL;
    }
    IndexMinPQ *pq = malloc(sizeof *pq);
    if (pq == NULL) {
        return NULL;
    }
    pq->max_capacity = max_index_count;
    pq->size = 0;
    pq->pq = malloc((max_index_count + 1) * sizeof(size_t));
    pq->qp = malloc(max_index_count * sizeof(size_t));
    pq->keys = malloc(max_index_count * sizeof(int64_t));
    if (pq->pq == NULL || pq->qp == NULL || pq->keys == NULL) {
        free(pq->pq);
        free(pq->qp);
        free(pq->keys);
        free(pq);
        return NULL;
    }
    for (size_t i = 0; i < max_index_count; ++i) {
        pq->qp[i] = SIZE_MAX;
    }
    return pq;
}

void index_min_pq_destroy(IndexMinPQ *pq) {
    if (pq == NULL) {
        return;
    }
    free(pq->pq);
    free(pq->qp);
    free(pq->keys);
    free(pq);
}

bool index_min_pq_contains(const IndexMinPQ *pq, size_t id) {
    if (pq == NULL || id >= pq->max_capacity) {
        return false;
    }
    return pq->qp[id] != SIZE_MAX;
}

static bool greater(const IndexMinPQ *pq, size_t i, size_t j) {
    return pq->keys[pq->pq[i]] > pq->keys[pq->pq[j]];
}

static void exch(IndexMinPQ *pq, size_t i, size_t j) {
    size_t tmp = pq->pq[i];
    pq->pq[i] = pq->pq[j];
    pq->pq[j] = tmp;
    pq->qp[pq->pq[i]] = i;
    pq->qp[pq->pq[j]] = j;
}

static void sift_up(IndexMinPQ *pq, size_t k) {
    while (k > 1 && greater(pq, k / 2, k)) {
        exch(pq, k, k / 2);
        k = k / 2;
    }
}

static void sift_down(IndexMinPQ *pq, size_t k) {
    while (2 * k <= pq->size) {
        size_t j = 2 * k;
        if (j < pq->size && greater(pq, j, j + 1)) {
            j++;
        }
        if (!greater(pq, k, j)) {
            break;
        }
        exch(pq, k, j);
        k = j;
    }
}

bool index_min_pq_push(IndexMinPQ *pq, size_t id, int64_t key) {
    if (pq == NULL || id >= pq->max_capacity || index_min_pq_contains(pq, id)) {
        return false;
    }
    pq->size++;
    pq->qp[id] = pq->size;
    pq->pq[pq->size] = id;
    pq->keys[id] = key;
    sift_up(pq, pq->size);
    return true;
}

bool index_min_pq_decrease_key(IndexMinPQ *pq, size_t id, int64_t new_key) {
    if (pq == NULL || id >= pq->max_capacity || !index_min_pq_contains(pq, id)) {
        return false;
    }
    if (new_key >= pq->keys[id]) {
        return false;
    }
    pq->keys[id] = new_key;
    sift_up(pq, pq->qp[id]);
    return true;
}

bool index_min_pq_peek_min(const IndexMinPQ *pq, size_t *out_id, int64_t *out_key) {
    if (pq == NULL || pq->size == 0 || out_id == NULL || out_key == NULL) {
        return false;
    }
    *out_id = pq->pq[1];
    *out_key = pq->keys[pq->pq[1]];
    return true;
}

bool index_min_pq_pop_min(IndexMinPQ *pq, size_t *out_id, int64_t *out_key) {
    if (pq == NULL || pq->size == 0 || out_id == NULL || out_key == NULL) {
        return false;
    }
    size_t min_id = pq->pq[1];
    *out_id = min_id;
    *out_key = pq->keys[min_id];
    exch(pq, 1, pq->size);
    pq->size--;
    pq->qp[min_id] = SIZE_MAX;
    if (pq->size > 0) {
        sift_down(pq, 1);
    }
    return true;
}

size_t index_min_pq_size(const IndexMinPQ *pq) {
    if (pq == NULL) {
        return 0;
    }
    return pq->size;
}

int main(void) {
    IndexMinPQ *pq = index_min_pq_create(10);
    assert(pq != NULL);
    assert(index_min_pq_size(pq) == 0);

    assert(index_min_pq_push(pq, 2, 50));
    assert(index_min_pq_push(pq, 5, 20));
    assert(index_min_pq_push(pq, 1, 70));
    assert(index_min_pq_push(pq, 8, 40));

    assert(index_min_pq_contains(pq, 5));
    assert(!index_min_pq_contains(pq, 3));

    size_t min_id = 0;
    int64_t min_key = 0;
    assert(index_min_pq_peek_min(pq, &min_id, &min_key));
    assert(min_id == 5 && min_key == 20);

    assert(index_min_pq_decrease_key(pq, 1, 10));
    assert(index_min_pq_peek_min(pq, &min_id, &min_key));
    assert(min_id == 1 && min_key == 10);

    assert(index_min_pq_pop_min(pq, &min_id, &min_key));
    assert(min_id == 1 && min_key == 10);

    assert(index_min_pq_pop_min(pq, &min_id, &min_key));
    assert(min_id == 5 && min_key == 20);

    assert(index_min_pq_pop_min(pq, &min_id, &min_key));
    assert(min_id == 8 && min_key == 40);

    assert(index_min_pq_pop_min(pq, &min_id, &min_key));
    assert(min_id == 2 && min_key == 50);

    assert(index_min_pq_size(pq) == 0);
    assert(!index_min_pq_pop_min(pq, &min_id, &min_key));

    index_min_pq_destroy(pq);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：
  - `index_min_pq_contains`：通过 `qp[id]` 做直接数组索引，严格为 $O(1)$。
  - `index_min_pq_decrease_key`：通过 `qp` 数组在 $O(1)$ 时间内定位元素在堆中的位置，随后沿树高执行上滤，时间复杂度严格为 $O(\log n)$。
  - `index_min_pq_push` 与 `index_min_pq_pop_min`：涉及一次上滤或下滤，时间复杂度为 $O(\log n)$。
  - `index_min_pq_peek_min` 与 `index_min_pq_size`：直接访问堆顶，时间复杂度为 $O(1)$。
- **空间复杂度**：为容量为 $M$ 的标识符分配了 `pq`、`qp` 与 `keys` 三个连续数组，空间复杂度为 $O(M)$。

## 延伸思考 / 模式总结

1. **图算法的效率支柱**：在经典的 Dijkstra 算法中，若使用不支持快速 `decrease-key` 的普通二叉堆，只能通过“重复插入惰性删除”策略处理更优路径，导致堆中元素膨胀至 $O(E)$ 个，且需占用额外内存；使用索引优先队列则能将堆内元素严格限制在 $V$ 个，且支持真正的就地更新，使得 Dijkstra 算法运行时间严格受限在 $O(E \log V)$。
2. **句柄模式与反向定位思想**：索引优先队列的本质是向外部客户端发放一个句柄（即 `id`），并在内部维护“外部句柄到内部物理槽位”的反向索引表。这种设计模式在虚拟内存页表、操作系统文件描述符表以及网络连接追踪中极为普遍。

# 36006 - 有向无环图（DAG）的最长路与关键路径（PERT 图事件最早/最迟发生时间）

## 题目

在工程管理中，**边表示活动网络（AOE 网 / PERT 图）**用于规划大型项目的进度。图中的顶点表示**事件（Event）**，有向边表示**活动（Activity）**，边上的权值表示完成该项活动所需的**持续时间（Duration）**。

求整个工程项目完成所需的**最短总工期**（即从源点到汇点的最长路径长度），并确定哪些活动属于**关键活动（Critical Activity）**。关键活动是指其延误将直接导致整个工程工期延长的活动。

函数签名如下：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct {
    size_t from;
    size_t to;
    uint64_t duration;
} Activity;

bool critical_path_analysis(
    size_t event_count,
    const Activity *activities,
    size_t activity_count,
    uint64_t *out_project_duration,
    bool *out_is_critical,
    bool *out_is_dag
);
```

要求：
1. 参数无效时返回 `false`。若图中存在有向环，设置 `*out_is_dag = false` 并返回 `true`。
2. 事件编号为 $0, 1, \dots, event\_count - 1$。源点为入度为 0 的事件，汇点为出度为 0 的事件。
3. 计算每个事件的最早发生时间 $ve[u]$（按拓扑序顺推）与最迟发生时间 $vl[u]$（按逆拓扑序逆推）。
4. 对每条活动边，若活动的最早开始时间等于最迟开始时间（$ve[u] == vl[v] - duration$），则标记该活动为关键活动（`out_is_critical[i] = true`）。

## 解题思路

1. **事件最早发生时间 $ve$（顺推动态规划）**：
   事件 $u$ 的最早发生时间 $ve[u]$ 决定了所有以 $u$ 为起点的活动的最早开工时间。
   - 源点：$ve[source] = 0$；
   - 按照**拓扑排序顺序**向前推进计算：
     $$ve[v] = \max_{(u, v) \in E} \{ ve[u] + duration(u, v) \}$$
   工程总工期即为所有汇点事件最早发生时间的最大值：$T = \max_u ve[u]$。

2. **事件最迟发生时间 $vl$（逆推动态规划）**：
   事件 $u$ 的最迟发生时间 $vl[u]$ 是指在不推迟整个工程工期的前提下，该事件必须发生的最晚时刻。
   - 汇点：$vl[sink] = T$；其余事件初始全设为 $T$；
   - 按照**逆拓扑排序顺序**向后逆推计算：
     $$vl[u] = \min_{(u, v) \in E} \{ vl[v] - duration(u, v) \}$$

3. **关键活动判定**：
   对于活动边 $a_i = (u, v)$：
   - 最早可能开始时间 $e(a_i) = ve[u]$；
   - 最迟必须开始时间 $l(a_i) = vl[v] - duration(a_i)$；
   - 时间余量（Slack）为 $l(a_i) - e(a_i)$。
   若时间余量为 0（即 $e(a_i) == l(a_i)$），则说明该活动没有任何缓冲余地，必须准时完成，是**关键活动**。所有关键活动串联起来的路径即为**关键路径**。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    size_t from;
    size_t to;
    uint64_t duration;
} Activity;

typedef struct EdgeNode EdgeNode;
struct EdgeNode {
    size_t to;
    uint64_t weight;
    size_t act_idx;
    EdgeNode *next;
};

bool critical_path_analysis(
    size_t event_count,
    const Activity *activities,
    size_t activity_count,
    uint64_t *out_project_duration,
    bool *out_is_critical,
    bool *out_is_dag
) {
    if (out_project_duration == NULL || out_is_critical == NULL ||
        out_is_dag == NULL || (activity_count > 0 && activities == NULL)) {
        return false;
    }
    if (event_count <= 1) {
        *out_project_duration = 0;
        *out_is_dag = true;
        return true;
    }

    EdgeNode **adj = calloc(event_count, sizeof(EdgeNode*));
    EdgeNode *pool = malloc(activity_count * sizeof(EdgeNode));
    size_t *in_degree = calloc(event_count, sizeof(size_t));
    size_t *topo_order = malloc(event_count * sizeof(size_t));
    size_t *queue = malloc(event_count * sizeof(size_t));
    uint64_t *ve = calloc(event_count, sizeof(uint64_t));
    uint64_t *vl = malloc(event_count * sizeof(uint64_t));

    if (adj == NULL || (activity_count > 0 && pool == NULL) || in_degree == NULL ||
        topo_order == NULL || queue == NULL || ve == NULL || vl == NULL) {
        free(adj); free(pool); free(in_degree); free(topo_order);
        free(queue); free(ve); free(vl);
        return false;
    }

    // 建立正向邻接表与入度
    for (size_t i = 0; i < activity_count; ++i) {
        size_t u = activities[i].from;
        size_t v = activities[i].to;
        if (u >= event_count || v >= event_count) {
            free(adj); free(pool); free(in_degree); free(topo_order);
            free(queue); free(ve); free(vl);
            return false;
        }
        pool[i].to = v;
        pool[i].weight = activities[i].duration;
        pool[i].act_idx = i;
        pool[i].next = adj[u];
        adj[u] = &pool[i];
        in_degree[v]++;
    }

    // Kahn 算法拓扑排序
    size_t head = 0;
    size_t tail = 0;
    for (size_t i = 0; i < event_count; ++i) {
        if (in_degree[i] == 0) {
            queue[tail++] = i;
        }
    }

    size_t topo_len = 0;
    while (head < tail) {
        size_t u = queue[head++];
        topo_order[topo_len++] = u;

        for (const EdgeNode *edge = adj[u]; edge != NULL; edge = edge->next) {
            size_t v = edge->to;
            in_degree[v]--;
            if (in_degree[v] == 0) {
                queue[tail++] = v;
            }
        }
    }

    if (topo_len < event_count) {
        *out_is_dag = false; // 存在环
        free(adj); free(pool); free(in_degree); free(topo_order);
        free(queue); free(ve); free(vl);
        return true;
    }
    *out_is_dag = true;

    // 顺推计算 ve (事件最早发生时间)
    uint64_t max_duration = 0;
    for (size_t i = 0; i < event_count; ++i) {
        size_t u = topo_order[i];
        for (const EdgeNode *edge = adj[u]; edge != NULL; edge = edge->next) {
            size_t v = edge->to;
            uint64_t cand = ve[u] + edge->weight;
            if (cand > ve[v]) {
                ve[v] = cand;
            }
        }
        if (ve[u] > max_duration) {
            max_duration = ve[u];
        }
    }

    *out_project_duration = max_duration;

    // 逆推计算 vl (事件最迟发生时间)
    for (size_t i = 0; i < event_count; ++i) {
        vl[i] = max_duration;
    }

    for (size_t step = event_count; step > 0; --step) {
        size_t u = topo_order[step - 1];
        for (const EdgeNode *edge = adj[u]; edge != NULL; edge = edge->next) {
            size_t v = edge->to;
            uint64_t cand = vl[v] - edge->weight;
            if (cand < vl[u]) {
                vl[u] = cand;
            }
        }
    }

    // 判定关键活动
    for (size_t i = 0; i < activity_count; ++i) {
        size_t u = activities[i].from;
        size_t v = activities[i].to;
        uint64_t dur = activities[i].duration;
        uint64_t e_early = ve[u];
        uint64_t l_late = vl[v] - dur;
        out_is_critical[i] = (e_early == l_late);
    }

    free(adj);
    free(pool);
    free(in_degree);
    free(topo_order);
    free(queue);
    free(ve);
    free(vl);
    return true;
}

int main(void) {
    uint64_t duration = 0;
    bool is_critical[16];
    bool is_dag = false;

    // 经典工程网络图 (6 个事件，8 项活动)
    // 活动 0: 0 -> 1 (dur 3)
    // 活动 1: 0 -> 2 (dur 2)
    // 活动 2: 1 -> 3 (dur 2)
    // 活动 3: 1 -> 4 (dur 4)
    // 活动 4: 2 -> 3 (dur 4)
    // 活动 5: 2 -> 5 (dur 3)
    // 活动 6: 3 -> 5 (dur 2)
    // 活动 7: 4 -> 5 (dur 1)
    // 关键路径为 0 -> 2 -> 3 -> 5, 总工期为 2 + 4 + 2 = 8
    const Activity acts[8] = {
        {0, 1, 3}, {0, 2, 2}, {1, 3, 2}, {1, 4, 4},
        {2, 3, 4}, {2, 5, 3}, {3, 5, 2}, {4, 5, 1}
    };

    assert(critical_path_analysis(6, acts, 8, &duration, is_critical, &is_dag));
    assert(is_dag);
    assert(duration == 8);

    // 校验关键活动:
    // 本例存在两条并列的关键路径: 0 -> 1 -> 4 -> 5 与 0 -> 2 -> 3 -> 5
    // 活动 0, 1, 3, 4, 6, 7 均为关键活动
    assert(is_critical[0] == true);
    assert(is_critical[1] == true);
    assert(is_critical[4] == true);
    assert(is_critical[6] == true);
    // 活动 2 (1->3) 与 活动 5 (2->5) 不是关键活动 (有余量)
    assert(is_critical[2] == false);
    assert(is_critical[5] == false);

    return 0;
}
```

## 复杂度分析

- **时间复杂度**：$O(V + E)$。拓扑排序耗时 $O(V + E)$，前向顺推与后向逆推各遍历图一次，总体在线性时间内完成。
- **空间复杂度**：$O(V + E)$。需要维护邻接表、正向时间数组与逆向时间数组。

## 延伸思考 / 模式总结

关键路径算法是运筹学与项目管理系统的基石。通过双向拓扑递推，算法将复杂的网络工期优化转化为局部事件的时间差检验。该算法亦被用于现代超大规模集成电路（VLSI）的时序分析（Static Timing Analysis）以识别关键延迟路径。

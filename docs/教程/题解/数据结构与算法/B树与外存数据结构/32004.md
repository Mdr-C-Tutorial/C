# 32004 - B 树节点删除之借键与合并

## 题目描述

实现 B 树完整的结点删除算法 `btree_delete`，采用自顶向下预先丰满策略，处理向左/右兄弟借键（Borrow / Rotate）以及与兄弟父子合并（Merge）的所有边界情形。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct BTree BTree;

bool btree_delete(
    BTree *tree,
    int64_t key,
    bool *out_removed
);
```

要求：

1. 当待删关键字存在于内部结点时，不能直接物理擦除，必须通过其前驱（左子树最大键）或后继（右子树最小键）替代，转化为叶子层删除。
2. 在沿树下降搜索待删结点的过程中，只要即将进入的子结点仅有 $t - 1$ 个键（即处于下溢的临界边缘）：
   - 若左（或右）相邻兄弟拥有至少 $t$ 个键，通过父结点借一个键（三方旋转）；
   - 若左右相邻兄弟均只有 $t - 1$ 个键，将父结点的分隔键连同兄弟结点合并为一个包含 $2t - 1$ 个键的满结点；
   - 由此确保进入的子结点至少包含 $t$ 个键，保证后续删除决不导致非法欠饱和。
3. 若根结点的全部关键字在合并中被抽取下沉，且根结点不为叶结点，必须收缩整棵树的高度，将唯一的孩子提升为新根。
4. 正确更新树的规模 `tree->size`，且删除后必须满足 B 树的全部结构与阶数不变式。

## 解题思路

### 1. 为什么删除必须预先补充（Borrow & Merge）？

在最小度数为 $t$ 的 B 树中，非根结点的键数下限为 $t - 1$。
如果像普通 BST 那样直接删到底层叶子，一旦叶子只剩 $t - 1$ 个键，删除后就会发生下溢（Underflow）。若此时再自底向上沿路径回溯借键或合并，可能会一路连锁合并到根结点，造成难以控制的回溯复杂度与并发锁阻塞。
CLRS 提出的优雅策略是：**在自顶向下搜索待删键的下探过程中，每当发现即将进入的子结点只有 $t - 1$ 个键时，立即预先将其“充盈”（通过向兄弟借键或合并兄弟）到至少 $t$ 个键**。这样后续无论是直接删除叶子还是递归替换，都绝不可能产生下溢！

### 2. 核心补救操作的几何推导

设当前内部结点为 $P$，即将进入第 $i$ 个孩子 $C_i$（当前包含 $t - 1$ 个键）：
- **情形 A：向左兄弟借键 (`borrow_from_prev`)**：
  若左兄弟 $C_{i-1}$ 拥有至少 $t$ 个键：
  1. 父结点的分隔键 $P.\text{keys}[i-1]$ 下沉至 $C_i$ 的最左侧；
  2. 左兄弟的最大键 $C_{i-1}.\text{keys}[\text{last}]$ 上升取代 $P.\text{keys}[i-1]$；
  3. 左兄弟的最右子指针转移到 $C_i$ 的最左侧。
- **情形 B：向右兄弟借键 (`borrow_from_next`)**：
  若右兄弟 $C_{i+1}$ 拥有至少 $t$ 个键：
  1. 父结点的分隔键 $P.\text{keys}[i]$ 下沉至 $C_i$ 的最右侧；
  2. 右兄弟的最小键 $C_{i+1}.\text{keys}[0]$ 上升取代 $P.\text{keys}[i]$；
  3. 右兄弟的最左子指针转移到 $C_i$ 的最右侧。
- **情形 C：父子合并 (`merge_nodes`)**：
  若左右兄弟均只有 $t - 1$ 个键，无法借调：
  1. 将父结点的分隔键 $P.\text{keys}[i]$ 下拉，与 $C_i$ 以及 $C_{i+1}$ 合并；
  2. 合并后的新结点恰好包含 $(t - 1) + 1 + (t - 1) = 2t - 1$ 个键，成为一个合法的满结点；
  3. 释放 $C_{i+1}$。此时父结点 $P$ 减少了一个键和一个孩子。若 $P$ 是根且键数变为 0，整棵树高度缩减 1。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct BTreeNode BTreeNode;

struct BTreeNode {
    bool leaf;
    size_t key_count;
    int64_t *keys;
    BTreeNode **children;
};

typedef struct BTree {
    BTreeNode *root;
    size_t t;
    size_t size;
} BTree;

static BTreeNode *create_node(size_t t, bool leaf) {
    BTreeNode *n = (BTreeNode *)malloc(sizeof(BTreeNode));
    assert(n != NULL);
    n->leaf = leaf;
    n->key_count = 0;
    n->keys = (int64_t *)malloc(sizeof(int64_t) * (2 * t - 1));
    assert(n->keys != NULL);
    n->children = (BTreeNode **)malloc(sizeof(BTreeNode *) * (2 * t));
    assert(n->children != NULL);
    for (size_t i = 0; i < 2 * t; ++i) {
        n->children[i] = NULL;
    }
    return n;
}

static void free_nodes(BTreeNode *n) {
    if (n == NULL) {
        return;
    }
    if (!n->leaf) {
        for (size_t i = 0; i <= n->key_count; ++i) {
            free_nodes(n->children[i]);
        }
    }
    free(n->keys);
    free(n->children);
    free(n);
}

BTree *btree_create(size_t t) {
    if (t < 2) {
        return NULL;
    }
    BTree *tree = (BTree *)malloc(sizeof(BTree));
    if (tree == NULL) {
        return NULL;
    }
    tree->t = t;
    tree->root = create_node(t, true);
    tree->size = 0;
    return tree;
}

void btree_destroy(BTree *tree) {
    if (tree == NULL) {
        return;
    }
    free_nodes(tree->root);
    free(tree);
}

static void btree_split_child(BTreeNode *parent, size_t i, size_t t) {
    BTreeNode *y = parent->children[i];
    BTreeNode *z = create_node(t, y->leaf);
    z->key_count = t - 1;

    for (size_t j = 0; j < t - 1; ++j) {
        z->keys[j] = y->keys[j + t];
    }
    if (!y->leaf) {
        for (size_t j = 0; j < t; ++j) {
            z->children[j] = y->children[j + t];
            y->children[j + t] = NULL;
        }
    }
    y->key_count = t - 1;

    for (size_t j = parent->key_count + 1; j > i + 1; --j) {
        parent->children[j] = parent->children[j - 1];
    }
    parent->children[i + 1] = z;

    for (size_t j = parent->key_count; j > i; --j) {
        parent->keys[j] = parent->keys[j - 1];
    }
    parent->keys[i] = y->keys[t - 1];
    parent->key_count++;
}

static bool btree_insert_non_full(BTreeNode *node, int64_t key, size_t t, bool *out_inserted) {
    size_t i = node->key_count;
    if (node->leaf) {
        while (i > 0 && key < node->keys[i - 1]) {
            node->keys[i] = node->keys[i - 1];
            --i;
        }
        if (i > 0 && node->keys[i - 1] == key) {
            for (size_t k = i; k < node->key_count; ++k) {
                node->keys[k] = node->keys[k + 1];
            }
            *out_inserted = false;
            return true;
        }
        node->keys[i] = key;
        node->key_count++;
        *out_inserted = true;
        return true;
    } else {
        while (i > 0 && key < node->keys[i - 1]) {
            --i;
        }
        if (i > 0 && node->keys[i - 1] == key) {
            *out_inserted = false;
            return true;
        }
        if (node->children[i]->key_count == 2 * t - 1) {
            btree_split_child(node, i, t);
            if (key == node->keys[i]) {
                *out_inserted = false;
                return true;
            }
            if (key > node->keys[i]) {
                i++;
            }
        }
        return btree_insert_non_full(node->children[i], key, t, out_inserted);
    }
}

bool btree_insert(BTree *tree, int64_t key, bool *out_inserted) {
    if (tree == NULL || out_inserted == NULL) {
        return false;
    }
    BTreeNode *r = tree->root;
    if (r->key_count == 2 * tree->t - 1) {
        BTreeNode *s = create_node(tree->t, false);
        tree->root = s;
        s->children[0] = r;
        btree_split_child(s, 0, tree->t);
        bool ins = false;
        bool ok = btree_insert_non_full(s, key, tree->t, &ins);
        if (ok && ins) {
            tree->size++;
        }
        *out_inserted = ins;
        return ok;
    } else {
        bool ins = false;
        bool ok = btree_insert_non_full(r, key, tree->t, &ins);
        if (ok && ins) {
            tree->size++;
        }
        *out_inserted = ins;
        return ok;
    }
}

static size_t find_key_slot(const BTreeNode *node, int64_t key) {
    size_t idx = 0;
    while (idx < node->key_count && node->keys[idx] < key) {
        idx++;
    }
    return idx;
}

static void merge_nodes(BTreeNode *parent, size_t idx, size_t t) {
    BTreeNode *child = parent->children[idx];
    BTreeNode *sibling = parent->children[idx + 1];

    /* 将父结点的键下移 */
    child->keys[t - 1] = parent->keys[idx];

    /* 拷贝右兄弟所有关键字 */
    for (size_t j = 0; j < sibling->key_count; ++j) {
        child->keys[j + t] = sibling->keys[j];
    }
    if (!child->leaf) {
        for (size_t j = 0; j <= sibling->key_count; ++j) {
            child->children[j + t] = sibling->children[j];
        }
    }
    child->key_count += sibling->key_count + 1;

    /* 父结点的键与子指针前移收缩 */
    for (size_t j = idx + 1; j < parent->key_count; ++j) {
        parent->keys[j - 1] = parent->keys[j];
    }
    for (size_t j = idx + 2; j <= parent->key_count; ++j) {
        parent->children[j - 1] = parent->children[j];
    }
    parent->key_count--;

    free(sibling->keys);
    free(sibling->children);
    free(sibling);
}

static void borrow_from_prev(BTreeNode *parent, size_t idx) {
    BTreeNode *child = parent->children[idx];
    BTreeNode *sibling = parent->children[idx - 1];

    for (size_t j = child->key_count; j > 0; --j) {
        child->keys[j] = child->keys[j - 1];
    }
    if (!child->leaf) {
        for (size_t j = child->key_count + 1; j > 0; --j) {
            child->children[j] = child->children[j - 1];
        }
    }

    child->keys[0] = parent->keys[idx - 1];
    if (!child->leaf) {
        child->children[0] = sibling->children[sibling->key_count];
    }

    parent->keys[idx - 1] = sibling->keys[sibling->key_count - 1];
    child->key_count++;
    sibling->key_count--;
}

static void borrow_from_next(BTreeNode *parent, size_t idx) {
    BTreeNode *child = parent->children[idx];
    BTreeNode *sibling = parent->children[idx + 1];

    child->keys[child->key_count] = parent->keys[idx];
    if (!child->leaf) {
        child->children[child->key_count + 1] = sibling->children[0];
    }

    parent->keys[idx] = sibling->keys[0];

    for (size_t j = 1; j < sibling->key_count; ++j) {
        sibling->keys[j - 1] = sibling->keys[j];
    }
    if (!sibling->leaf) {
        for (size_t j = 1; j <= sibling->key_count; ++j) {
            sibling->children[j - 1] = sibling->children[j];
        }
    }

    child->key_count++;
    sibling->key_count--;
}

static void fill_child(BTreeNode *node, size_t idx, size_t t) {
    if (idx != 0 && node->children[idx - 1]->key_count >= t) {
        borrow_from_prev(node, idx);
    } else if (idx != node->key_count && node->children[idx + 1]->key_count >= t) {
        borrow_from_next(node, idx);
    } else {
        if (idx != node->key_count) {
            merge_nodes(node, idx, t);
        } else {
            merge_nodes(node, idx - 1, t);
        }
    }
}

static int64_t get_predecessor(BTreeNode *node) {
    BTreeNode *curr = node;
    while (!curr->leaf) {
        curr = curr->children[curr->key_count];
    }
    return curr->keys[curr->key_count - 1];
}

static int64_t get_successor(BTreeNode *node) {
    BTreeNode *curr = node;
    while (!curr->leaf) {
        curr = curr->children[0];
    }
    return curr->keys[0];
}

static bool btree_delete_node(BTreeNode *node, int64_t key, size_t t) {
    size_t idx = find_key_slot(node, key);

    if (idx < node->key_count && node->keys[idx] == key) {
        if (node->leaf) {
            /* 情形 1：关键字位于叶结点中 */
            for (size_t j = idx + 1; j < node->key_count; ++j) {
                node->keys[j - 1] = node->keys[j];
            }
            node->key_count--;
            return true;
        } else {
            /* 情形 2：关键字位于内部结点中 */
            if (node->children[idx]->key_count >= t) {
                int64_t pred = get_predecessor(node->children[idx]);
                node->keys[idx] = pred;
                return btree_delete_node(node->children[idx], pred, t);
            } else if (node->children[idx + 1]->key_count >= t) {
                int64_t succ = get_successor(node->children[idx + 1]);
                node->keys[idx] = succ;
                return btree_delete_node(node->children[idx + 1], succ, t);
            } else {
                merge_nodes(node, idx, t);
                return btree_delete_node(node->children[idx], key, t);
            }
        }
    } else {
        if (node->leaf) {
            return false;
        }
        bool flag = (idx == node->key_count);
        if (node->children[idx]->key_count < t) {
            fill_child(node, idx, t);
        }
        if (flag && idx > node->key_count) {
            return btree_delete_node(node->children[idx - 1], key, t);
        } else {
            return btree_delete_node(node->children[idx], key, t);
        }
    }
}

bool btree_delete(BTree *tree, int64_t key, bool *out_removed) {
    if (tree == NULL || out_removed == NULL) {
        return false;
    }
    if (tree->root->key_count == 0 && tree->root->leaf) {
        *out_removed = false;
        return true;
    }

    bool res = btree_delete_node(tree->root, key, tree->t);
    *out_removed = res;
    if (res) {
        tree->size--;
        /* 根结点键数被抽空且有孩子时，收缩树高度 */
        if (tree->root->key_count == 0 && !tree->root->leaf) {
            BTreeNode *tmp = tree->root;
            tree->root = tree->root->children[0];
            free(tmp->keys);
            free(tmp->children);
            free(tmp);
        }
    }
    return true;
}

static bool validate_helper(
    const BTreeNode *node,
    size_t t,
    bool is_root,
    size_t current_depth,
    size_t *expected_leaf_depth,
    bool *has_leaf_depth,
    bool has_low,
    int64_t low,
    bool has_high,
    int64_t high
) {
    if (node == NULL) {
        return false;
    }
    size_t max_k = 2 * t - 1;
    size_t min_k = is_root ? 1 : (t - 1);
    if (is_root && node->key_count == 0 && node->leaf) {
        return true;
    }
    if (node->key_count > max_k || node->key_count < min_k) {
        return false;
    }

    for (size_t i = 0; i < node->key_count; ++i) {
        if ((has_low && node->keys[i] <= low) || (has_high && node->keys[i] >= high)) {
            return false;
        }
        if (i > 0 && node->keys[i] <= node->keys[i - 1]) {
            return false;
        }
    }

    if (node->leaf) {
        if (!*has_leaf_depth) {
            *has_leaf_depth = true;
            *expected_leaf_depth = current_depth;
        } else if (*expected_leaf_depth != current_depth) {
            return false;
        }
        return true;
    }

    for (size_t i = 0; i <= node->key_count; ++i) {
        if (node->children[i] == NULL) {
            return false;
        }
        bool child_has_low = (i > 0) ? true : has_low;
        int64_t child_low = (i > 0) ? node->keys[i - 1] : low;

        bool child_has_high = (i < node->key_count) ? true : has_high;
        int64_t child_high = (i < node->key_count) ? node->keys[i] : high;

        if (!validate_helper(
                node->children[i], t, false, current_depth + 1,
                expected_leaf_depth, has_leaf_depth,
                child_has_low, child_low, child_has_high, child_high)) {
            return false;
        }
    }
    return true;
}

bool btree_validate(const BTree *tree) {
    if (tree == NULL || tree->t < 2) {
        return false;
    }
    size_t expected_depth = 0;
    bool has_depth = false;
    return validate_helper(tree->root, tree->t, true, 0, &expected_depth, &has_depth, false, 0, false, 0);
}

int main(void) {
    BTree *tree = btree_create(3);
    assert(tree != NULL);

    bool ins = false;
    for (int64_t i = 1; i <= 60; ++i) {
        assert(btree_insert(tree, i, &ins) && ins);
        assert(btree_validate(tree));
    }
    assert(tree->size == 60);

    bool rem = false;
    /* 阶段 1：交替删除偶数键，频繁触发借键与合并 */
    for (int64_t i = 2; i <= 60; i += 2) {
        assert(btree_delete(tree, i, &rem) && rem);
        assert(btree_validate(tree));
    }
    assert(tree->size == 30);

    /* 阶段 2：删除奇数键直到全树清空，测试根结点高度缩减 */
    for (int64_t i = 1; i <= 60; i += 2) {
        assert(btree_delete(tree, i, &rem) && rem);
        assert(btree_validate(tree));
    }
    assert(tree->size == 0);
    assert(tree->root->key_count == 0 && tree->root->leaf);

    btree_destroy(tree);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：从根到叶只做单一向下行程。每一层在进入前至多触发常数次借键或合并（移动 $O(t)$ 个键）。因此单次删除的时间复杂度为严格的 $O(t \log_t n)$；对于固定的阶数 $t$，时间复杂度为 $O(\log n)$。
- **空间复杂度**：递归栈深度为树的高度 $h = O(\log n)$，额外空间复杂度为 $O(\log n)$。

## 延伸思考 / 模式总结

自顶向下借键合并算法将“预防性维护”发挥到了极致：不在发生下溢后惊慌失措地向上回溯拆补，而是在下降探索的路上主动扫清一切隐患。这种模式不仅彻底免除了维护回溯父指针的负担，更为外存数据库的页锁（Page Latch）解耦提供了极佳的理论支撑。

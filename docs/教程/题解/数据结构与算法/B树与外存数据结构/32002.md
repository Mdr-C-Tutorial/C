# 32002 - B 树满节点分裂与根节点上升裂变

## 题目描述

实现 B 树满子结点分裂（Node Split）与根结点上升裂变（Root Elevation）原子操作，并提供全树叶子等深性与多路有序性验证器。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct BTreeNode BTreeNode;

struct BTreeNode {
    bool leaf;
    size_t key_count;
    int64_t *keys;
    BTreeNode **children;
};

void btree_split_child(
    BTreeNode *parent,
    size_t i,
    BTreeNode *y,
    size_t t
);

BTreeNode *btree_split_root(
    BTreeNode *old_root,
    size_t t
);

bool btree_validate(
    const BTreeNode *root,
    size_t t
);
```

要求：

1. 当子结点 $y = \text{parent}.\text{children}[i]$ 达到满容量（恰好包含 $2t - 1$ 个关键字）时，执行分裂操作：
   - 将中位数键 $y.\text{keys}[t - 1]$ 上升提升到父结点中；
   - 分配新结点 $z$（右兄弟），将 $y$ 中位于中位数右侧的 $t - 1$ 个较大键以及 $t$ 个子树指针转移到 $z$ 中；
   - 原结点 $y$ 保留左侧 $t - 1$ 个较小键以及 $t$ 个子树指针；
   - 父结点安全向右搬移后续键与孩子指针，容纳新提升的中位数与指向 $z$ 的指针。
2. 当整个 B 树的根结点变满时，通过 `btree_split_root` 创建空新根并执行分裂，使整棵树的高度安全增加 1。
3. `btree_validate` 验证每个结点的键数范围、键的严格单调递增性、所有叶结点的绝对深度完全相等，以及指针引用的合法性。

## 解题思路

### 1. 裂变几何对称性

一个包含 $2t - 1$ 个键的满结点，若包含子指针则必有 $2t$ 个子指针。
将其切分为三部分：
1. **左半部分**：包含下标 $0 \dots t - 2$ 共 $t - 1$ 个键，以及下标 $0 \dots t - 1$ 共 $t$ 个子指针。原结点 $y$ 保留这部分；
2. **中位数项**：下标为 $t - 1$ 的键 $y.\text{keys}[t - 1]$。该项不留在底层，而是向上提升插入到父结点的槽位 $i$ 中；
3. **右半部分**：包含下标 $t \dots 2t - 2$ 共 $t - 1$ 个键，以及下标 $t \dots 2t - 1$ 共 $t$ 个子指针。新创建的右兄弟结点 $z$ 接收这部分。

分裂后，$y$ 和 $z$ 各自包含 $t - 1$ 个键，均严格满足 B 树结点的最小容量不变式！

### 2. 根结点分裂与树高增长

B 树的高度**只在且仅在根结点分裂时增加**：
- 当根结点已满且待插新键必须穿透时，分配一个新的内部结点 `new_root`，此时 `new_root->key_count = 0`；
- 令 `new_root->children[0] = old_root`；
- 调用 `btree_split_child(new_root, 0, old_root, t)`；
- 此时 `old_root` 被分裂为左右两个半满孩子，中位数成为 `new_root` 的唯一键。树的高度恰好增加 1，且所有叶子结点的深度同步加 1，依然保持严格等深！

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct BTreeNode BTreeNode;

struct BTreeNode {
    bool leaf;
    size_t key_count;
    int64_t *keys;
    BTreeNode **children;
};

static BTreeNode *create_node(size_t t, bool leaf) {
    BTreeNode *n = (BTreeNode *)malloc(sizeof(BTreeNode));
    assert(n != NULL);
    n->leaf = leaf;
    n->key_count = 0;
    n->keys = (int64_t *)malloc(sizeof(int64_t) * (2 * t - 1));
    assert(n->keys != NULL);
    n->children = (BTreeNode **)malloc(sizeof(BTreeNode *) * (2 * t));
    assert(n->children != NULL);
    for (size_t i = 0; i < 2 * t; ++i) {
        n->children[i] = NULL;
    }
    return n;
}

static void free_tree(BTreeNode *n) {
    if (n == NULL) {
        return;
    }
    if (!n->leaf) {
        for (size_t i = 0; i <= n->key_count; ++i) {
            free_tree(n->children[i]);
        }
    }
    free(n->keys);
    free(n->children);
    free(n);
}

void btree_split_child(
    BTreeNode *parent,
    size_t i,
    BTreeNode *y,
    size_t t
) {
    assert(y->key_count == 2 * t - 1);
    assert(parent->key_count < 2 * t - 1);

    BTreeNode *z = create_node(t, y->leaf);
    z->key_count = t - 1;

    /* 将 y 的右半段关键字复制到新结点 z */
    for (size_t j = 0; j < t - 1; ++j) {
        z->keys[j] = y->keys[j + t];
    }

    /* 若非叶子，将 y 的右半段子树指针复制到 z */
    if (!y->leaf) {
        for (size_t j = 0; j < t; ++j) {
            z->children[j] = y->children[j + t];
            y->children[j + t] = NULL;
        }
    }

    y->key_count = t - 1;

    /* 父结点向右腾出孩子指针槽位 */
    for (size_t j = parent->key_count + 1; j > i + 1; --j) {
        parent->children[j] = parent->children[j - 1];
    }
    parent->children[i + 1] = z;

    /* 父结点向右腾出关键字槽位，并插入中位数 */
    for (size_t j = parent->key_count; j > i; --j) {
        parent->keys[j] = parent->keys[j - 1];
    }
    parent->keys[i] = y->keys[t - 1];
    parent->key_count++;
}

BTreeNode *btree_split_root(BTreeNode *old_root, size_t t) {
    assert(old_root->key_count == 2 * t - 1);
    BTreeNode *new_root = create_node(t, false);
    new_root->children[0] = old_root;
    btree_split_child(new_root, 0, old_root, t);
    return new_root;
}

static bool validate_helper(
    const BTreeNode *node,
    size_t t,
    bool is_root,
    size_t current_depth,
    size_t *expected_leaf_depth,
    bool *has_leaf_depth,
    bool has_low,
    int64_t low,
    bool has_high,
    int64_t high
) {
    if (node == NULL) {
        return false;
    }

    size_t max_k = 2 * t - 1;
    size_t min_k = is_root ? 1 : (t - 1);
    if (node->key_count > max_k || node->key_count < min_k) {
        return false;
    }

    /* 关键字有序性与区间边界校验 */
    for (size_t i = 0; i < node->key_count; ++i) {
        if ((has_low && node->keys[i] <= low) ||
            (has_high && node->keys[i] >= high)) {
            return false;
        }
        if (i > 0 && node->keys[i] <= node->keys[i - 1]) {
            return false;
        }
    }

    /* 叶子结点：校验全树叶子等深性 */
    if (node->leaf) {
        if (!*has_leaf_depth) {
            *has_leaf_depth = true;
            *expected_leaf_depth = current_depth;
        } else if (*expected_leaf_depth != current_depth) {
            return false;
        }
        return true;
    }

    /* 内部结点：递归校验所有孩子分支 */
    for (size_t i = 0; i <= node->key_count; ++i) {
        if (node->children[i] == NULL) {
            return false;
        }
        bool child_has_low = (i > 0) ? true : has_low;
        int64_t child_low = (i > 0) ? node->keys[i - 1] : low;

        bool child_has_high = (i < node->key_count) ? true : has_high;
        int64_t child_high = (i < node->key_count) ? node->keys[i] : high;

        if (!validate_helper(
                node->children[i], t, false, current_depth + 1,
                expected_leaf_depth, has_leaf_depth,
                child_has_low, child_low, child_has_high, child_high)) {
            return false;
        }
    }
    return true;
}

bool btree_validate(const BTreeNode *root, size_t t) {
    if (root == NULL || t < 2) {
        return false;
    }
    if (root->key_count == 0 && root->leaf) {
        return true;
    }
    size_t expected_depth = 0;
    bool has_depth = false;
    return validate_helper(root, t, true, 0, &expected_depth, &has_depth, false, 0, false, 0);
}

int main(void) {
    size_t t = 2; /* 2-3-4 树，最大键数为 3 */

    /* 1. 构造满根结点: [10, 20, 30] */
    BTreeNode *root = create_node(t, true);
    root->key_count = 3;
    root->keys[0] = 10;
    root->keys[1] = 20;
    root->keys[2] = 30;

    assert(btree_validate(root, t));

    /* 2. 根裂变上升 */
    BTreeNode *new_root = btree_split_root(root, t);
    /* 裂变后:
       new_root 拥有 1 个键 20
       child 0 拥有 1 个键 10
       child 1 拥有 1 个键 30
    */
    assert(new_root->key_count == 1 && new_root->keys[0] == 20);
    assert(new_root->children[0]->key_count == 1 && new_root->children[0]->keys[0] == 10);
    assert(new_root->children[1]->key_count == 1 && new_root->children[1]->keys[0] == 30);
    assert(btree_validate(new_root, t));

    /* 3. 填充 child 1 达到满容量: [30, 40, 50] */
    BTreeNode *c1 = new_root->children[1];
    c1->key_count = 3;
    c1->keys[0] = 30;
    c1->keys[1] = 40;
    c1->keys[2] = 50;

    assert(btree_validate(new_root, t));

    /* 4. 分裂内部结点的孩子 child 1 */
    btree_split_child(new_root, 1, c1, t);
    /* 分裂后:
       new_root 拥有 2 个键: [20, 40]
       children[0] 拥有 [10]
       children[1] 拥有 [30]
       children[2] 拥有 [50]
    */
    assert(new_root->key_count == 2);
    assert(new_root->keys[0] == 20 && new_root->keys[1] == 40);
    assert(new_root->children[0]->key_count == 1 && new_root->children[0]->keys[0] == 10);
    assert(new_root->children[1]->key_count == 1 && new_root->children[1]->keys[0] == 30);
    assert(new_root->children[2]->key_count == 1 && new_root->children[2]->keys[0] == 50);

    assert(btree_validate(new_root, t));

    free_tree(new_root);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：分裂一个满结点只涉及 $O(t)$ 个键和指针的局部内存复制与平移，执行时间为 $O(t)$。由于 $t$ 为固定阶数，单次分裂操作为严格的 $O(1)$。
- **空间复杂度**：每次分裂分配且仅分配一个新结点结构，额外空间复杂度为 $O(t) = O(1)$。全树验证器遍历整棵树的每一个结点，时间复杂度为 $O(P)$（$P$ 为总结点数）。

## 延伸思考 / 模式总结

满结点分裂是 B 树能够维持绝对平衡的核心奥秘。二叉搜索树由于是“从上往下生长”，新元素总是作为最底层的叶子插入，极易因倾斜导致高度失控；而 B 树是“**自底向上生长**”——所有插入首先发生在叶子层，只有当容量饱和时才通过中位数向上传递，迫使树从根部向上“长高”。这种反向生长机制天然抹平了所有分支的深度差异。

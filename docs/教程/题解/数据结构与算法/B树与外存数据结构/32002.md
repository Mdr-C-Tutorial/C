# 32002：B 树内存页的按键查找与闭区间扫描

## 题目

在 `32001` 的 `BTreePage` 内存页数组上实现按键查找与有序闭区间扫描：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

bool btree_search_key(
    const BTreePage *pages,
    size_t page_count,
    size_t root,
    int64_t key,
    size_t *out_page,
    size_t *out_slot,
    bool *out_found
);

bool btree_range_scan(
    const BTreePage *pages,
    size_t page_count,
    size_t root,
    int64_t low,
    int64_t high,
    int64_t *out_keys,
    size_t capacity,
    size_t *out_count
);
```

要求：

1. 复用 `32001` 的 `BTreePage` 结构与合法性要求；当传入的页数组不构成合法 B 树或参数无效时，两个函数均返回 `false` 且不得修改输出对象。
2. `btree_search_key` 从 `root` 出发按多路搜索树规则查找 `key`：若命中，写入所在页下标 `*out_page`、页内槽位 `*out_slot`、设置 `*out_found = true` 并返回 `true`；若未命中，成功返回 `true`、设置 `*out_found = false`，且不得修改 `*out_page` 与 `*out_slot`。
3. `btree_range_scan` 按严格升序导出闭区间 `[low, high]` 内的全部键：在内部页多路遍历时，必须利用分隔键剪枝不可能与 `[low, high]` 相交的子树分支。
4. 当 `low > high`、命中键数超过 `capacity`（或命中键数大于 `0` 但 `out_keys == NULL`）时返回 `false`，不得修改输出数组与 `*out_count`。
5. 说明在合法 B 树上执行按键查找与带剪枝闭区间扫描的时间复杂度。

## 解析

首先调用 `btree_validate` 校验传入的内存页数组确实构成一棵无环、边界严格有序、叶子等深的合法 B 树，从而保证后续下探不会死循环或越界。

1. **多路按键查找 (`btree_search_key`)**：
   从根页 `current = root` 开始循环。在当前页内找第一个满足 `page->keys[slot] >= key` 的槽位 `slot`：
   - 若 `slot < page->key_count && page->keys[slot] == key`，则查找命中，将 `current` 与 `slot` 写入输出对象并置 `*out_found = true`；
   - 若当前页是叶子页（`page->leaf`），说明键不在树中，仅置 `*out_found = false` 并保留 `*out_page`、`*out_slot` 原值不变；
   - 若当前页是内部页，则转入子页 `current = page->children[slot]` 继续下一层查找。
2. **多路闭区间剪枝扫描 (`btree_range_scan`)**：
   在内部页 `page` 中，第 `i` 个子页 `children[i]` 的键严格位于分隔键 `(keys[i - 1], keys[i])` 之间（当 `i == 0` 时无下界，当 `i == key_count` 时无上界）。因此，只有同时满足：
   - 上界条件：`i == page->key_count || page->keys[i] > low`（否则该子树全部键 `< low`）；
   - 下界条件：`i == 0 || page->keys[i - 1] < high`（否则该子树全部键 `> high`），
   才需要递归进入 `children[i]`。先执行一趟带剪枝的计数遍历，确认命中总数不超过 `capacity` 后再执行第二趟升序写入，保证容量不足时不污染输出数组。

## 答案

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

enum {
    BTREE_MAX_KEYS = 3,
    BTREE_MAX_CHILDREN = 4
};

typedef struct {
    bool leaf;
    size_t key_count;
    int64_t keys[BTREE_MAX_KEYS];
    size_t children[BTREE_MAX_CHILDREN];
} BTreePage;

typedef struct {
    bool has_leaf_depth;
    size_t leaf_depth;
} ValidationContext;

static bool valid_reference(size_t page_count, size_t index) {
    return index != SIZE_MAX && index < page_count;
}

static bool validate_page(
    const BTreePage *pages,
    size_t page_count,
    size_t index,
    unsigned char *state,
    bool is_root,
    bool has_lower,
    int64_t lower,
    bool has_upper,
    int64_t upper,
    size_t depth,
    ValidationContext *context
) {
    if (!valid_reference(page_count, index) || state[index] != 0) {
        return false;
    }
    const BTreePage *page = &pages[index];
    if (page->key_count > BTREE_MAX_KEYS ||
        (!is_root && page->key_count == 0) ||
        (is_root && page->key_count == 0 && !page->leaf)) {
        return false;
    }
    for (size_t i = 0; i < page->key_count; ++i) {
        if ((i > 0 && page->keys[i - 1] >= page->keys[i]) ||
            (has_lower && page->keys[i] <= lower) ||
            (has_upper && page->keys[i] >= upper)) {
            return false;
        }
    }
    state[index] = 1;

    if (page->leaf) {
        for (size_t i = 0; i < BTREE_MAX_CHILDREN; ++i) {
            if (page->children[i] != SIZE_MAX) {
                return false;
            }
        }
        if (!context->has_leaf_depth) {
            context->has_leaf_depth = true;
            context->leaf_depth = depth;
        } else if (context->leaf_depth != depth) {
            return false;
        }
    } else {
        for (size_t i = 0; i <= page->key_count; ++i) {
            if (!valid_reference(page_count, page->children[i])) {
                return false;
            }
        }
        for (size_t i = page->key_count + 1;
             i < BTREE_MAX_CHILDREN;
             ++i) {
            if (page->children[i] != SIZE_MAX) {
                return false;
            }
        }
        if (depth == SIZE_MAX) {
            return false;
        }
        for (size_t i = 0; i <= page->key_count; ++i) {
            bool child_has_lower = has_lower;
            bool child_has_upper = has_upper;
            int64_t child_lower = lower;
            int64_t child_upper = upper;
            if (i > 0) {
                child_has_lower = true;
                child_lower = page->keys[i - 1];
            }
            if (i < page->key_count) {
                child_has_upper = true;
                child_upper = page->keys[i];
            }
            if (!validate_page(
                    pages, page_count, page->children[i], state,
                    false, child_has_lower, child_lower,
                    child_has_upper, child_upper, depth + 1,
                    context
                )) {
                return false;
            }
        }
    }
    state[index] = 2;
    return true;
}

bool btree_validate(
    const BTreePage *pages,
    size_t page_count,
    size_t root
) {
    if (pages == NULL || page_count == 0 || root >= page_count ||
        page_count > SIZE_MAX / sizeof(unsigned char)) {
        return false;
    }
    unsigned char *state = calloc(page_count, sizeof *state);
    if (state == NULL) {
        return false;
    }
    ValidationContext context = {false, 0};
    bool valid = validate_page(
        pages, page_count, root, state,
        true, false, 0, false, 0, 0, &context
    );
    if (valid) {
        for (size_t i = 0; i < page_count; ++i) {
            if (state[i] != 2) {
                valid = false;
                break;
            }
        }
    }
    free(state);
    return valid;
}

bool btree_search_key(
    const BTreePage *pages,
    size_t page_count,
    size_t root,
    int64_t key,
    size_t *out_page,
    size_t *out_slot,
    bool *out_found
) {
    if (out_page == NULL || out_slot == NULL || out_found == NULL ||
        !btree_validate(pages, page_count, root)) {
        return false;
    }

    size_t current = root;
    for (;;) {
        const BTreePage *page = &pages[current];
        size_t slot = 0;
        while (slot < page->key_count && page->keys[slot] < key) {
            ++slot;
        }
        if (slot < page->key_count && page->keys[slot] == key) {
            *out_page = current;
            *out_slot = slot;
            *out_found = true;
            return true;
        }
        if (page->leaf) {
            *out_found = false;
            return true;
        }
        current = page->children[slot];
    }
}

static void count_btree_range(
    const BTreePage *pages,
    size_t index,
    int64_t low,
    int64_t high,
    size_t *counter
) {
    const BTreePage *page = &pages[index];
    for (size_t i = 0; i < page->key_count; ++i) {
        if (!page->leaf &&
            page->keys[i] > low &&
            (i == 0 || page->keys[i - 1] < high)) {
            count_btree_range(
                pages, page->children[i], low, high, counter
            );
        }
        if (page->keys[i] >= low && page->keys[i] <= high) {
            ++(*counter);
        }
    }
    if (!page->leaf &&
        (page->key_count == 0 ||
         page->keys[page->key_count - 1] < high)) {
        count_btree_range(
            pages, page->children[page->key_count],
            low, high, counter
        );
    }
}

static void collect_btree_range(
    const BTreePage *pages,
    size_t index,
    int64_t low,
    int64_t high,
    int64_t *out_keys,
    size_t *cursor
) {
    const BTreePage *page = &pages[index];
    for (size_t i = 0; i < page->key_count; ++i) {
        if (!page->leaf &&
            page->keys[i] > low &&
            (i == 0 || page->keys[i - 1] < high)) {
            collect_btree_range(
                pages, page->children[i], low, high, out_keys, cursor
            );
        }
        if (page->keys[i] >= low && page->keys[i] <= high) {
            out_keys[(*cursor)++] = page->keys[i];
        }
    }
    if (!page->leaf &&
        (page->key_count == 0 ||
         page->keys[page->key_count - 1] < high)) {
        collect_btree_range(
            pages, page->children[page->key_count],
            low, high, out_keys, cursor
        );
    }
}

bool btree_range_scan(
    const BTreePage *pages,
    size_t page_count,
    size_t root,
    int64_t low,
    int64_t high,
    int64_t *out_keys,
    size_t capacity,
    size_t *out_count
) {
    if (out_count == NULL || low > high ||
        !btree_validate(pages, page_count, root)) {
        return false;
    }

    size_t matched = 0;
    count_btree_range(pages, root, low, high, &matched);
    if (matched > capacity || (matched > 0 && out_keys == NULL)) {
        return false;
    }

    size_t cursor = 0;
    collect_btree_range(pages, root, low, high, out_keys, &cursor);
    *out_count = cursor;
    return true;
}
```

在已知合法的 B 树上（不含全表结构验证耗时 $O(P)$），单次按键查找每层最多比较 $2t - 1 = 3$ 个键，时间复杂度为 $O(\log n)$；带剪枝的闭区间扫描仅访问两条边界路径上的页以及含命中键的页，查找与扫描阶段本身的时间复杂度为 $O(\log n + k)$。

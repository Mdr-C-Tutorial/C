# 32005 - B+ 树叶子双向链表与范围扫描

## 题目描述

实现 B+ 树（B+ Tree）结构，其内部索引结点仅保存路由路标，全部数据键值对均驻留在底层叶结点中，底层所有叶结点通过双向链表线性串接，支持极速的有序闭区间范围扫描 `bptree_range_scan`。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct BpTree BpTree;

BpTree *bptree_create(size_t t);
void bptree_destroy(BpTree *tree);
bool bptree_insert(
    BpTree *tree,
    int64_t key,
    int64_t val,
    bool *out_inserted
);
bool bptree_range_scan(
    const BpTree *tree,
    int64_t low,
    int64_t high,
    int64_t *out_keys,
    int64_t *out_vals,
    size_t capacity,
    size_t *out_count
);
```

要求：

1. **叶子数据化与双向链表**：所有实际数据项只存放在叶子结点中；叶子结点之间维护 `prev` 与 `next` 指针，构成双向有序链表。
2. **叶子分裂的键复制特性**：当叶子满裂变时，中位数键不仅要上升复制到父结点作为路标，**还必须保留在右叶结点的头部**（因为数据必须在叶子层完整存在，不可丢失）。
3. **范围扫描的线性穿透性**：
   - 仅在定位起点时沿树下降一层层查找，耗时 $O(\log_t n)$；
   - 一旦找到首个满足 `key >= low` 的叶子结点，后续的数据收集**全部沿着叶子结点的 `next` 指针水平顺序遍历**，直到遇到首个严格大于 `high` 的键立即终止；
   - 扫描过程中严禁折回内部结点进行中序回溯。
4. 保证强异常安全：当匹配记录数超过调用方给定的缓冲区 `capacity` 时，返回 `false` 且不污染调用方的数据。

## 解题思路

### 1. B+ 树相较于普通 B 树的两大核心进化

- **进化 1：内部结点不存数据，专注路由**：
  普通 B 树的内部结点既包含键也包含真实数据记录（或行指针），使得内部结点的分支因子被庞大的数据体大幅压缩；而 B+ 树内部结点只存放索引键和子页地址，分支因子极大，使得数亿行的数据表树高依然只有 3 到 4 层。
- **进化 2：叶子结点双向穿透（链表直通）**：
  在普通 B 树中执行有序范围扫描，控制流必须在内部父结点和各个子树之间反复回溯（中序遍历），产生大量的垂直上下跳转与随机 I/O；
  而 B+ 树将所有叶结点直接链接成双向链表，范围查询只需一次 $O(\log_B n)$ 的垂直下探定位起点，后续的连续范围读取完全退化为外存连续块的顺序读取（Sequential I/O），吞吐量提升数个数量级！

### 2. 范围扫描的双趟保护机制

为了防止在缓冲区 `capacity` 不足时向调用方写入半截数据，我们采取两趟扫描：
1. **第一趟（只读测算）**：从定位到的起始叶结点开始沿 `next` 遍历计数，统计落在 `[low, high]` 内的元素总数 `matched`。若 `matched > capacity`，直接返回 `false`；
2. **第二趟（有序导出）**：重新从起始叶结点导出数据到 `out_keys` 与 `out_vals` 中，确保操作的原子性与安全保证。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct BpNode BpNode;

struct BpNode {
    bool leaf;
    size_t key_count;
    int64_t *keys;
    union {
        BpNode **children; /* 内部结点：子指针数组，容量为 2 * t */
        int64_t *vals;     /* 叶子结点：数据值数组，容量为 2 * t - 1 */
    } u;
    BpNode *prev;          /* 双向叶子链表 */
    BpNode *next;
};

typedef struct BpTree {
    BpNode *root;
    BpNode *leaf_head;
    size_t t;
    size_t size;
} BpTree;

static BpNode *create_node(size_t t, bool leaf) {
    BpNode *n = (BpNode *)malloc(sizeof(BpNode));
    assert(n != NULL);
    n->leaf = leaf;
    n->key_count = 0;
    n->keys = (int64_t *)malloc(sizeof(int64_t) * (2 * t - 1));
    assert(n->keys != NULL);
    if (leaf) {
        n->u.vals = (int64_t *)malloc(sizeof(int64_t) * (2 * t - 1));
        assert(n->u.vals != NULL);
    } else {
        n->u.children = (BpNode **)malloc(sizeof(BpNode *) * (2 * t));
        assert(n->u.children != NULL);
        for (size_t i = 0; i < 2 * t; ++i) {
            n->u.children[i] = NULL;
        }
    }
    n->prev = NULL;
    n->next = NULL;
    return n;
}

static void free_nodes(BpNode *n) {
    if (n == NULL) {
        return;
    }
    if (!n->leaf) {
        for (size_t i = 0; i <= n->key_count; ++i) {
            free_nodes(n->u.children[i]);
        }
        free(n->u.children);
    } else {
        free(n->u.vals);
    }
    free(n->keys);
    free(n);
}

BpTree *bptree_create(size_t t) {
    if (t < 2) {
        return NULL;
    }
    BpTree *tree = (BpTree *)malloc(sizeof(BpTree));
    if (tree == NULL) {
        return NULL;
    }
    tree->t = t;
    tree->root = create_node(t, true);
    tree->leaf_head = tree->root;
    tree->size = 0;
    return tree;
}

void bptree_destroy(BpTree *tree) {
    if (tree == NULL) {
        return;
    }
    free_nodes(tree->root);
    free(tree);
}

static void bptree_split_child(BpNode *parent, size_t i, size_t t) {
    BpNode *y = parent->u.children[i];
    BpNode *z = create_node(t, y->leaf);

    if (y->leaf) {
        /* 叶结点裂变：数据必须保存在右叶子，首个键复制到父结点 */
        size_t mid = t - 1;
        z->key_count = y->key_count - mid;
        for (size_t j = 0; j < z->key_count; ++j) {
            z->keys[j] = y->keys[j + mid];
            z->u.vals[j] = y->u.vals[j + mid];
        }
        y->key_count = mid;

        /* 串接双向链表 */
        z->next = y->next;
        z->prev = y;
        if (y->next != NULL) {
            y->next->prev = z;
        }
        y->next = z;

        /* 将 z 的首个键上提作为父结点的路由路标 */
        for (size_t j = parent->key_count + 1; j > i + 1; --j) {
            parent->u.children[j] = parent->u.children[j - 1];
        }
        parent->u.children[i + 1] = z;

        for (size_t j = parent->key_count; j > i; --j) {
            parent->keys[j] = parent->keys[j - 1];
        }
        parent->keys[i] = z->keys[0];
        parent->key_count++;
    } else {
        /* 内部结点裂变：中位数上移，不留在子层 */
        z->key_count = t - 1;
        for (size_t j = 0; j < t - 1; ++j) {
            z->keys[j] = y->keys[j + t];
        }
        for (size_t j = 0; j < t; ++j) {
            z->u.children[j] = y->u.children[j + t];
            y->u.children[j + t] = NULL;
        }
        y->key_count = t - 1;

        for (size_t j = parent->key_count + 1; j > i + 1; --j) {
            parent->u.children[j] = parent->u.children[j - 1];
        }
        parent->u.children[i + 1] = z;

        for (size_t j = parent->key_count; j > i; --j) {
            parent->keys[j] = parent->keys[j - 1];
        }
        parent->keys[i] = y->keys[t - 1];
        parent->key_count++;
    }
}

static bool bptree_insert_non_full(BpNode *node, int64_t key, int64_t val, size_t t, bool *out_inserted) {
    size_t i = node->key_count;
    if (node->leaf) {
        while (i > 0 && key < node->keys[i - 1]) {
            node->keys[i] = node->keys[i - 1];
            node->u.vals[i] = node->u.vals[i - 1];
            --i;
        }
        if (i > 0 && node->keys[i - 1] == key) {
            node->u.vals[i - 1] = val;
            *out_inserted = false;
            return true;
        }
        node->keys[i] = key;
        node->u.vals[i] = val;
        node->key_count++;
        *out_inserted = true;
        return true;
    } else {
        while (i > 0 && key < node->keys[i - 1]) {
            --i;
        }
        if (node->u.children[i]->key_count == 2 * t - 1) {
            bptree_split_child(node, i, t);
            if (key >= node->keys[i]) {
                i++;
            }
        }
        return bptree_insert_non_full(node->u.children[i], key, val, t, out_inserted);
    }
}

bool bptree_insert(BpTree *tree, int64_t key, int64_t val, bool *out_inserted) {
    if (tree == NULL || out_inserted == NULL) {
        return false;
    }
    BpNode *r = tree->root;
    if (r->key_count == 2 * tree->t - 1) {
        BpNode *s = create_node(tree->t, false);
        tree->root = s;
        s->u.children[0] = r;
        bptree_split_child(s, 0, tree->t);
        bool ins = false;
        bool ok = bptree_insert_non_full(s, key, val, tree->t, &ins);
        if (ok && ins) {
            tree->size++;
        }
        *out_inserted = ins;
        return ok;
    } else {
        bool ins = false;
        bool ok = bptree_insert_non_full(r, key, val, tree->t, &ins);
        if (ok && ins) {
            tree->size++;
        }
        *out_inserted = ins;
        return ok;
    }
}

bool bptree_range_scan(
    const BpTree *tree,
    int64_t low,
    int64_t high,
    int64_t *out_keys,
    int64_t *out_vals,
    size_t capacity,
    size_t *out_count
) {
    if (tree == NULL || out_count == NULL || low > high) {
        return false;
    }

    /* 1. 垂直下探定位可能包含 low 的起始叶结点 */
    const BpNode *curr = tree->root;
    while (!curr->leaf) {
        size_t idx = 0;
        while (idx < curr->key_count && curr->keys[idx] <= low) {
            idx++;
        }
        curr = curr->u.children[idx];
    }

    /* 2. 第一趟：只读计数校验容量 */
    size_t matched = 0;
    const BpNode *leaf = curr;
    bool done = false;
    while (leaf != NULL && !done) {
        for (size_t i = 0; i < leaf->key_count; ++i) {
            if (leaf->keys[i] > high) {
                done = true;
                break;
            }
            if (leaf->keys[i] >= low) {
                matched++;
            }
        }
        leaf = leaf->next;
    }

    if (matched > capacity || (matched > 0 && (out_keys == NULL || out_vals == NULL))) {
        return false;
    }

    /* 3. 第二趟：水平沿双向链表极速导出 */
    size_t collected = 0;
    leaf = curr;
    done = false;
    while (leaf != NULL && !done) {
        for (size_t i = 0; i < leaf->key_count; ++i) {
            if (leaf->keys[i] > high) {
                done = true;
                break;
            }
            if (leaf->keys[i] >= low) {
                if (out_keys != NULL) {
                    out_keys[collected] = leaf->keys[i];
                }
                if (out_vals != NULL) {
                    out_vals[collected] = leaf->u.vals[i];
                }
                collected++;
            }
        }
        leaf = leaf->next;
    }

    *out_count = collected;
    return true;
}

int main(void) {
    BpTree *tree = bptree_create(3);
    assert(tree != NULL);

    bool ins = false;
    for (int64_t i = 1; i <= 100; ++i) {
        assert(bptree_insert(tree, i * 10, i * 100, &ins) && ins);
    }
    assert(tree->size == 100);

    /* 闭区间扫描 [250, 600] -> 包含 250, 260, ..., 600 共 36 条记录 */
    int64_t keys[64];
    int64_t vals[64];
    size_t count = 0;

    assert(bptree_range_scan(tree, 250, 600, keys, vals, 64, &count));
    assert(count == 36);
    for (size_t i = 0; i < count; ++i) {
        int64_t expected_k = 250 + (int64_t)i * 10;
        assert(keys[i] == expected_k);
        assert(vals[i] == expected_k * 10);
    }

    /* 区间无命中测试 */
    assert(bptree_range_scan(tree, 1005, 1050, keys, vals, 64, &count));
    assert(count == 0);

    /* 输出缓冲区不足防护：容量 10 < 36，安全返回 false */
    assert(!bptree_range_scan(tree, 250, 600, keys, vals, 10, &count));

    bptree_destroy(tree);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：
  - 起始定位：从根到叶只下探一次，耗时 $O(\log_t n)$；
  - 范围遍历：设落在区间内的目标记录数为 $k$，每个叶结点包含约 $t$ 个记录，因此仅需访问约 $\lceil k / t \rceil$ 个叶结点，遍历耗时为 $O(k)$。
  - 总时间复杂度为严格的 $O(\log_t n + k)$，在外存模型中其 I/O 复杂度为极佳的 $O(\log_B N + K / B)$。
- **空间复杂度**：水平扫描无需任何递归调用栈与辅助队列，额外空间复杂度为严格的 $O(1)$。

## 延伸思考 / 模式总结

B+ 树几乎是关系型数据库（如 MySQL InnoDB、PostgreSQL）存储引擎索引的标准代名词。其成功的核心哲学就是**读写语义的分离与数据归一**：内部结点做纯粹的宽扇出导航，叶子结点组织高密度的顺序流。这种结构将范围扫描从高昂的随机“找道寻址”转换成了对底层块的“顺序预读”，是外存数据结构设计的终极教科书范式。

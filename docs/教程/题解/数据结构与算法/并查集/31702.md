# 31702：带权并查集与奇偶关系维护

## 题目

实现维护二元同类/异类关系的带权并查集 `ParityDisjointSet`，支持关系声明与冲突检测（`parity_dsu_relate`）以及关系查询（`parity_dsu_query`）。

## 解析

在标准并查集的 `parent` 与 `sizes` 数组之上，增加边权数组 `parity`，其中 `parity[x] ∈ {0, 1}` 表示结点 `x` 与其父结点 `parent[x]` 的相对奇偶关系（`0` 代表同类，`1` 代表异类，根结点满足 `parity[root] == 0`）。任意结点 `x` 到其所在树根 `root` 的相对关系 $d(x)$ 即为沿途所有边权的异或和（模 $2$ 加法）。

核心转移方程由以下两部分组成：

1. **两趟迭代带权路径压缩**：
   - 第一趟从 `x` 沿 `parent` 向上走到 `root`，累异或得到 `x` 相对于 `root` 的总奇偶距离 `total_xor`。
   - 第二趟从 `x` 重新向上走，记当前结点 `cursor` 到 `root` 的剩余异或距离为 `remaining_xor`（初值为 `total_xor`）。在将 `parent[cursor]` 直接改写为 `root` 前，先保存旧边权 `edge = parity[cursor]`，然后将 `parity[cursor]` 设为 `remaining_xor`，并令 `remaining_xor ^= edge` 推进到原父结点。这样无需递归即可在 $O(1)$ 额外空间内完成带权路径压缩。
2. **按大小合并时的根间边权计算**：
   设 `left` 到其根 $r_L$ 的异或距离为 $d_L$，`right` 到其根 $r_R$ 的异或距离为 $d_R$，期望建立的关系为 $w \in \{0, 1\}$：
   - 若 $r_L = r_R$，则树中已隐含两者的相对关系 $d_L \oplus d_R$。当 $(d_L \oplus d_R) \neq w$ 时说明出现奇环矛盾，置 `*out_consistent = false` 并直接返回；
   - 若 $r_L \neq r_R$，假设按大小合并将 $r_R$ 挂到 $r_L$ 之下（`parent[r_R] = r_L`），由于合并后路径 `left -> r_L <- r_R <- right` 的异或和必须等于 $w$，即 $d_L \oplus \text{parity}[r_R] \oplus d_R = w$，两边同异或 $d_L \oplus d_R$ 即得新边权：
     $$\text{parity}[r_R] = d_L \oplus d_R \oplus w$$

## 答案

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct ParityDisjointSet ParityDisjointSet;

struct ParityDisjointSet {
    size_t *parent;
    size_t *sizes;
    unsigned char *parity;
    size_t count;
};

ParityDisjointSet *parity_dsu_create(size_t count) {
    ParityDisjointSet *set = malloc(sizeof *set);
    if (set == NULL) {
        return NULL;
    }
    set->parent = NULL;
    set->sizes = NULL;
    set->parity = NULL;
    set->count = count;

    if (count > 0) {
        if (count > SIZE_MAX / sizeof *set->parent) {
            free(set);
            return NULL;
        }
        set->parent = malloc(count * sizeof *set->parent);
        set->sizes = malloc(count * sizeof *set->sizes);
        set->parity = calloc(count, sizeof *set->parity);
        if (set->parent == NULL || set->sizes == NULL ||
            set->parity == NULL) {
            free(set->parent);
            free(set->sizes);
            free(set->parity);
            free(set);
            return NULL;
        }
        for (size_t i = 0; i < count; ++i) {
            set->parent[i] = i;
            set->sizes[i] = 1;
        }
    }
    return set;
}

void parity_dsu_destroy(ParityDisjointSet *set) {
    if (set == NULL) {
        return;
    }
    free(set->parent);
    free(set->sizes);
    free(set->parity);
    free(set);
}

static size_t find_compress(
    ParityDisjointSet *set,
    size_t element,
    unsigned char *out_parity
) {
    size_t root = element;
    unsigned char total_xor = 0;
    while (set->parent[root] != root) {
        total_xor ^= set->parity[root];
        root = set->parent[root];
    }

    unsigned char remaining_xor = total_xor;
    size_t cursor = element;
    while (cursor != root) {
        size_t next = set->parent[cursor];
        unsigned char edge = set->parity[cursor];
        set->parent[cursor] = root;
        set->parity[cursor] = remaining_xor;
        remaining_xor ^= edge;
        cursor = next;
    }

    *out_parity = total_xor;
    return root;
}

static size_t find_const(
    const ParityDisjointSet *set,
    size_t element,
    unsigned char *out_parity
) {
    size_t cursor = element;
    unsigned char total_xor = 0;
    while (set->parent[cursor] != cursor) {
        total_xor ^= set->parity[cursor];
        cursor = set->parent[cursor];
    }
    *out_parity = total_xor;
    return cursor;
}

bool parity_dsu_relate(
    ParityDisjointSet *set,
    size_t left,
    size_t right,
    bool different,
    bool *out_consistent
) {
    if (set == NULL || out_consistent == NULL ||
        left >= set->count || right >= set->count) {
        return false;
    }

    unsigned char left_parity = 0;
    unsigned char right_parity = 0;
    size_t left_root = find_compress(set, left, &left_parity);
    size_t right_root = find_compress(set, right, &right_parity);
    unsigned char expected = different ? 1u : 0u;

    if (left_root == right_root) {
        *out_consistent = ((left_parity ^ right_parity) == expected);
        return true;
    }

    if (set->sizes[left_root] < set->sizes[right_root]) {
        size_t temp_root = left_root;
        left_root = right_root;
        right_root = temp_root;

        unsigned char temp_parity = left_parity;
        left_parity = right_parity;
        right_parity = temp_parity;
    }

    set->parent[right_root] = left_root;
    set->parity[right_root] =
        (unsigned char)(left_parity ^ right_parity ^ expected);
    set->sizes[left_root] += set->sizes[right_root];
    *out_consistent = true;
    return true;
}

bool parity_dsu_query(
    const ParityDisjointSet *set,
    size_t left,
    size_t right,
    bool *out_known,
    bool *out_different
) {
    if (set == NULL || out_known == NULL || out_different == NULL ||
        left >= set->count || right >= set->count) {
        return false;
    }

    unsigned char left_parity = 0;
    unsigned char right_parity = 0;
    size_t left_root = find_const(set, left, &left_parity);
    size_t right_root = find_const(set, right, &right_parity);
    if (left_root != right_root) {
        *out_known = false;
        return true;
    }

    *out_known = true;
    *out_different = ((left_parity ^ right_parity) != 0u);
    return true;
}
```

结合按大小合并与路径压缩后，每次关系维护的摊还时间复杂度为 $O(\alpha(n))$，只读查询的最坏时间复杂度不超过树高上界 $O(\log n)$（与压缩操作交替时摊还为 $O(\alpha(n))$），空间复杂度为 $O(n)$。

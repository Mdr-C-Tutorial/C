# 31308 - 滑窗重复检测与定和子数组计数

## 题目

结合滑动窗口、前缀和与散列表，在期望 $O(n)$ 时间复杂度下实现以下两个序列统计函数：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

bool has_nearby_duplicate(
    const int64_t *nums,
    size_t n,
    size_t k,
    bool *out_found
);

bool count_subarrays_with_sum(
    const int64_t *nums,
    size_t n,
    int64_t target_sum,
    size_t *out_count
);
```

要求：

1. `has_nearby_duplicate` 判定是否存在下标 $0 \le i < j < n$ 满足 `nums[i] == nums[j]` 且 $j - i \le k$，将结果写入 `*out_found`。要求维护一个元素个数不超过 $k$ 的散列集合（当窗口右移超过 $k$ 时及时从集合中移除滑出窗口的 `nums[j - k]`），使空间复杂度受限于 $O(\min(n, k))$。
2. `count_subarrays_with_sum` 统计满足连续非空子数组元素之和等于 `target_sum` 的区间 $[i, j]$（$0 \le i \le j < n$）的总个数并写入 `*out_count`（提示：采用 `uint64_t` 无符号模运算或假定前缀和在 `int64_t` 范围内，用散列映射维护历史前缀和 `prefix` 的出现频次，每次累加 `freq[prefix - target_sum]`）。
3. 参数无效或内部散列表内存分配失败时返回 `false`，不得修改输出对象。

## 解析

### 1. 滑动窗口与定长散列集合

若每次暴力向前扫描 $k$ 个元素，总耗时为 $O(n \cdot k)$。
借助散列表维护滑动窗口内的元素：
- 集合容量上限设为 $\min(n, k) + 1$；
- 遍历 $j = 0, 1, \dots, n - 1$：
  - 先在集合中查找 `nums[j]`，若命中则立即判定存在，置 `*out_found = true` 并返回；
  - 若未命中，将 `nums[j]` 插入集合；
  - 当 $j \ge k$ 时，最左侧元素 `nums[j - k]` 已滑出窗口，将其从集合中删除。
- 窗口内元素个数至多为 $k$ 个，空间复杂度为严格的 $O(\min(n, k))$，时间复杂度为期望 $O(n)$。

### 2. 前缀和与频次映射

设连续子数组 $[i, j]$ 之和为 $\sum_{t=i}^j \text{nums}[t] = P_j - P_{i-1}$，其中 $P_t = \sum_{m=0}^t \text{nums}[m]$ 是前缀和（规定 $P_{-1} = 0$）。
区间和等于目标值即：

$$P_j - P_{i-1} = \text{target\_sum} \iff P_{i-1} = P_j - \text{target\_sum}$$

因此，从左到右维护当前前缀和 $P_j$，在历史前缀和频次映射中查询所需前缀和 $P_j - \text{target\_sum}$ 的出现次数并累加到答案中。
初始时，必须在频次映射中先记录前缀和为 $0$ 的出现次数为 $1$（对应 $i = 0$ 时空前缀的情况）。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

// --- 滑窗散列集合（支持删除） ---
typedef enum {
    SLOT_EMPTY = 0,
    SLOT_OCCUPIED = 1,
    SLOT_TOMBSTONE = 2
} SlotState;

typedef struct {
    int64_t key;
    SlotState state;
} SetSlot;

typedef struct {
    SetSlot *slots;
    size_t cap;
    size_t size;
} SlidingSet;

static uint64_t hash64(int64_t x) {
    uint64_t v = (uint64_t)x;
    v = (v ^ (v >> 30)) * UINT64_C(0xbf58476d1ce4e5b9);
    v = (v ^ (v >> 27)) * UINT64_C(0x94d049bb133111eb);
    return v ^ (v >> 31);
}

static bool sset_init(SlidingSet *s, size_t min_cap) {
    size_t cap = 8;
    while (cap < min_cap * 3) {
        if (cap > SIZE_MAX / 2) return false;
        cap *= 2;
    }
    s->slots = calloc(cap, sizeof(SetSlot));
    if (!s->slots) return false;
    s->cap = cap;
    s->size = 0;
    return true;
}

static void sset_free(SlidingSet *s) {
    free(s->slots);
}

static bool sset_contains(const SlidingSet *s, int64_t key) {
    if (s->cap == 0) return false;
    size_t mask = s->cap - 1;
    size_t idx = (size_t)(hash64(key) & mask);
    for (size_t step = 0; step < s->cap; ++step) {
        if (s->slots[idx].state == SLOT_EMPTY) return false;
        if (s->slots[idx].state == SLOT_OCCUPIED && s->slots[idx].key == key) {
            return true;
        }
        idx = (idx + 1) & mask;
    }
    return false;
}

static void sset_insert(SlidingSet *s, int64_t key) {
    size_t mask = s->cap - 1;
    size_t idx = (size_t)(hash64(key) & mask);
    size_t first_tomb = SIZE_MAX;
    for (;;) {
        SlotState st = s->slots[idx].state;
        if (st == SLOT_EMPTY) {
            size_t target = (first_tomb != SIZE_MAX) ? first_tomb : idx;
            s->slots[target].key = key;
            s->slots[target].state = SLOT_OCCUPIED;
            s->size++;
            return;
        }
        if (st == SLOT_OCCUPIED && s->slots[idx].key == key) {
            return;
        }
        if (st == SLOT_TOMBSTONE && first_tomb == SIZE_MAX) {
            first_tomb = idx;
        }
        idx = (idx + 1) & mask;
    }
}

static void sset_remove(SlidingSet *s, int64_t key) {
    size_t mask = s->cap - 1;
    size_t idx = (size_t)(hash64(key) & mask);
    for (size_t step = 0; step < s->cap; ++step) {
        SlotState st = s->slots[idx].state;
        if (st == SLOT_EMPTY) return;
        if (st == SLOT_OCCUPIED && s->slots[idx].key == key) {
            s->slots[idx].state = SLOT_TOMBSTONE;
            s->size--;
            return;
        }
        idx = (idx + 1) & mask;
    }
}

bool has_nearby_duplicate(
    const int64_t *nums,
    size_t n,
    size_t k,
    bool *out_found
) {
    if (nums == NULL || out_found == NULL) {
        return false;
    }
    if (n <= 1 || k == 0) {
        *out_found = false;
        return true;
    }

    size_t window_cap = k < n ? k : n;
    SlidingSet set;
    if (!sset_init(&set, window_cap)) {
        return false;
    }

    for (size_t j = 0; j < n; ++j) {
        if (sset_contains(&set, nums[j])) {
            *out_found = true;
            sset_free(&set);
            return true;
        }
        sset_insert(&set, nums[j]);
        if (j >= k) {
            sset_remove(&set, nums[j - k]);
        }
    }

    *out_found = false;
    sset_free(&set);
    return true;
}

// --- 前缀和频次映射 ---
typedef struct FreqNode {
    int64_t prefix;
    size_t freq;
    struct FreqNode *next;
} FreqNode;

typedef struct {
    FreqNode **buckets;
    size_t cap;
} FreqMap;

static bool fmap_init(FreqMap *m, size_t cap) {
    m->buckets = calloc(cap, sizeof(FreqNode *));
    if (!m->buckets) return false;
    m->cap = cap;
    return true;
}

static void fmap_free(FreqMap *m) {
    for (size_t i = 0; i < m->cap; ++i) {
        FreqNode *cur = m->buckets[i];
        while (cur) {
            FreqNode *next = cur->next;
            free(cur);
            cur = next;
        }
    }
    free(m->buckets);
}

static size_t fmap_get(const FreqMap *m, int64_t prefix) {
    size_t idx = (size_t)(hash64(prefix) & (m->cap - 1));
    for (FreqNode *cur = m->buckets[idx]; cur; cur = cur->next) {
        if (cur->prefix == prefix) {
            return cur->freq;
        }
    }
    return 0;
}

static bool fmap_add(FreqMap *m, int64_t prefix) {
    size_t idx = (size_t)(hash64(prefix) & (m->cap - 1));
    for (FreqNode *cur = m->buckets[idx]; cur; cur = cur->next) {
        if (cur->prefix == prefix) {
            cur->freq++;
            return true;
        }
    }
    FreqNode *node = malloc(sizeof(FreqNode));
    if (!node) return false;
    node->prefix = prefix;
    node->freq = 1;
    node->next = m->buckets[idx];
    m->buckets[idx] = node;
    return true;
}

bool count_subarrays_with_sum(
    const int64_t *nums,
    size_t n,
    int64_t target_sum,
    size_t *out_count
) {
    if (nums == NULL || out_count == NULL) {
        return false;
    }
    if (n == 0) {
        *out_count = 0;
        return true;
    }

    size_t cap = 16;
    while (cap < n * 2) {
        if (cap > SIZE_MAX / 2) return false;
        cap *= 2;
    }

    FreqMap map;
    if (!fmap_init(&map, cap)) {
        return false;
    }

    // 初始前缀和为 0 出现 1 次
    if (!fmap_add(&map, 0)) {
        fmap_free(&map);
        return false;
    }

    size_t total_count = 0;
    int64_t running_sum = 0;

    for (size_t j = 0; j < n; ++j) {
        running_sum += nums[j];
        int64_t need = running_sum - target_sum;
        total_count += fmap_get(&map, need);

        if (!fmap_add(&map, running_sum)) {
            fmap_free(&map);
            return false;
        }
    }

    *out_count = total_count;
    fmap_free(&map);
    return true;
}

int main(void) {
    // 1. 验证 has_nearby_duplicate
    const int64_t arr1[] = {1, 2, 3, 1};
    bool found = false;
    assert(has_nearby_duplicate(arr1, 4, 3, &found) && found);
    assert(has_nearby_duplicate(arr1, 4, 2, &found) && !found);

    const int64_t arr2[] = {1, 0, 1, 1};
    assert(has_nearby_duplicate(arr2, 4, 1, &found) && found);

    // 2. 验证 count_subarrays_with_sum
    const int64_t arr3[] = {1, 1, 1};
    size_t cnt = 0;
    assert(count_subarrays_with_sum(arr3, 3, 2, &cnt) && cnt == 2);

    const int64_t arr4[] = {1, -1, 0};
    assert(count_subarrays_with_sum(arr4, 3, 0, &cnt) && cnt == 3);

    return 0;
}
```

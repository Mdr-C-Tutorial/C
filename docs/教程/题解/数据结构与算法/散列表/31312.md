# 题解：31312

## 题目

在线性探测开放寻址表中，除了使用墓碑，还可以采用 Knuth 在《计算机程序设计艺术》第 3 卷 6.4 节给出的 **Algorithm R（反向移位删除，Backward-Shift Deletion）**，使槽仅需 `EMPTY` 和 `OCCUPIED` 两种状态，彻底消除墓碑：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct LinearShiftMap LinearShiftMap;

LinearShiftMap *lshift_map_create(void);
void lshift_map_destroy(LinearShiftMap *map);
bool lshift_map_put(
    LinearShiftMap *map,
    int64_t key,
    int64_t value,
    bool *out_replaced
);
bool lshift_map_get(
    const LinearShiftMap *map,
    int64_t key,
    int64_t *out_value
);
bool lshift_map_remove(
    LinearShiftMap *map,
    int64_t key,
    bool *out_removed
);
size_t lshift_map_size(const LinearShiftMap *map);
```

要求：

1. 槽状态只允许包含“空”与“占用”两态，**不得使用墓碑状态**，删除操作也不得触发整表重散列。
2. 当删除位于槽下标 $i$ 的元素并将槽 $i$ 腾空后，令扫描指针 $j = (i + 1) \bmod m$ 依次检查紧随其后的连续占用簇（直到遇到真正的空槽为止）：对簇中位于槽 $j$ 的元素，设其初始散列槽为 $r = h(\text{key}_j) \bmod m$；若 $r$ **不在**环形半开区间 $(i, j]$ 内（即当初插入该元素时曾越过槽 $i$），则将槽 $j$ 的元素搬移到空洞槽 $i$，随后令 $i \gets j$ 并继续向后扫描。
3. 写出环形半开区间判定函数 `bool in_cyclic_open_closed(size_t i, size_t j, size_t r)`（正确处理 $i < j$ 与跨表尾回绕 $i > j$ 两种情形），证明为什么按该规则前移后，簇内所有剩余元素的探测链均保持连通。

## 解析

### 1. 为什么开放寻址可以直接前移填补

线性探测中，同义词链条之所以会被破坏，是因为删除操作在探测链中间制造了断点（空槽）。墓碑机制通过引入假占用态绕过断点，但会长期污染表并拖慢后续查询。
Knuth Algorithm R 的思想是：**直接在物理上将后方的合法元素前移，缝合空洞**。

### 2. 环形半开区间 $(i, j]$ 的几何含义

当前槽 $i$ 是由于删除或前移刚产生的空洞。考察其后继占用元素所在槽 $j$ 及其理想初始散列位置 $r$：
- 顺时针环形区间 $(i, j]$ 表示“从空洞的下一个位置 $i+1$ 开始顺延到当前位置 $j$ 的区域”；
- 若 $r \in (i, j]$，说明该元素的理想初始位置本身就落在空洞 $i$ 的右侧，它在插入时**从未越过槽 $i$**（它的合法探测起始位置在 $i$ 之后），因此它绝不能被向前搬移到 $i$，否则会使它落到其理想位置 $r$ 的前方（产生负向探测偏离）；
- 若 $r \notin (i, j]$，说明从 $r$ 顺时针前进到 $j$ 的探测路径必然**跨过了空洞槽 $i$**。当初是因为槽 $i$ 被占用，该元素才被迫顺延存放在 $j$。现在槽 $i$ 已空出，将其搬移至 $i$ 恰好拉近了它与其理想位置 $r$ 的距离，且保留了连续性！
搬移完成后，原位置 $j$ 变成新的空洞，令 $i \gets j$ 继续考察后续元素，直至遇到真正的空槽终止。

### 3. 环形区间判定实现

在模 $m$ 环形数组中：
- 当 $i < j$ 时：$r \in (i, j] \iff i < r \le j$；
- 当 $i > j$ 时（区间跨越了下标 0）：$r \in (i, j] \iff r > i \lor r \le j$。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct LinearShiftMap LinearShiftMap;

typedef enum {
    LSM_EMPTY = 0,
    LSM_OCCUPIED = 1
} LsmState;

typedef struct {
    int64_t key;
    int64_t value;
    LsmState state;
} LsmSlot;

struct LinearShiftMap {
    LsmSlot *slots;
    size_t capacity;
    size_t size;
};

static uint64_t mix_hash(int64_t key) {
    uint64_t x = (uint64_t)key;
    x = (x ^ (x >> 30)) * UINT64_C(0xbf58476d1ce4e5b9);
    x = (x ^ (x >> 27)) * UINT64_C(0x94d049bb133111eb);
    return x ^ (x >> 31);
}

static inline bool in_cyclic_open_closed(size_t i, size_t j, size_t r) {
    if (i < j) {
        return r > i && r <= j;
    } else {
        return r > i || r <= j;
    }
}

LinearShiftMap *lshift_map_create(void) {
    LinearShiftMap *map = malloc(sizeof(LinearShiftMap));
    if (!map) return NULL;
    map->capacity = 16;
    map->size = 0;
    map->slots = calloc(map->capacity, sizeof(LsmSlot));
    if (!map->slots) {
        free(map);
        return NULL;
    }
    return map;
}

void lshift_map_destroy(LinearShiftMap *map) {
    if (!map) return;
    free(map->slots);
    free(map);
}

static bool lshift_rehash(LinearShiftMap *map, size_t new_cap) {
    LsmSlot *new_slots = calloc(new_cap, sizeof(LsmSlot));
    if (!new_slots) return false;
    size_t mask = new_cap - 1;

    for (size_t i = 0; i < map->capacity; ++i) {
        if (map->slots[i].state == LSM_OCCUPIED) {
            int64_t k = map->slots[i].key;
            int64_t v = map->slots[i].value;
            size_t idx = (size_t)(mix_hash(k) & mask);
            while (new_slots[idx].state == LSM_OCCUPIED) {
                idx = (idx + 1) & mask;
            }
            new_slots[idx].key = k;
            new_slots[idx].value = v;
            new_slots[idx].state = LSM_OCCUPIED;
        }
    }

    free(map->slots);
    map->slots = new_slots;
    map->capacity = new_cap;
    return true;
}

bool lshift_map_put(
    LinearShiftMap *map,
    int64_t key,
    int64_t value,
    bool *out_replaced
) {
    if (!map || !out_replaced) return false;

    if ((map->size + 1) * 4 > map->capacity * 3) {
        if (map->capacity > SIZE_MAX / 2 ||
            !lshift_rehash(map, map->capacity * 2)) {
            return false;
        }
    }

    size_t mask = map->capacity - 1;
    size_t idx = (size_t)(mix_hash(key) & mask);

    for (;;) {
        if (map->slots[idx].state == LSM_EMPTY) {
            map->slots[idx].key = key;
            map->slots[idx].value = value;
            map->slots[idx].state = LSM_OCCUPIED;
            map->size++;
            *out_replaced = false;
            return true;
        }
        if (map->slots[idx].key == key) {
            map->slots[idx].value = value;
            *out_replaced = true;
            return true;
        }
        idx = (idx + 1) & mask;
    }
}

bool lshift_map_get(
    const LinearShiftMap *map,
    int64_t key,
    int64_t *out_value
) {
    if (!map || !out_value || map->capacity == 0) return false;
    size_t mask = map->capacity - 1;
    size_t idx = (size_t)(mix_hash(key) & mask);

    for (size_t step = 0; step < map->capacity; ++step) {
        if (map->slots[idx].state == LSM_EMPTY) return false;
        if (map->slots[idx].key == key) {
            *out_value = map->slots[idx].value;
            return true;
        }
        idx = (idx + 1) & mask;
    }
    return false;
}

bool lshift_map_remove(
    LinearShiftMap *map,
    int64_t key,
    bool *out_removed
) {
    if (!map || !out_removed || map->capacity == 0) return false;
    size_t mask = map->capacity - 1;
    size_t idx = (size_t)(mix_hash(key) & mask);

    // 1. 定位待删目标
    bool found = false;
    for (size_t step = 0; step < map->capacity; ++step) {
        if (map->slots[idx].state == LSM_EMPTY) break;
        if (map->slots[idx].key == key) {
            found = true;
            break;
        }
        idx = (idx + 1) & mask;
    }

    if (!found) {
        *out_removed = false;
        return true;
    }

    // 2. Knuth Algorithm R 反向移位消洞
    size_t i = idx;
    size_t j = (i + 1) & mask;

    while (map->slots[j].state == LSM_OCCUPIED) {
        size_t r = (size_t)(mix_hash(map->slots[j].key) & mask);
        if (!in_cyclic_open_closed(i, j, r)) {
            // 将槽 j 元素移到槽 i
            map->slots[i] = map->slots[j];
            i = j;
        }
        j = (j + 1) & mask;
    }

    map->slots[i].state = LSM_EMPTY;
    map->size--;
    *out_removed = true;
    return true;
}

size_t lshift_map_size(const LinearShiftMap *map) {
    return map ? map->size : 0;
}

int main(void) {
    LinearShiftMap *map = lshift_map_create();
    assert(map != NULL);

    bool replaced = false;
    bool removed = false;
    int64_t val = 0;

    // 插入元素测试
    for (int64_t i = 1; i <= 10; ++i) {
        assert(lshift_map_put(map, i * 7, i * 100, &replaced));
        assert(!replaced);
    }
    assert(lshift_map_size(map) == 10);

    // 删除测试
    assert(lshift_map_remove(map, 28, &removed) && removed);
    assert(lshift_map_size(map) == 9);
    assert(!lshift_map_get(map, 28, &val));

    // 验证删除后其余冲突链上元素依然能被正确定位
    for (int64_t i = 1; i <= 10; ++i) {
        if (i == 4) continue; // 28
        assert(lshift_map_get(map, i * 7, &val) && val == i * 100);
    }

    lshift_map_destroy(map);
    return 0;
}
```

# 题解：31307

## 题目

利用散列表将以下两个经典的数组查找问题优化至期望 $O(n)$ 时间复杂度：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

bool two_sum_indices(
    const int64_t *nums,
    size_t n,
    int64_t target,
    size_t *out_i,
    size_t *out_j
);

size_t longest_consecutive_run(const int64_t *nums, size_t n);
```

要求：

1. `two_sum_indices` 在长度为 `n` 的数组 `nums` 中查找满足 $\text{nums}[i] + \text{nums}[j] = \text{target}$（数学精确和，不得发生有符号整数溢出 UB）且 $i < j$ 的一对下标。若存在多对，返回使右端点 $j$ 最小（若仍相同则 $i$ 最小）的那一对并写入 `*out_i, *out_j`，返回 `true`；不存在或参数无效时返回 `false` 且不修改输出对象。
2. `longest_consecutive_run` 返回数组 `nums` 中元素所能组成的最长**数值连续递增（步长为 1）**子集的长度（原数组中位置不必相邻，重复元素只计一次；例如 `[100, 4, 200, 1, 3, 2]` 返回 `4`）。
3. 在 `longest_consecutive_run` 中，先把全部元素放入散列集合，随后仅当 `x == INT64_MIN` 或 `x - 1` 不在集合中时，才以 `x` 为起点向后探测 `x + 1, x + 2, ...`（遇到 `INT64_MAX` 时停止继续加 1 以防溢出）。证明为什么内层 `while` 循环在整个算法运行期间累计最多执行 $n$ 次迭代。

## 解析

### 1. 两数之和与溢出防护

若采用暴力双重循环比对，时间复杂度为 $O(n^2)$。
利用散列表维护历史元素 `{数值 -> 下标}` 的映射：
从左到右遍历 $j = 0, 1, \dots, n - 1$：
- 需要查找满足 $\text{complement} + \text{nums}[j] = \text{target}$ 的前驱补数 $\text{complement} = \text{target} - \text{nums}[j]$。
- **防止有符号减法溢出**：若直接计算 `target - nums[j]`，当 `nums[j] < 0` 且 `target > INT64_MAX + nums[j]` 时会发生上溢；当 `nums[j] > 0` 且 `target < INT64_MIN + nums[j]` 时会发生下溢。若发生溢出，说明在 `int64_t` 的数值范围内不可能存在满足要求的补数，直接跳过当前项。
- 若补数无溢出且在散列表中查到了历史下标 $i$，由于 $j$ 是从左向右首次命中，且表中记录的是补数最早（或任一有效前驱）下标，因此该对下标自动保证了 $j$ 最小；直接写入输出对象并返回 `true`。
- 若未查到，将当前元素 `nums[j]` 及其下标 `j` 插入散列表（若键已存在可保留较小下标以满足 tie-breaking 规则）。

### 2. 最长连续序列与线性时间复杂度分析

若先对数组排序再统计，耗时为 $O(n \log n)$。利用散列集合可压制到期望 $O(n)$：
1. 将 `nums` 中的全部元素插入一个散列集合 `set` 中；
2. 再次遍历集合中的每个数 $x$：
   - 只有当 $x$ 是**连续区间的起点**时才启动向后探索。$x$ 为起点的充要条件是：$x - 1$ **不在**集合中（若 $x == \text{INT64\_MIN}$ 则前驱不存在，直接作为起点）；
   - 若 $x - 1$ 在集合中，说明 $x$ 只是某个更长连续段的中间元素，直接跳过。
3. **复杂度证明**：
   - 每个连续区间有且仅有一个起点；
   - 任何非起点元素在第 2 步的判断中仅耗费 $O(1)$ 查找即被跳过；
   - 包含在某个连续段中的元素，仅在从该段的起点开始向后探测时被访问恰好一次；
   - 因此内层 `while` 循环在整个算法执行期间，对所有区间的向后迭代次数之和严格不超过元素总数 $n$。
   - 集合插入与查询的期望时间均为 $O(1)$，故总体期望时间复杂度为 $O(n)$。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

// --- 轻量级开放寻址散列表 ---
enum { SLOT_FREE = 0, SLOT_BUSY = 1 };

typedef struct {
    int64_t key;
    size_t val;
    unsigned char state;
} KVSlot;

typedef struct {
    KVSlot *slots;
    size_t cap;
    size_t count;
} KVMap;

static uint64_t mix64(uint64_t x) {
    x = (x ^ (x >> 30)) * UINT64_C(0xbf58476d1ce4e5b9);
    x = (x ^ (x >> 27)) * UINT64_C(0x94d049bb133111eb);
    return x ^ (x >> 31);
}

static bool kv_init(KVMap *m, size_t cap) {
    m->slots = calloc(cap, sizeof(KVSlot));
    if (!m->slots) return false;
    m->cap = cap;
    m->count = 0;
    return true;
}

static void kv_free(KVMap *m) {
    free(m->slots);
}

static bool kv_find(const KVMap *m, int64_t k, size_t *out_v) {
    if (m->cap == 0) return false;
    size_t mask = m->cap - 1;
    size_t idx = (size_t)(mix64((uint64_t)k) & mask);
    for (size_t step = 0; step < m->cap; ++step) {
        if (m->slots[idx].state == SLOT_FREE) return false;
        if (m->slots[idx].key == k) {
            if (out_v) *out_v = m->slots[idx].val;
            return true;
        }
        idx = (idx + 1) & mask;
    }
    return false;
}

static void kv_insert_min(KVMap *m, int64_t k, size_t v) {
    size_t mask = m->cap - 1;
    size_t idx = (size_t)(mix64((uint64_t)k) & mask);
    while (m->slots[idx].state == SLOT_BUSY) {
        if (m->slots[idx].key == k) {
            // 保留较小下标
            if (v < m->slots[idx].val) m->slots[idx].val = v;
            return;
        }
        idx = (idx + 1) & mask;
    }
    m->slots[idx].key = k;
    m->slots[idx].val = v;
    m->slots[idx].state = SLOT_BUSY;
    m->count++;
}

bool two_sum_indices(
    const int64_t *nums,
    size_t n,
    int64_t target,
    size_t *out_i,
    size_t *out_j
) {
    if (nums == NULL || n < 2 || out_i == NULL || out_j == NULL) {
        return false;
    }

    size_t cap = 16;
    while (cap < n * 2) {
        if (cap > SIZE_MAX / 2) return false;
        cap *= 2;
    }

    KVMap map;
    if (!kv_init(&map, cap)) return false;

    for (size_t j = 0; j < n; ++j) {
        int64_t cur = nums[j];
        bool has_complement = true;
        int64_t comp = 0;

        // 防有符号减法溢出检查
        if (cur > 0 && target < INT64_MIN + cur) {
            has_complement = false;
        } else if (cur < 0 && target > INT64_MAX + cur) {
            has_complement = false;
        } else {
            comp = target - cur;
        }

        size_t prev_i = 0;
        if (has_complement && kv_find(&map, comp, &prev_i)) {
            *out_i = prev_i;
            *out_j = j;
            kv_free(&map);
            return true;
        }

        kv_insert_min(&map, cur, j);
    }

    kv_free(&map);
    return false;
}

size_t longest_consecutive_run(const int64_t *nums, size_t n) {
    if (nums == NULL || n == 0) {
        return 0;
    }

    size_t cap = 16;
    while (cap < n * 2) {
        if (cap > SIZE_MAX / 2) return 0;
        cap *= 2;
    }

    KVMap set;
    if (!kv_init(&set, cap)) return 0;

    for (size_t i = 0; i < n; ++i) {
        kv_insert_min(&set, nums[i], 0);
    }

    size_t max_run = 0;

    for (size_t i = 0; i < n; ++i) {
        int64_t x = nums[i];

        // 仅当 x 是序列起点（x - 1 不在表中）时向后探索
        bool is_start = false;
        if (x == INT64_MIN) {
            is_start = true;
        } else if (!kv_find(&set, x - 1, NULL)) {
            is_start = true;
        }

        if (is_start) {
            size_t current_run = 1;
            int64_t cur = x;
            while (cur < INT64_MAX && kv_find(&set, cur + 1, NULL)) {
                cur++;
                current_run++;
            }
            if (current_run > max_run) {
                max_run = current_run;
            }
        }
    }

    kv_free(&set);
    return max_run;
}

int main(void) {
    // 1. 验证 two_sum_indices 正常与极值溢出防护
    const int64_t arr1[] = {2, 7, 11, 15};
    size_t idx_i = 0, idx_j = 0;
    assert(two_sum_indices(arr1, 4, 9, &idx_i, &idx_j));
    assert(idx_i == 0 && idx_j == 1);

    // 极值溢出测试：target = INT64_MIN, nums 包含负数与正数
    const int64_t arr_overflow[] = {INT64_MAX, 1, INT64_MIN};
    assert(!two_sum_indices(arr_overflow, 3, INT64_MIN, &idx_i, &idx_j));

    // 2. 验证 longest_consecutive_run
    const int64_t arr2[] = {100, 4, 200, 1, 3, 2};
    assert(longest_consecutive_run(arr2, 6) == 4);

    const int64_t arr3[] = {0, 3, 7, 2, 5, 8, 4, 6, 0, 1};
    assert(longest_consecutive_run(arr3, 10) == 9);

    const int64_t arr4[] = {INT64_MAX, INT64_MAX - 1, INT64_MIN};
    assert(longest_consecutive_run(arr4, 3) == 2);

    return 0;
}
```

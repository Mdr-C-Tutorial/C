# 31309 - 分离链接整数散列映射

## 题目

实现采用分离链接法（Separate Chaining）的 64 位整数键值映射 `IntHashMap`：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct IntHashMap IntHashMap;

IntHashMap *int_map_create(void);
void int_map_destroy(IntHashMap *map);
bool int_map_put(
    IntHashMap *map,
    int64_t key,
    uint64_t value,
    bool *out_replaced
);
bool int_map_get(
    const IntHashMap *map,
    int64_t key,
    uint64_t *out_value
);
bool int_map_remove(
    IntHashMap *map,
    int64_t key,
    bool *out_removed
);
size_t int_map_size(const IntHashMap *map);
```

要求：

1. 桶数组容量始终保持为 $2$ 的整数次幂，通过位与掩码定位桶下标；映射表拥有全部链表结点。
2. `int_map_put` 遇到已存在的键时原地更新其值并置 `*out_replaced = true`；插入新键若使 `size > capacity`，先将桶数组容量扩容为 $2$ 倍，并将旧桶中的已有链表结点直接重新挂接到新桶数组中（重散列时不得重新 `malloc` 链表结点）。
3. `int_map_get` 命中时写入 `*out_value` 并返回 `true`；键不存在或参数无效时返回 `false`，且不得修改 `*out_value`。
4. `int_map_remove` 命中时摘除并释放对应结点，置 `*out_removed = true`；键不存在时成功返回并置 `*out_removed = false`。
5. 内存分配失败、容量或大小计算溢出、参数无效时返回 `false`，原映射表与输出对象保持不变；`int_map_size(NULL)` 返回 `0`，`int_map_destroy(NULL)` 安全。

## 解析

分离链接散列表将桶数组 `buckets` 表示为长度为 `capacity`（$2$ 的整数次幂）的单向链表头指针数组。对于任意键 `key`，使用 64 位混合散列函数计算散列值后，通过位掩码 `hash & (capacity - 1)` 在 $O(1)$ 时间内定位桶下标。

实现中有两个核心工程要点：

1. **强异常安全的插入与扩容顺序**：当 `int_map_put` 确认键不存在、需要插入新结点时，若 `size + 1 > capacity` 则需同时完成“新结点分配”和“新桶数组分配”。为了保证任何一步失败都不破坏原表，必须**先分配新链表结点**，再按需执行桶数组扩容；若扩容分配新桶数组失败，直接释放刚分配的孤立结点并返回 `false`，此时旧桶数组中的链表尚未发生任何指针改动。
2. **零结点分配的原地重挂接（Node Relinking）**：扩容时仅调用 `calloc` 分配长度为 `2 * capacity` 的新桶头指针数组，随后遍历旧桶中的每个现有结点 `node`，直接修改其 `node->next` 将其头插挂入 `new_buckets[index]`，最后释放旧桶指针数组本身。整个重散列过程不重新 `malloc` 或 `free` 任何链表结点。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct IntHashMap IntHashMap;
typedef struct IntMapNode IntMapNode;

struct IntMapNode {
    int64_t key;
    uint64_t value;
    IntMapNode *next;
};

struct IntHashMap {
    IntMapNode **buckets;
    size_t capacity;
    size_t size;
};

static uint64_t hash_int64(int64_t key) {
    uint64_t x = (uint64_t)key;
    x = (x ^ (x >> 30)) * UINT64_C(0xbf58476d1ce4e5b9);
    x = (x ^ (x >> 27)) * UINT64_C(0x94d049bb133111eb);
    return x ^ (x >> 31);
}

static size_t bucket_index(int64_t key, size_t capacity) {
    return (size_t)(hash_int64(key) & (uint64_t)(capacity - 1));
}

static bool rehash_relink(IntHashMap *map, size_t new_capacity) {
    if (new_capacity == 0 ||
        new_capacity > SIZE_MAX / sizeof *map->buckets) {
        return false;
    }
    IntMapNode **new_buckets = calloc(
        new_capacity, sizeof *new_buckets
    );
    if (new_buckets == NULL) {
        return false;
    }

    for (size_t i = 0; i < map->capacity; ++i) {
        IntMapNode *cursor = map->buckets[i];
        while (cursor != NULL) {
            IntMapNode *next = cursor->next;
            size_t target = bucket_index(cursor->key, new_capacity);
            cursor->next = new_buckets[target];
            new_buckets[target] = cursor;
            cursor = next;
        }
    }

    free(map->buckets);
    map->buckets = new_buckets;
    map->capacity = new_capacity;
    return true;
}

IntHashMap *int_map_create(void) {
    IntHashMap *map = malloc(sizeof *map);
    if (map == NULL) {
        return NULL;
    }
    size_t initial_capacity = 8;
    map->buckets = calloc(initial_capacity, sizeof *map->buckets);
    if (map->buckets == NULL) {
        free(map);
        return NULL;
    }
    map->capacity = initial_capacity;
    map->size = 0;
    return map;
}

void int_map_destroy(IntHashMap *map) {
    if (map == NULL) {
        return;
    }
    for (size_t i = 0; i < map->capacity; ++i) {
        IntMapNode *cursor = map->buckets[i];
        while (cursor != NULL) {
            IntMapNode *next = cursor->next;
            free(cursor);
            cursor = next;
        }
    }
    free(map->buckets);
    free(map);
}

bool int_map_put(
    IntHashMap *map,
    int64_t key,
    uint64_t value,
    bool *out_replaced
) {
    if (map == NULL || out_replaced == NULL) {
        return false;
    }

    size_t index = bucket_index(key, map->capacity);
    for (IntMapNode *cursor = map->buckets[index];
         cursor != NULL;
         cursor = cursor->next) {
        if (cursor->key == key) {
            cursor->value = value;
            *out_replaced = true;
            return true;
        }
    }

    if (map->size == SIZE_MAX) {
        return false;
    }
    IntMapNode *node = malloc(sizeof *node);
    if (node == NULL) {
        return false;
    }

    if (map->size + 1 > map->capacity) {
        if (map->capacity > SIZE_MAX / 2 ||
            !rehash_relink(map, map->capacity * 2)) {
            free(node);
            return false;
        }
        index = bucket_index(key, map->capacity);
    }

    node->key = key;
    node->value = value;
    node->next = map->buckets[index];
    map->buckets[index] = node;
    ++map->size;
    *out_replaced = false;
    return true;
}

bool int_map_get(
    const IntHashMap *map,
    int64_t key,
    uint64_t *out_value
) {
    if (map == NULL || out_value == NULL) {
        return false;
    }
    size_t index = bucket_index(key, map->capacity);
    for (const IntMapNode *cursor = map->buckets[index];
         cursor != NULL;
         cursor = cursor->next) {
        if (cursor->key == key) {
            *out_value = cursor->value;
            return true;
        }
    }
    return false;
}

bool int_map_remove(
    IntHashMap *map,
    int64_t key,
    bool *out_removed
) {
    if (map == NULL || out_removed == NULL) {
        return false;
    }
    size_t index = bucket_index(key, map->capacity);
    IntMapNode **link = &map->buckets[index];
    while (*link != NULL) {
        if ((*link)->key == key) {
            IntMapNode *victim = *link;
            *link = victim->next;
            free(victim);
            --map->size;
            *out_removed = true;
            return true;
        }
        link = &(*link)->next;
    }
    *out_removed = false;
    return true;
}

size_t int_map_size(const IntHashMap *map) {
    return map == NULL ? 0 : map->size;
}

int main(void) {
    IntHashMap *map = int_map_create();
    assert(map != NULL);
    bool replaced = false;
    bool removed = false;
    uint64_t val = 0;

    for (int64_t i = 0; i < 32; ++i) {
        assert(int_map_put(map, i * 17 - 100, (uint64_t)(i + 1000), &replaced));
        assert(!replaced);
    }
    assert(int_map_size(map) == 32);
    assert(int_map_put(map, -100, 9999, &replaced) && replaced);
    assert(int_map_get(map, -100, &val) && val == 9999);
    assert(int_map_remove(map, -100, &removed) && removed);
    assert(!int_map_get(map, -100, &val));
    assert(int_map_remove(map, -100, &removed) && !removed);
    assert(int_map_size(map) == 31);

    int_map_destroy(map);
    int_map_destroy(NULL);
    return 0;
}
```

装载因子控制在 $\alpha = n / m \le 1$ 以内，在简单一致散列假设下，`put`、`get` 与 `remove` 的期望时间复杂度均为 $O(1)$；单次扩容重挂接耗时 $O(n)$，分摊到每次插入仍为 $O(1)$，空间复杂度为 $O(n + m)$。

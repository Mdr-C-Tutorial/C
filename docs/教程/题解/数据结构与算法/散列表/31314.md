# 31314 - Robin Hood 散列表

## 题目

实现带 **探测序列长度（Probe Sequence Length, PSL）** 追踪、**早停查找（Early-Exit Lookup）** 与 **后移删除** 的 **Robin Hood 散列表** `RobinHoodMap`：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct RobinHoodMap RobinHoodMap;

RobinHoodMap *rh_map_create(void);
void rh_map_destroy(RobinHoodMap *map);
bool rh_map_put(
    RobinHoodMap *map,
    int64_t key,
    int64_t value,
    bool *out_replaced
);
bool rh_map_get(
    const RobinHoodMap *map,
    int64_t key,
    int64_t *out_value
);
bool rh_map_remove(
    RobinHoodMap *map,
    int64_t key,
    bool *out_removed
);
size_t rh_map_size(const RobinHoodMap *map);
size_t rh_map_max_psl(const RobinHoodMap *map);
```

要求：

1. 每个占用槽记录其当前元素偏离初始散列槽 $h(k) \bmod m$ 的探测步数 `psl`（在初始槽处 `psl == 0`）。
2. **插入与更新**：沿线性探测序列前进，每走一步待插项的 `curr_psl` 加 1。若在发生交换前遇到相同键，则原地更新其值并返回；一旦遇到某个占用槽满足 `curr_psl > slot.psl`（“劫富济贫”条件），先在剩余探测段确认目标键是否已存在（或在插入前先查找），随后将待插项与该槽元素交换，继续为被置换出的元素寻找后续槽位，直至放入空槽。
3. **早停查找**：在 `rh_map_get` 与 `rh_map_remove` 的定位阶段，若当前探测步数 `psl` 严格大于当前槽内元素的 `slot.psl`，立即终止探测并返回未命中。证明为什么在 Robin Hood 不变式下，后续槽中绝不可能再出现目标键。
4. **后移删除**：删除槽 `idx` 后，只要下一个槽 `next = (idx + 1) & mask` 被占用且 `slot[next].psl > 0`，就将其前移到 `idx` 并令其 `psl` 减 1；遇到空槽或 `psl == 0` 的槽时停止，并将最后的空洞置为空槽。
5. `rh_map_max_psl` 返回当前表中所有占用槽的最大 `psl`（空表返回 `0`）。

## 解析

### 1. Robin Hood 核心思想：“劫富济贫”

在线性探测中，先插入的元素哪怕独占大段连续槽，后插入的元素也只能持续向后退让，导致探测序列长度（PSL）方差极大，个别元素的查找耗时极长。
Robin Hood 散列定义：**谁的 PSL 小（离理想位置近），谁就是“富人”；谁的 PSL 大（漂泊更远），谁就是“穷人”**。
当插入一个带当前步数 `curr_psl` 的元素时：
- 若遭遇的槽已有元素 `slot.psl < curr_psl`，说明已有元素比当前元素更富裕，立刻“劫富”：**将当前元素安放在此处，而将旧元素踢出并继续向后为其寻找新槽位**！
- 这一规则极大平抑了 PSL 的方差，使得所有元素的 PSL 紧密聚集在一个极小的均值附近。

### 2. 早停查找（Early Exit）不变式证明

**不变式**：对于同属于一个冲突探测簇的连续元素，由于 Robin Hood 交换规则，它们各自的 PSL 沿着探测前进方向是**非递减**的（经过初始槽差异补偿后有序排列）。
当我们在查找键 $k$ 时：
- 从理想槽 $h(k) \bmod m$ 开始，每向前一步，我们如果存在于该表中，当前所应具有的探测步数即为 `psl`；
- 如果当前检查的槽内元素也是从前面漂移过来的，而它的 `slot.psl < psl`，说明在当时插入时，如果键 $k$ 在表中，它的 `curr_psl` 必然比当前槽的住户大，它早就应该在此槽之前（或者在此槽发生交换）将住户踢走；
- 它既然没有出现在这里，而且当前住户的偏离距离比它还小，那么在后续槽中更加不可能出现键 $k$！
- 因此，一旦遇到 `slot.empty || psl > slot.psl`，**可以立刻断定键不存在，提前返回未命中**！这使未命中查找的耗时大幅降低。

### 3. 无墓碑后移删除

当槽 `idx` 被删除腾空后：
- 考察下一个槽 `next = (idx + 1) & mask`；
- 若 `slot[next]` 是空槽，或者 `slot[next].psl == 0`（说明它正好坐落在自己的理想槽上，不能再往前移），停止后移；
- 若 `slot[next].psl > 0`，说明它曾经向后漂移过至少一格，将其安全地前移到 `idx`，并将其 `psl` 减 1；随后令 `idx = next` 继续考察后续。整个过程完全无需墓碑。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct RobinHoodMap RobinHoodMap;

bool rh_map_get(const RobinHoodMap *map, int64_t key, int64_t *out_value);

#define RH_EMPTY_PSL -1

typedef struct {
    int64_t key;
    int64_t value;
    int32_t psl; // -1 表示槽位空闲
} RhBucket;

struct RobinHoodMap {
    RhBucket *buckets;
    size_t capacity;
    size_t size;
};

static uint64_t rh_hash(int64_t key) {
    uint64_t x = (uint64_t)key;
    x = (x ^ (x >> 30)) * UINT64_C(0xbf58476d1ce4e5b9);
    x = (x ^ (x >> 27)) * UINT64_C(0x94d049bb133111eb);
    return x ^ (x >> 31);
}

RobinHoodMap *rh_map_create(void) {
    RobinHoodMap *map = malloc(sizeof(RobinHoodMap));
    if (!map) return NULL;
    map->capacity = 16;
    map->size = 0;
    map->buckets = malloc(map->capacity * sizeof(RhBucket));
    if (!map->buckets) {
        free(map);
        return NULL;
    }
    for (size_t i = 0; i < map->capacity; ++i) {
        map->buckets[i].psl = RH_EMPTY_PSL;
    }
    return map;
}

void rh_map_destroy(RobinHoodMap *map) {
    if (!map) return;
    free(map->buckets);
    free(map);
}

static void rh_insert_internal(RobinHoodMap *map, int64_t key, int64_t value) {
    RhBucket curr = { .key = key, .value = value, .psl = 0 };
    size_t mask = map->capacity - 1;
    size_t idx = (size_t)(rh_hash(curr.key) & mask);

    for (;;) {
        RhBucket *target = &map->buckets[idx];

        if (target->psl == RH_EMPTY_PSL) {
            *target = curr;
            map->size++;
            return;
        }

        if (target->key == curr.key) {
            target->value = curr.value;
            return;
        }

        // 劫富济贫：若当前项漂移距离更长，交换之
        if (curr.psl > target->psl) {
            RhBucket tmp = *target;
            *target = curr;
            curr = tmp;
        }

        curr.psl++;
        idx = (idx + 1) & mask;
    }
}

static bool rh_resize(RobinHoodMap *map, size_t new_cap) {
    RhBucket *old_buckets = map->buckets;
    size_t old_cap = map->capacity;

    RhBucket *new_buckets = malloc(new_cap * sizeof(RhBucket));
    if (!new_buckets) return false;
    for (size_t i = 0; i < new_cap; ++i) {
        new_buckets[i].psl = RH_EMPTY_PSL;
    }

    map->buckets = new_buckets;
    map->capacity = new_cap;
    map->size = 0;

    for (size_t i = 0; i < old_cap; ++i) {
        if (old_buckets[i].psl != RH_EMPTY_PSL) {
            rh_insert_internal(map, old_buckets[i].key, old_buckets[i].value);
        }
    }

    free(old_buckets);
    return true;
}

bool rh_map_put(
    RobinHoodMap *map,
    int64_t key,
    int64_t value,
    bool *out_replaced
) {
    if (!map || !out_replaced) return false;

    // 先检查是否存在以确定 replaced
    int64_t existing = 0;
    if (rh_map_get(map, key, &existing)) {
        // 原地更新
        size_t mask = map->capacity - 1;
        size_t idx = (size_t)(rh_hash(key) & mask);
        int32_t psl = 0;
        while (true) {
            RhBucket *target = &map->buckets[idx];
            if (target->psl == RH_EMPTY_PSL || psl > target->psl) break;
            if (target->key == key) {
                target->value = value;
                *out_replaced = true;
                return true;
            }
            psl++;
            idx = (idx + 1) & mask;
        }
    }

    // 新增元素前检查扩容 (负载上限 0.75)
    if ((map->size + 1) * 4 > map->capacity * 3) {
        if (map->capacity > SIZE_MAX / 2 ||
            !rh_resize(map, map->capacity * 2)) {
            return false;
        }
    }

    rh_insert_internal(map, key, value);
    *out_replaced = false;
    return true;
}

bool rh_map_get(
    const RobinHoodMap *map,
    int64_t key,
    int64_t *out_value
) {
    if (!map || !out_value || map->capacity == 0) return false;

    size_t mask = map->capacity - 1;
    size_t idx = (size_t)(rh_hash(key) & mask);
    int32_t psl = 0;

    for (;;) {
        const RhBucket *target = &map->buckets[idx];

        // 早停查找：若槽空，或当前探测步数已超越目标槽内住户的偏离步数，键必然不存在
        if (target->psl == RH_EMPTY_PSL || psl > target->psl) {
            return false;
        }

        if (target->key == key) {
            *out_value = target->value;
            return true;
        }

        psl++;
        idx = (idx + 1) & mask;
    }
}

bool rh_map_remove(
    RobinHoodMap *map,
    int64_t key,
    bool *out_removed
) {
    if (!map || !out_removed || map->capacity == 0) return false;

    size_t mask = map->capacity - 1;
    size_t idx = (size_t)(rh_hash(key) & mask);
    int32_t psl = 0;
    bool found = false;
    size_t del_idx = 0;

    // 1. 定位
    for (;;) {
        RhBucket *target = &map->buckets[idx];
        if (target->psl == RH_EMPTY_PSL || psl > target->psl) {
            *out_removed = false;
            return true;
        }
        if (target->key == key) {
            del_idx = idx;
            found = true;
            break;
        }
        psl++;
        idx = (idx + 1) & mask;
    }

    if (!found) {
        *out_removed = false;
        return true;
    }

    // 2. 后移删除
    size_t cur = del_idx;
    for (;;) {
        size_t next = (cur + 1) & mask;
        RhBucket *nxt = &map->buckets[next];

        if (nxt->psl == RH_EMPTY_PSL || nxt->psl == 0) {
            map->buckets[cur].psl = RH_EMPTY_PSL;
            break;
        }

        map->buckets[cur] = *nxt;
        map->buckets[cur].psl--;
        cur = next;
    }

    map->size--;
    *out_removed = true;
    return true;
}

size_t rh_map_size(const RobinHoodMap *map) {
    return map ? map->size : 0;
}

size_t rh_map_max_psl(const RobinHoodMap *map) {
    if (!map) return 0;
    int32_t max_p = 0;
    for (size_t i = 0; i < map->capacity; ++i) {
        if (map->buckets[i].psl > max_p) {
            max_p = map->buckets[i].psl;
        }
    }
    return (size_t)max_p;
}

int main(void) {
    RobinHoodMap *map = rh_map_create();
    assert(map != NULL);

    bool replaced = false;
    bool removed = false;
    int64_t val = 0;

    for (int64_t i = 1; i <= 30; ++i) {
        assert(rh_map_put(map, i * 13, i * 100, &replaced));
        assert(!replaced);
    }
    assert(rh_map_size(map) == 30);

    // 验证最大 PSL 保持在极小常数水平（平抑方差）
    size_t max_p = rh_map_max_psl(map);
    assert(max_p < 10);

    for (int64_t i = 1; i <= 30; ++i) {
        assert(rh_map_get(map, i * 13, &val) && val == i * 100);
    }

    // 验证未命中早停查找
    assert(!rh_map_get(map, 999999, &val));

    // 验证删除与后移
    assert(rh_map_remove(map, 26, &removed) && removed);
    assert(!rh_map_get(map, 26, &val));
    assert(rh_map_size(map) == 29);

    rh_map_destroy(map);
    return 0;
}
```

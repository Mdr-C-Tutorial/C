# 题解：31315

## 题目

实现保证**最坏情况查找与删除时间复杂度为严格 $O(1)$** 的 **布谷鸟散列集合 (Cuckoo Hash Set)**，并分析散列洪泛攻击（HashDoS）的构造与防御：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct CuckooIntSet CuckooIntSet;

CuckooIntSet *cuckoo_set_create(void);
void cuckoo_set_destroy(CuckooIntSet *set);
bool cuckoo_set_contains(const CuckooIntSet *set, int64_t key);
bool cuckoo_set_insert(
    CuckooIntSet *set,
    int64_t key,
    bool *out_inserted
);
bool cuckoo_set_remove(
    CuckooIntSet *set,
    int64_t key,
    bool *out_removed
);
size_t cuckoo_set_size(const CuckooIntSet *set);
```

要求：

1. 维护两张容量均为 $m$（$2$ 的幂）的子表 $T_1, T_2$ 以及两个带独立 64 位随机种子的散列函数 $h_1(k), h_2(k)$。任意已存在键 $k$ 必须且仅能存放在 $T_1[h_1(k) \bmod m]$ 或 $T_2[h_2(k) \bmod m]$ 之一。
2. `cuckoo_set_contains` 与 `cuckoo_set_remove` **最多只允许检查这两个槽位**（严格最坏情况 $O(1)$ 次比较），不得进行任何线性扫描或链表遍历。
3. `cuckoo_set_insert`：若键已存在于 $T_1[h_1(k)]$ 或 $T_2[h_2(k)]$，置 `*out_inserted = false` 并返回 `true`；若两槽之一为空，直接放入。若两槽均满，将新键放入 $T_1[h_1(k)]$ 并踢出原住户 $x$，再将 $x$ 放入其位于 $T_2[h_2(x)]$ 的位置并踢出其中原住户，如此在 $T_1$ 与 $T_2$ 之间交替踢出。
4. 若单次插入的交替踢出次数达到上限（例如 `MAX_KICKS = 64`）仍未找到空槽，或者总元素数超过容量阈值（$2m$ 的 $45\%$），则将子表容量扩容（或更新随机种子 `seed1, seed2`）并将全部元素（含当前被踢出的悬空键）重新散列插入新表；重散列失败时必须保持原集合与输出对象不变。
5. 在题解中回答：为什么无密钥确定性散列函数容易遭受碰撞洪泛攻击（HashDoS），以及布谷鸟散列在遭遇小规模环路冲突时为什么必须支持**更换散列种子重散列**而不能仅靠固定散列函数扩容。

## 解析

### 1. 布谷鸟散列的最坏情况 $O(1)$ 保证

布谷鸟散列（Pagh & Rodler, 2004）的结构特性：
- 任意键 $k$ 在整个数据结构中**最多只有两个合法家园**：$T_1[h_1(k) \bmod m]$ 和 $T_2[h_2(k) \bmod m]$；
- 查询操作只需要读取这两个固定槽位，执行两次相等比对；删除操作同样只需要在这两个槽位中将命中项置为空。
- 因此，查找与删除的时间复杂度在**最坏情况下**都是严格确定性的 $O(1)$，不存在任何链表遍历、墓碑跳步或探测簇拉长的问题。

### 2. 插入踢出机制与二分图环路

插入新键时模拟布谷鸟借巢产卵的行为：
- 若首选槽已被占用，新键强行进驻该槽，将原住户 $x$ 踢出；
- 被踢出的 $x$ 转向其在另一张子表中的备选槽，若该备选槽也被占用，则再次踢出该处的住户；
- 这一过程在两张表之间交替进行。若二分图中不存在有向环，踢出链条会在几步之内找到一个空槽并停机；
- 若键集合在当前的散列映射下形成了闭合置换环（环路内边的数量超过了顶点的容纳能力），踢出过程将陷入无限死循环。因此设定最大踢出步数（例如 64 次），一旦超标即判定当前种子下的二分图拓扑不可行，触发重散列。

### 3. HashDoS 攻击与种子轮换的必要性

1. **HashDoS 攻击机理**：
   若散列表使用公开、无密钥的确定性散列函数（例如单凭字符串 ASCII 码计算的简单多项式），攻击者可以在本地离线穷举，构造出数万个散列到完全相同槽位的恶意键（如 `Aa`, `BB` 碰撞，或针对模 $2^p$ 的特定构造）。当攻击者向服务端发送包含这些键的单个请求（如万级字段的 POST 表单）时，拉链法或开放寻址表会被迫退化为单链表或全表线性扫描，单次请求消耗数十秒 CPU 时间，耗尽服务器计算资源。
2. **为什么布谷鸟散列必须支持更换种子**：
   在布谷鸟散列中，如果仅仅因为几个特定的键在当前固定散列函数下产生了局部的拓扑死锁环（例如 3 个键仅争夺 2 个槽），如果只是翻倍容量但不更换散列函数，这几个键在扩容后若仍然以相同的相对模式映射到对应的两个槽，死锁环依然存在；
   更换 64 位独立随机种子相当于从散列函数族中重新随机选取新的独立边投影，能以极高概率打破局部小环路，使二分图恢复为无环或稀疏图，从而以很高的空间利用率完成插入。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct CuckooIntSet CuckooIntSet;

#define CUCKOO_MAX_KICKS 64

typedef struct {
    int64_t key;
    bool occupied;
} CuckooSlot;

struct CuckooIntSet {
    CuckooSlot *t1;
    CuckooSlot *t2;
    size_t capacity; // 单张子表的槽数 (2 的幂)
    size_t size;     // 两表有效元素总和
    uint64_t seed1;
    uint64_t seed2;
};

static uint64_t splitmix64(uint64_t x) {
    x = (x ^ (x >> 30)) * UINT64_C(0xbf58476d1ce4e5b9);
    x = (x ^ (x >> 27)) * UINT64_C(0x94d049bb133111eb);
    return x ^ (x >> 31);
}

static inline size_t h1_idx(const CuckooIntSet *set, int64_t key) {
    return (size_t)(splitmix64((uint64_t)key ^ set->seed1) & (set->capacity - 1));
}

static inline size_t h2_idx(const CuckooIntSet *set, int64_t key) {
    return (size_t)(splitmix64((uint64_t)key ^ set->seed2) & (set->capacity - 1));
}

CuckooIntSet *cuckoo_set_create(void) {
    CuckooIntSet *set = malloc(sizeof(CuckooIntSet));
    if (!set) return NULL;
    set->capacity = 16;
    set->size = 0;
    set->seed1 = UINT64_C(0x123456789abcdef0);
    set->seed2 = UINT64_C(0xfedcba9876543210);

    set->t1 = calloc(set->capacity, sizeof(CuckooSlot));
    set->t2 = calloc(set->capacity, sizeof(CuckooSlot));
    if (!set->t1 || !set->t2) {
        free(set->t1);
        free(set->t2);
        free(set);
        return NULL;
    }
    return set;
}

void cuckoo_set_destroy(CuckooIntSet *set) {
    if (!set) return;
    free(set->t1);
    free(set->t2);
    free(set);
}

// 严格最坏情况 O(1) 查找：只检查两个固定槽位
bool cuckoo_set_contains(const CuckooIntSet *set, int64_t key) {
    if (!set || set->capacity == 0) return false;
    size_t p1 = h1_idx(set, key);
    if (set->t1[p1].occupied && set->t1[p1].key == key) {
        return true;
    }
    size_t p2 = h2_idx(set, key);
    if (set->t2[p2].occupied && set->t2[p2].key == key) {
        return true;
    }
    return false;
}

// 严格最坏情况 O(1) 删除：只检查两个固定槽位
bool cuckoo_set_remove(
    CuckooIntSet *set,
    int64_t key,
    bool *out_removed
) {
    if (!set || !out_removed) return false;
    size_t p1 = h1_idx(set, key);
    if (set->t1[p1].occupied && set->t1[p1].key == key) {
        set->t1[p1].occupied = false;
        set->size--;
        *out_removed = true;
        return true;
    }
    size_t p2 = h2_idx(set, key);
    if (set->t2[p2].occupied && set->t2[p2].key == key) {
        set->t2[p2].occupied = false;
        set->size--;
        *out_removed = true;
        return true;
    }
    *out_removed = false;
    return true;
}

size_t cuckoo_set_size(const CuckooIntSet *set) {
    return set ? set->size : 0;
}

static bool cuckoo_rehash_and_insert(
    CuckooIntSet *set,
    size_t new_cap,
    uint64_t extra_key
);

static bool cuckoo_insert_no_resize(
    CuckooIntSet *set,
    int64_t key
) {
    int64_t cur_key = key;

    for (int kick = 0; kick < CUCKOO_MAX_KICKS; ++kick) {
        // 尝试进驻 T1
        size_t p1 = h1_idx(set, cur_key);
        if (!set->t1[p1].occupied) {
            set->t1[p1].key = cur_key;
            set->t1[p1].occupied = true;
            set->size++;
            return true;
        }

        // 踢出 T1 原住户
        int64_t victim1 = set->t1[p1].key;
        set->t1[p1].key = cur_key;

        // 尝试将 victim1 放入 T2
        size_t p2 = h2_idx(set, victim1);
        if (!set->t2[p2].occupied) {
            set->t2[p2].key = victim1;
            set->t2[p2].occupied = true;
            set->size++;
            return true;
        }

        // 踢出 T2 原住户
        cur_key = set->t2[p2].key;
        set->t2[p2].key = victim1;
    }

    // 踢出超限，触发扩容/换种子
    return false;
}

static bool cuckoo_rehash_and_insert(
    CuckooIntSet *set,
    size_t new_cap,
    uint64_t extra_key
) {
    if (new_cap > SIZE_MAX / sizeof(CuckooSlot) / 2) return false;

    // 收集旧表中所有元素
    size_t total_keys = set->size + 1;
    int64_t *all_keys = malloc(total_keys * sizeof(int64_t));
    if (!all_keys) return false;

    size_t pos = 0;
    for (size_t i = 0; i < set->capacity; ++i) {
        if (set->t1[i].occupied) all_keys[pos++] = set->t1[i].key;
        if (set->t2[i].occupied) all_keys[pos++] = set->t2[i].key;
    }
    all_keys[pos++] = (int64_t)extra_key;

    // 分配新子表
    CuckooSlot *nt1 = calloc(new_cap, sizeof(CuckooSlot));
    CuckooSlot *nt2 = calloc(new_cap, sizeof(CuckooSlot));
    if (!nt1 || !nt2) {
        free(all_keys);
        free(nt1);
        free(nt2);
        return false;
    }

    // 临时更新拓扑尝试插入
    CuckooSlot *old_t1 = set->t1;
    CuckooSlot *old_t2 = set->t2;
    uint64_t old_s1 = set->seed1;
    uint64_t old_s2 = set->seed2;

    set->t1 = nt1;
    set->t2 = nt2;
    set->capacity = new_cap;
    set->size = 0;
    // 轮换新随机种子
    set->seed1 = splitmix64(old_s1 + UINT64_C(0x9e3779b97f4a7c15));
    set->seed2 = splitmix64(old_s2 + UINT64_C(0xbf58476d1ce4e5b9));

    for (size_t i = 0; i < total_keys; ++i) {
        if (!cuckoo_insert_no_resize(set, all_keys[i])) {
            // 重散列依然踢出超限，递归扩大容量重试
            int64_t failed_key = all_keys[i];
            free(all_keys);
            return cuckoo_rehash_and_insert(set, new_cap * 2, (uint64_t)failed_key);
        }
    }

    free(all_keys);
    free(old_t1);
    free(old_t2);
    return true;
}

bool cuckoo_set_insert(
    CuckooIntSet *set,
    int64_t key,
    bool *out_inserted
) {
    if (!set || !out_inserted) return false;

    if (cuckoo_set_contains(set, key)) {
        *out_inserted = false;
        return true;
    }

    // 装载因子阈值：总容量 2 * capacity 的 45%
    if ((set->size + 1) * 100 > set->capacity * 2 * 45) {
        if (!cuckoo_rehash_and_insert(set, set->capacity * 2, (uint64_t)key)) {
            return false;
        }
        *out_inserted = true;
        return true;
    }

    if (cuckoo_insert_no_resize(set, key)) {
        *out_inserted = true;
        return true;
    }

    // 踢出超限，更换种子或扩容重建
    if (!cuckoo_rehash_and_insert(set, set->capacity * 2, (uint64_t)key)) {
        return false;
    }
    *out_inserted = true;
    return true;
}

int main(void) {
    CuckooIntSet *set = cuckoo_set_create();
    assert(set != NULL);

    bool inserted = false;
    bool removed = false;

    for (int64_t i = 1; i <= 50; ++i) {
        assert(cuckoo_set_insert(set, i * 19, &inserted));
        assert(inserted);
    }
    assert(cuckoo_set_size(set) == 50);

    // 验证严格 O(1) 的存在性检查
    for (int64_t i = 1; i <= 50; ++i) {
        assert(cuckoo_set_contains(set, i * 19));
    }
    assert(!cuckoo_set_contains(set, 1000000));

    // 重复插入幂等性
    assert(cuckoo_set_insert(set, 19, &inserted) && !inserted);
    assert(cuckoo_set_size(set) == 50);

    // 删除
    assert(cuckoo_set_remove(set, 19, &removed) && removed);
    assert(!cuckoo_set_contains(set, 19));
    assert(cuckoo_set_remove(set, 19, &removed) && !removed);
    assert(cuckoo_set_size(set) == 49);

    cuckoo_set_destroy(set);
    return 0;
}
```

# 题解：31313

## 题目

结合散列表与双向链表，实现固定容量、所有操作期望时间复杂度均为 $O(1)$ 的 **LRU（最近最少使用，Least Recently Used）缓存** `LruCache`：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct LruCache LruCache;

LruCache *lru_create(size_t capacity);
void lru_destroy(LruCache *cache);
bool lru_get(LruCache *cache, int64_t key, int64_t *out_value);
bool lru_put(
    LruCache *cache,
    int64_t key,
    int64_t value,
    bool *out_evicted,
    int64_t *out_evicted_key
);
size_t lru_size(const LruCache *cache);
```

要求：

1. `lru_create(capacity)` 在 `capacity == 0` 或分配失败时返回 `NULL`。
2. `lru_get` 查询键 `key`：若命中则写出 `*out_value`、将该条目提升为“最近刚使用”（移至双向链表头部）并返回 `true`；未命中或参数无效时返回 `false` 且不修改 `*out_value`。
3. `lru_put` 插入或更新键值对：
   - 若 `key` 已存在，更新其值并移至链表头部，置 `*out_evicted = false`（若 `out_evicted != NULL`）；
   - 若 `key` 不存在且当前 `size == capacity`，先淘汰双向链表尾部“最久未使用”的条目，从散列表中移除该旧键，置 `*out_evicted = true` 并将被淘汰的旧键写入 `*out_evicted_key`（若对应输出指针非空），再插入新键到链表头部（可直接复用被淘汰结点的内存）。
4. `lru_get` 与 `lru_put` 的期望时间复杂度必须为 $O(1)$，不得在每次访问时线性扫描链表。

## 解析

LRU 缓存要求在 $O(1)$ 时间内完成：
1. **按键索引**：给定一个键，能立即定位到对应的值和访问状态（由**散列表**负责，$O(1)$ 期望查找）；
2. **时序维护**：任意访问命中时，能以 $O(1)$ 代价将该元素提升至“最新使用”状态；淘汰时能以 $O(1)$ 代价取出“最久未使用”元素（由带哨兵结点的**双向循环链表**负责）。

### 数据结构协同设计
- 链表结点结构 `LruNode` 同时包含数据载荷与链表指针：
  `{ int64_t key, int64_t value, LruNode *prev, LruNode *next }`；
- 散列表桶中存放指向 `LruNode` 的指针；为了拉链法消解冲突，结点内部还可以包含 `LruNode *hash_next`；
- 链表维护哑哨兵节点 `head`，`head->next` 为最新元素，`head->prev` 为最旧元素；
- 当缓存满且插入新键时，尾结点 `victim = head->prev` 即为淘汰目标。从散列表中摘除 `victim->key` 后，直接改写 `victim->key = new_key, victim->value = new_val` 并将其重新挂接在散列表与链表头部，免去了重复的 `free` 与 `malloc` 开销。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct LruCache LruCache;
typedef struct LruNode LruNode;

struct LruNode {
    int64_t key;
    int64_t value;
    LruNode *prev;
    LruNode *next;
    LruNode *hnext; // 散列表拉链指针
};

struct LruCache {
    size_t capacity;
    size_t size;
    LruNode head; // 双向循环链表哑哨兵
    LruNode **buckets;
    size_t bucket_count;
};

static uint64_t hash_key64(int64_t k) {
    uint64_t x = (uint64_t)k;
    x = (x ^ (x >> 30)) * UINT64_C(0xbf58476d1ce4e5b9);
    x = (x ^ (x >> 27)) * UINT64_C(0x94d049bb133111eb);
    return x ^ (x >> 31);
}

static void list_detach(LruNode *node) {
    node->prev->next = node->next;
    node->next->prev = node->prev;
}

static void list_push_front(LruCache *c, LruNode *node) {
    node->next = c->head.next;
    node->prev = &c->head;
    c->head.next->prev = node;
    c->head.next = node;
}

static LruNode *hash_find(const LruCache *c, int64_t key) {
    size_t idx = (size_t)(hash_key64(key) & (c->bucket_count - 1));
    for (LruNode *cur = c->buckets[idx]; cur != NULL; cur = cur->hnext) {
        if (cur->key == key) {
            return cur;
        }
    }
    return NULL;
}

static void hash_insert(LruCache *c, LruNode *node) {
    size_t idx = (size_t)(hash_key64(node->key) & (c->bucket_count - 1));
    node->hnext = c->buckets[idx];
    c->buckets[idx] = node;
}

static void hash_remove(LruCache *c, int64_t key) {
    size_t idx = (size_t)(hash_key64(key) & (c->bucket_count - 1));
    LruNode **link = &c->buckets[idx];
    while (*link != NULL) {
        if ((*link)->key == key) {
            *link = (*link)->hnext;
            return;
        }
        link = &(*link)->hnext;
    }
}

LruCache *lru_create(size_t capacity) {
    if (capacity == 0 || capacity > SIZE_MAX / 4) {
        return NULL;
    }
    LruCache *c = malloc(sizeof(LruCache));
    if (!c) return NULL;

    c->capacity = capacity;
    c->size = 0;
    c->head.prev = &c->head;
    c->head.next = &c->head;

    size_t bcount = 16;
    while (bcount < capacity * 2) {
        bcount *= 2;
    }
    c->buckets = calloc(bcount, sizeof(LruNode *));
    if (!c->buckets) {
        free(c);
        return NULL;
    }
    c->bucket_count = bcount;
    return c;
}

void lru_destroy(LruCache *cache) {
    if (!cache) return;
    LruNode *cur = cache->head.next;
    while (cur != &cache->head) {
        LruNode *next = cur->next;
        free(cur);
        cur = next;
    }
    free(cache->buckets);
    free(cache);
}

bool lru_get(LruCache *cache, int64_t key, int64_t *out_value) {
    if (!cache || !out_value) return false;
    LruNode *node = hash_find(cache, key);
    if (!node) return false;

    // 提升至最近使用
    list_detach(node);
    list_push_front(cache, node);
    *out_value = node->value;
    return true;
}

bool lru_put(
    LruCache *cache,
    int64_t key,
    int64_t value,
    bool *out_evicted,
    int64_t *out_evicted_key
) {
    if (!cache) return false;
    LruNode *node = hash_find(cache, key);
    if (node) {
        node->value = value;
        list_detach(node);
        list_push_front(cache, node);
        if (out_evicted) *out_evicted = false;
        return true;
    }

    if (cache->size == cache->capacity) {
        // 淘汰最久未使用的尾结点
        LruNode *victim = cache->head.prev;
        if (out_evicted) *out_evicted = true;
        if (out_evicted_key) *out_evicted_key = victim->key;

        hash_remove(cache, victim->key);
        list_detach(victim);

        // 原地复用结点内存
        victim->key = key;
        victim->value = value;
        hash_insert(cache, victim);
        list_push_front(cache, victim);
        return true;
    }

    LruNode *new_node = malloc(sizeof(LruNode));
    if (!new_node) return false;
    new_node->key = key;
    new_node->value = value;

    hash_insert(cache, new_node);
    list_push_front(cache, new_node);
    cache->size++;
    if (out_evicted) *out_evicted = false;
    return true;
}

size_t lru_size(const LruCache *cache) {
    return cache ? cache->size : 0;
}

int main(void) {
    LruCache *cache = lru_create(2);
    assert(cache != NULL);

    bool evicted = false;
    int64_t evicted_k = 0;
    int64_t val = 0;

    // 插入 1 -> 10, 2 -> 20
    assert(lru_put(cache, 1, 10, &evicted, &evicted_k) && !evicted);
    assert(lru_put(cache, 2, 20, &evicted, &evicted_k) && !evicted);
    assert(lru_size(cache) == 2);

    // 查询 1，使 1 变为最新，2 成为最久未使用
    assert(lru_get(cache, 1, &val) && val == 10);

    // 插入 3 -> 30，应淘汰 2
    assert(lru_put(cache, 3, 30, &evicted, &evicted_k) && evicted);
    assert(evicted_k == 2);

    // 2 应当不存在，1 和 3 存在
    assert(!lru_get(cache, 2, &val));
    assert(lru_get(cache, 1, &val) && val == 10);
    assert(lru_get(cache, 3, &val) && val == 30);

    lru_destroy(cache);
    return 0;
}
```

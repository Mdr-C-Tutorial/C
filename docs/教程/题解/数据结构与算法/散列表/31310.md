# 题解：31310

## 题目

实现采用**素数表容量**与**双重散列 (Double Hashing)** 的 64 位整数键值映射 `DoubleHashMap`：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct DoubleHashMap DoubleHashMap;

DoubleHashMap *dhash_map_create(void);
void dhash_map_destroy(DoubleHashMap *map);
bool dhash_map_put(
    DoubleHashMap *map,
    int64_t key,
    int64_t value,
    bool *out_replaced
);
bool dhash_map_get(
    const DoubleHashMap *map,
    int64_t key,
    int64_t *out_value
);
bool dhash_map_remove(
    DoubleHashMap *map,
    int64_t key,
    bool *out_removed
);
size_t dhash_map_size(const DoubleHashMap *map);
size_t dhash_map_capacity(const DoubleHashMap *map);
```

要求：

1. 槽容量 $m$ 从递增素数表（例如 $11, 23, 47, 97, 197, 397, 797, 1597, \dots$）中选取；设 `uint64_t h = splitmix64_mix((uint64_t)key)`，主散列位置为 $h_1(k) = h \bmod m$，探测步长为 $h_2(k) = 1 + ((h >> 32) \bmod (m - 1))$，第 $i$ 次探测槽下标为 $(h_1(k) + i \cdot h_2(k)) \bmod m$。
2. 每个槽维护 `EMPTY`、`OCCUPIED`、`TOMBSTONE` 三种状态；插入时可复用探测链上遇到的首个墓碑，但必须继续探测直至确认目标键不存在。
3. 当 `(size + tombstones + 1) * 10 > capacity * 7`（装载上限 $0.7$）时，若 `size * 2 <= capacity` 可保持原素数容量原地重建以清理墓碑，否则升级到素数表中的下一个素数容量重散列。
4. 分配失败或参数无效时返回 `false`，原映射表与输出对象保持不变。

## 解析

双重散列结合素数容量能从根本上消除线性探测的一次聚集与二次探测的二次聚集：
1. **步长互素性质**：因为容量 $m$ 为素数，而步长公式为 $h_2(k) = 1 + ((h \gg 32) \bmod (m - 1))$，显然 $1 \le h_2(k) \le m - 1$。因为 $m$ 为质数，任何小于 $m$ 的正整数均与 $m$ 互素，即 $\gcd(h_2(k), m) = 1$，从而保证了探测序列必然遍历全部 $m$ 个槽。
2. **墓碑复用与强异常安全重建**：
   在插入时记录遇到的首个墓碑下标 `first_tomb`，但绝不可提前终止探测，必须继续向后比对确认是否有已存在的相同键；遇到空槽后，若此前遇见过墓碑，则直接将新键值写入该墓碑槽，同时将墓碑计数器递减。
   重散列时先在独立的缓冲区中完成新素数容量槽的分配与有效键搬移，完全成功后再提交替换旧槽指针，确保分配失败时不破坏原映射表状态。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct DoubleHashMap DoubleHashMap;

typedef enum {
    DH_SLOT_EMPTY = 0,
    DH_SLOT_OCCUPIED = 1,
    DH_SLOT_TOMBSTONE = 2
} DhSlotState;

typedef struct {
    int64_t key;
    int64_t value;
    DhSlotState state;
} DhSlot;

struct DoubleHashMap {
    DhSlot *slots;
    size_t capacity;
    size_t size;
    size_t tombstones;
    size_t prime_index;
};

static const size_t PRIMES[] = {
    11, 23, 47, 97, 197, 397, 797, 1597, 3203, 6421,
    12853, 25717, 51437, 102877, 205759, 411527, 823117
};
static const size_t PRIMES_COUNT = sizeof(PRIMES) / sizeof(PRIMES[0]);

static uint64_t splitmix64(uint64_t x) {
    x = (x ^ (x >> 30)) * UINT64_C(0xbf58476d1ce4e5b9);
    x = (x ^ (x >> 27)) * UINT64_C(0x94d049bb133111eb);
    return x ^ (x >> 31);
}

static void compute_hash(int64_t key, size_t cap, size_t *out_h1, size_t *out_h2) {
    uint64_t h = splitmix64((uint64_t)key);
    *out_h1 = (size_t)(h % cap);
    *out_h2 = (size_t)(1 + ((h >> 32) % (cap - 1)));
}

DoubleHashMap *dhash_map_create(void) {
    DoubleHashMap *map = malloc(sizeof(DoubleHashMap));
    if (!map) return NULL;
    map->prime_index = 0;
    map->capacity = PRIMES[0];
    map->size = 0;
    map->tombstones = 0;
    map->slots = calloc(map->capacity, sizeof(DhSlot));
    if (!map->slots) {
        free(map);
        return NULL;
    }
    return map;
}

void dhash_map_destroy(DoubleHashMap *map) {
    if (!map) return;
    free(map->slots);
    free(map);
}

static bool dhash_rehash(DoubleHashMap *map, size_t new_prime_idx) {
    size_t new_cap = PRIMES[new_prime_idx];
    DhSlot *new_slots = calloc(new_cap, sizeof(DhSlot));
    if (!new_slots) return false;

    for (size_t i = 0; i < map->capacity; ++i) {
        if (map->slots[i].state == DH_SLOT_OCCUPIED) {
            int64_t k = map->slots[i].key;
            int64_t v = map->slots[i].value;
            size_t h1 = 0, h2 = 0;
            compute_hash(k, new_cap, &h1, &h2);
            size_t idx = h1;
            while (new_slots[idx].state == DH_SLOT_OCCUPIED) {
                idx = (idx + h2) % new_cap;
            }
            new_slots[idx].key = k;
            new_slots[idx].value = v;
            new_slots[idx].state = DH_SLOT_OCCUPIED;
        }
    }

    free(map->slots);
    map->slots = new_slots;
    map->capacity = new_cap;
    map->prime_index = new_prime_idx;
    map->tombstones = 0;
    return true;
}

bool dhash_map_put(
    DoubleHashMap *map,
    int64_t key,
    int64_t value,
    bool *out_replaced
) {
    if (!map || !out_replaced) return false;

    if ((map->size + map->tombstones + 1) * 10 > map->capacity * 7) {
        size_t next_idx = map->prime_index;
        if (map->size * 2 > map->capacity && next_idx + 1 < PRIMES_COUNT) {
            next_idx++;
        }
        if (!dhash_rehash(map, next_idx)) {
            return false;
        }
    }

    size_t h1 = 0, h2 = 0;
    compute_hash(key, map->capacity, &h1, &h2);
    size_t idx = h1;
    size_t first_tomb = SIZE_MAX;

    for (size_t step = 0; step < map->capacity; ++step) {
        DhSlotState st = map->slots[idx].state;
        if (st == DH_SLOT_EMPTY) {
            size_t target = (first_tomb != SIZE_MAX) ? first_tomb : idx;
            if (first_tomb != SIZE_MAX) {
                map->tombstones--;
            }
            map->slots[target].key = key;
            map->slots[target].value = value;
            map->slots[target].state = DH_SLOT_OCCUPIED;
            map->size++;
            *out_replaced = false;
            return true;
        }
        if (st == DH_SLOT_OCCUPIED && map->slots[idx].key == key) {
            map->slots[idx].value = value;
            *out_replaced = true;
            return true;
        }
        if (st == DH_SLOT_TOMBSTONE && first_tomb == SIZE_MAX) {
            first_tomb = idx;
        }
        idx = (idx + h2) % map->capacity;
    }

    if (first_tomb != SIZE_MAX) {
        map->tombstones--;
        map->slots[first_tomb].key = key;
        map->slots[first_tomb].value = value;
        map->slots[first_tomb].state = DH_SLOT_OCCUPIED;
        map->size++;
        *out_replaced = false;
        return true;
    }
    return false;
}

bool dhash_map_get(
    const DoubleHashMap *map,
    int64_t key,
    int64_t *out_value
) {
    if (!map || !out_value) return false;
    size_t h1 = 0, h2 = 0;
    compute_hash(key, map->capacity, &h1, &h2);
    size_t idx = h1;

    for (size_t step = 0; step < map->capacity; ++step) {
        DhSlotState st = map->slots[idx].state;
        if (st == DH_SLOT_EMPTY) return false;
        if (st == DH_SLOT_OCCUPIED && map->slots[idx].key == key) {
            *out_value = map->slots[idx].value;
            return true;
        }
        idx = (idx + h2) % map->capacity;
    }
    return false;
}

bool dhash_map_remove(
    DoubleHashMap *map,
    int64_t key,
    bool *out_removed
) {
    if (!map || !out_removed) return false;
    size_t h1 = 0, h2 = 0;
    compute_hash(key, map->capacity, &h1, &h2);
    size_t idx = h1;

    for (size_t step = 0; step < map->capacity; ++step) {
        DhSlotState st = map->slots[idx].state;
        if (st == DH_SLOT_EMPTY) {
            *out_removed = false;
            return true;
        }
        if (st == DH_SLOT_OCCUPIED && map->slots[idx].key == key) {
            map->slots[idx].state = DH_SLOT_TOMBSTONE;
            map->size--;
            map->tombstones++;
            *out_removed = true;
            return true;
        }
        idx = (idx + h2) % map->capacity;
    }
    *out_removed = false;
    return true;
}

size_t dhash_map_size(const DoubleHashMap *map) {
    return map ? map->size : 0;
}

size_t dhash_map_capacity(const DoubleHashMap *map) {
    return map ? map->capacity : 0;
}

int main(void) {
    DoubleHashMap *map = dhash_map_create();
    assert(map != NULL);
    assert(dhash_map_capacity(map) == 11);

    bool replaced = false;
    bool removed = false;
    int64_t val = 0;

    for (int64_t i = 1; i <= 20; ++i) {
        assert(dhash_map_put(map, i * 10, i * 100, &replaced));
        assert(!replaced);
    }
    assert(dhash_map_size(map) == 20);
    assert(dhash_map_capacity(map) > 11); // 已触发素数扩容

    for (int64_t i = 1; i <= 20; ++i) {
        assert(dhash_map_get(map, i * 10, &val) && val == i * 100);
    }

    assert(dhash_map_remove(map, 50, &removed) && removed);
    assert(!dhash_map_get(map, 50, &val));
    assert(dhash_map_remove(map, 50, &removed) && !removed);

    dhash_map_destroy(map);
    return 0;
}
```

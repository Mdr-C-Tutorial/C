# 31802 - 动态邻接表表示法与图遍历

## 题目描述

实现基于链表数组的动态邻接表图结构 `IntGraph`，支持有向图与无向图的高效追加构建、按加边顺序的广度优先搜索（BFS）、出入度统计以及转置图构造：

```c
#include <stdbool.h>
#include <stddef.h>

typedef struct IntGraph IntGraph;

IntGraph *int_graph_create(size_t vertex_count, bool directed);
void int_graph_destroy(IntGraph *graph);
bool int_graph_add_edge(IntGraph *graph, size_t from, size_t to);
bool int_graph_bfs(
    const IntGraph *graph,
    size_t source,
    size_t *out_order,
    size_t capacity,
    size_t *out_count
);
bool int_graph_degrees(
    const IntGraph *graph,
    size_t vertex,
    size_t *out_in_degree,
    size_t *out_out_degree
);
IntGraph *int_graph_transpose(const IntGraph *graph);
```

要求：

1. 顶点编号为 $[0, \text{vertex\_count} - 1]$。图对象拥有全部邻接链表节点的独立内存存储。
2. 每个顶点维护头尾双指针 `head` 与 `tail`，确保以严格 $O(1)$ 时间按实际加边次序将新边挂载于链表末尾。
3. `int_graph_bfs` 按加边顺序将从源点 `source` 可达的所有顶点写入 `out_order`。入队时即时打上已访问标记，杜绝重边导致的重复入队；容量不足或参数非法时不得修改输出数组。
4. `int_graph_degrees` 仅对有向图生效，统计指定顶点的出度与入度；`int_graph_transpose` 生成并返回每条边反转后的独立转置图，中途若分配失败必须完整释放已分配结构并返回 `NULL`。
5. 详细阐述动态邻接表在稀疏图场景下的内存优势及对象所有权生命周期管理准则。

## 解题思路

### 1. 稀疏图与邻接链表的设计哲学

在现实世界的网络拓扑（社交网络、网页链接、道路交通图）中，绝大多数图均为**稀疏图（Sparse Graph）**，边数 $E$ 远小于 $V^2$（通常 $E = O(V)$）。
此时邻接矩阵会退化出大量的内存浪费（绝大多数单元为 0）。邻接表通过为每个顶点维护一条单向链表，仅为真实存在的边分配存储空间：
- 节点结构：
  ```c
  struct GraphEdge {
      size_t to;
      GraphEdge *next;
  };
  ```
- 顶点结构包含首尾指针 `head` 与 `tail`，使得追加边无需遍历链表即可在 $O(1)$ 时间完成，同时完美维系了边的输入时间拓扑序。

### 2. 广度优先搜索（BFS）的访问队列与幂等标记

在执行 BFS 遍历时：
- 维护一个环形/线性队列 `queue` 与布尔访问标记数组 `visited`；
- **防重复入队不变式**：顶点的 `visited` 标记必须在**入队的那一瞬间**置为 `true`，而绝不能推迟到出队时。若推迟到出队，当同一个顶点被多个已入队的前驱共同指向时，该顶点会被重复推入队列多次，破坏 $O(V + E)$ 的线性复杂度保证。

### 3. 转置图的有向反转与异常安全回滚

有向图的转置图 $G^T = (V, E^T)$ 将所有有向边反转（$u \to v$ 转变为 $v \to u$）：
- 首先创建拥有相同顶点数的空有向图 `transposed`；
- 遍历原图的每一条边 $u \to v$，向 `transposed` 中插入反向边 $v \to u$；
- 若某次节点分配中途失败，直接调用 `int_graph_destroy(transposed)` 利用成熟的析构链条完整回收此前已成功挂接的半成品节点，保证零内存泄漏。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct IntGraph IntGraph;
typedef struct GraphEdge GraphEdge;

struct GraphEdge {
    size_t to;
    GraphEdge *next;
};

typedef struct {
    GraphEdge *head;
    GraphEdge *tail;
} Adjacency;

struct IntGraph {
    Adjacency *adj;
    size_t vertex_count;
    bool directed;
};

IntGraph *int_graph_create(size_t vertex_count, bool directed) {
    IntGraph *graph = malloc(sizeof *graph);
    if (graph == NULL) {
        return NULL;
    }
    graph->adj = NULL;
    graph->vertex_count = vertex_count;
    graph->directed = directed;
    if (vertex_count > 0) {
        if (vertex_count > SIZE_MAX / sizeof *graph->adj) {
            free(graph);
            return NULL;
        }
        graph->adj = calloc(vertex_count, sizeof *graph->adj);
        if (graph->adj == NULL) {
            free(graph);
            return NULL;
        }
    }
    return graph;
}

static GraphEdge *make_edge(size_t to) {
    GraphEdge *edge = malloc(sizeof *edge);
    if (edge == NULL) {
        return NULL;
    }
    edge->to = to;
    edge->next = NULL;
    return edge;
}

static void append_edge(Adjacency *adj, GraphEdge *edge) {
    if (adj->tail == NULL) {
        adj->head = edge;
    } else {
        adj->tail->next = edge;
    }
    adj->tail = edge;
}

void int_graph_destroy(IntGraph *graph) {
    if (graph == NULL) {
        return;
    }
    for (size_t i = 0; i < graph->vertex_count; ++i) {
        GraphEdge *curr = graph->adj[i].head;
        while (curr != NULL) {
            GraphEdge *next = curr->next;
            free(curr);
            curr = next;
        }
    }
    free(graph->adj);
    free(graph);
}

bool int_graph_add_edge(IntGraph *graph, size_t from, size_t to) {
    if (graph == NULL || from >= graph->vertex_count || to >= graph->vertex_count) {
        return false;
    }
    GraphEdge *first = make_edge(to);
    if (first == NULL) {
        return false;
    }
    GraphEdge *second = NULL;
    if (!graph->directed) {
        second = make_edge(from);
        if (second == NULL) {
            free(first);
            return false;
        }
    }
    append_edge(&graph->adj[from], first);
    if (second != NULL) {
        append_edge(&graph->adj[to], second);
    }
    return true;
}

bool int_graph_bfs(
    const IntGraph *graph,
    size_t source,
    size_t *out_order,
    size_t capacity,
    size_t *out_count
) {
    if (graph == NULL || out_count == NULL || source >= graph->vertex_count) {
        return false;
    }
    if (graph->vertex_count == 0) {
        *out_count = 0;
        return true;
    }
    bool *visited = calloc(graph->vertex_count, sizeof *visited);
    size_t *queue = malloc(graph->vertex_count * sizeof *queue);
    if (visited == NULL || queue == NULL) {
        free(visited);
        free(queue);
        return false;
    }

    size_t head = 0;
    size_t tail = 0;
    queue[tail++] = source;
    visited[source] = true;

    while (head < tail) {
        size_t u = queue[head++];
        for (const GraphEdge *e = graph->adj[u].head; e != NULL; e = e->next) {
            if (!visited[e->to]) {
                visited[e->to] = true;
                queue[tail++] = e->to;
            }
        }
    }

    if (tail > capacity || (tail > 0 && out_order == NULL)) {
        free(visited);
        free(queue);
        return false;
    }

    memcpy(out_order, queue, tail * sizeof *out_order);
    *out_count = tail;
    free(visited);
    free(queue);
    return true;
}

bool int_graph_degrees(
    const IntGraph *graph,
    size_t vertex,
    size_t *out_in_degree,
    size_t *out_out_degree
) {
    if (graph == NULL || !graph->directed || out_in_degree == NULL ||
        out_out_degree == NULL || vertex >= graph->vertex_count) {
        return false;
    }
    size_t in_deg = 0;
    size_t out_deg = 0;
    for (const GraphEdge *e = graph->adj[vertex].head; e != NULL; e = e->next) {
        out_deg++;
    }
    for (size_t u = 0; u < graph->vertex_count; ++u) {
        for (const GraphEdge *e = graph->adj[u].head; e != NULL; e = e->next) {
            if (e->to == vertex) {
                in_deg++;
            }
        }
    }
    *out_in_degree = in_deg;
    *out_out_degree = out_deg;
    return true;
}

IntGraph *int_graph_transpose(const IntGraph *graph) {
    if (graph == NULL || !graph->directed) {
        return NULL;
    }
    IntGraph *t = int_graph_create(graph->vertex_count, true);
    if (t == NULL) {
        return NULL;
    }
    for (size_t u = 0; u < graph->vertex_count; ++u) {
        for (const GraphEdge *e = graph->adj[u].head; e != NULL; e = e->next) {
            if (!int_graph_add_edge(t, e->to, u)) {
                int_graph_destroy(t);
                return NULL;
            }
        }
    }
    return t;
}

int main(void) {
    IntGraph *g = int_graph_create(4, true);
    assert(g != NULL);
    assert(int_graph_add_edge(g, 0, 1));
    assert(int_graph_add_edge(g, 0, 2));
    assert(int_graph_add_edge(g, 1, 2));
    assert(int_graph_add_edge(g, 2, 3));

    size_t order[4] = {0};
    size_t count = 0;
    assert(int_graph_bfs(g, 0, order, 4, &count));
    assert(count == 4);
    assert(order[0] == 0 && order[1] == 1 && order[2] == 2 && order[3] == 3);

    size_t in_deg = 0;
    size_t out_deg = 0;
    assert(int_graph_degrees(g, 2, &in_deg, &out_deg));
    assert(in_deg == 2 && out_deg == 1);

    IntGraph *gt = int_graph_transpose(g);
    assert(gt != NULL);
    assert(int_graph_degrees(gt, 2, &in_deg, &out_deg));
    assert(in_deg == 1 && out_deg == 2);

    int_graph_destroy(gt);
    int_graph_destroy(g);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：
  - `int_graph_add_edge`：维护了链表尾指针，无论是有向边还是无向边，加边时间严格为 $O(1)$。
  - `int_graph_bfs`：每个连通顶点恰入队一次、出队一次，每条出边被检查一次，时间复杂度为标准的 $O(V + E)$。
  - `int_graph_degrees`：出度统计耗时 $O(\text{out\_deg}(v))$，入度统计需遍历所有顶点的出边，耗时 $O(V + E)$。
  - `int_graph_transpose`：遍历全图每条边并反向挂载，时间复杂度为 $O(V + E)$。
- **空间复杂度**：为 $V$ 个顶点分配表头，为 $E$ 条边分配链表节点，空间复杂度严格为 $O(V + E)$。

## 延伸思考 / 模式总结

1. **单向邻接表的入度痛点**：标准邻接表仅沿出边方向组织数据，这导致查询某个单点的入度需要“反向扫全图”，耗时高达 $O(V + E)$。若业务场景需要高频双向查询入度与出度，应转向后续章节的**十字链表（Orthogonal List）**或在顶点上同时维护正反两张邻接表。
2. **指针跳跃与缓存代价**：虽然邻接表在渐近空间复杂度 $O(V + E)$ 上极其理想，但链表节点在物理堆内存中离散分布。在图算法的高频遍历中，指针反引用（Pointer Chasing）会引发大量 CPU 缓存未命中，这也催生了完全基于连续内存数组的**链式前向星**设计。

# 31406 - 二叉搜索树删除与前驱后继替换

## 题目描述

在二叉搜索树（BST）中实现结点的动态删除操作 `int_bst_remove`，正确处理无子结点、单子结点以及双子结点的三种结构变形，并采用中序后继（或前驱）结点替换策略。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct IntBst IntBst;

bool int_bst_remove(
    IntBst *tree,
    int64_t key,
    bool *out_removed
);
```

要求：

1. 当待删结点为叶结点（无子结点）时，直接释放该结点，并将其父结点对应指针置为 NULL。
2. 当待删结点仅有单侧孩子时，将其子结点直接提升并链接至父结点，绕过被删结点。
3. 当待删结点同时拥有左右两棵子树时，在右子树中找到中序后继（右子树最小值结点），以后继结点的键替换待删结点的键，再递归在右子树中删除该后继结点。
4. 正确维护树中结点总数 `tree->size`；若目标键不存在，设置 `*out_removed = false` 并返回 `true`，树保持不变。
5. 删除后树必须依然严格满足二叉搜索树不变式，中序遍历结果严格单调递增。

## 解题思路

### 1. 三种删除情形的状态转移

- **情形 1：叶结点**：直接 `free(node)`，返回 `NULL` 供上层链接。
- **情形 2：单孩子结点**：
  - 若 `node->left == NULL`，暂存 `node->right`，释放 `node` 并返回暂存的右孩子。
  - 若 `node->right == NULL`，暂存 `node->left`，释放 `node` 并返回暂存的左孩子。
- **情形 3：双孩子结点**：
  - 待删结点拥有两个孩子，不能直接改变父子指针拓扑。
  - 观察发现：当前结点在中序遍历中的直接后继（右子树的最左结点）必然**没有左孩子**！
  - 因此，我们将后继结点的键值复制到当前结点，然后将删除任务转化为“在右子树中删除该后继键”。由于后继结点最多只有一个右孩子，该次递归删除必然落在情形 1 或情形 2，保证算法终止且结构简单。

### 2. 递归返回指针更新技术

采用函数式递归更新指针模式：`node->left = remove_node_rec(node->left, key, ...)`。当删除发生在子树中时，递归调用直接返回子树重构后的新根结点指针，父结点直接接收更新，代码结构紧凑且完全杜绝了悬空指针。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct IntBst IntBst;
typedef struct BstNode BstNode;

struct BstNode {
    int64_t key;
    BstNode *left;
    BstNode *right;
};

struct IntBst {
    BstNode *root;
    size_t size;
};

IntBst *int_bst_create(void) {
    IntBst *tree = (IntBst *)malloc(sizeof *tree);
    if (tree == NULL) {
        return NULL;
    }
    tree->root = NULL;
    tree->size = 0;
    return tree;
}

static void destroy_nodes(BstNode *node) {
    if (node == NULL) {
        return;
    }
    destroy_nodes(node->left);
    destroy_nodes(node->right);
    free(node);
}

void int_bst_destroy(IntBst *tree) {
    if (tree == NULL) {
        return;
    }
    destroy_nodes(tree->root);
    free(tree);
}

bool int_bst_insert(IntBst *tree, int64_t key, bool *out_inserted) {
    if (tree == NULL || out_inserted == NULL) {
        return false;
    }
    BstNode **link = &tree->root;
    while (*link != NULL) {
        if (key == (*link)->key) {
            *out_inserted = false;
            return true;
        }
        link = key < (*link)->key ? &(*link)->left : &(*link)->right;
    }
    BstNode *node = (BstNode *)malloc(sizeof *node);
    if (node == NULL) {
        return false;
    }
    node->key = key;
    node->left = NULL;
    node->right = NULL;
    *link = node;
    ++tree->size;
    *out_inserted = true;
    return true;
}

static BstNode *remove_node_rec(BstNode *root, int64_t key, bool *removed) {
    if (root == NULL) {
        *removed = false;
        return NULL;
    }
    if (key < root->key) {
        root->left = remove_node_rec(root->left, key, removed);
    } else if (key > root->key) {
        root->right = remove_node_rec(root->right, key, removed);
    } else {
        *removed = true;
        /* 情形 1 与情形 2：0 个或 1 个孩子 */
        if (root->left == NULL) {
            BstNode *right_child = root->right;
            free(root);
            return right_child;
        }
        if (root->right == NULL) {
            BstNode *left_child = root->left;
            free(root);
            return left_child;
        }
        /* 情形 3：拥有两个孩子 -> 寻找右子树中的中序后继结点 */
        BstNode *succ = root->right;
        while (succ->left != NULL) {
            succ = succ->left;
        }
        root->key = succ->key;
        bool succ_removed = false;
        root->right = remove_node_rec(root->right, succ->key, &succ_removed);
    }
    return root;
}

bool int_bst_remove(IntBst *tree, int64_t key, bool *out_removed) {
    if (tree == NULL || out_removed == NULL) {
        return false;
    }
    bool removed = false;
    tree->root = remove_node_rec(tree->root, key, &removed);
    if (removed) {
        tree->size--;
    }
    *out_removed = removed;
    return true;
}

bool int_bst_contains(const IntBst *tree, int64_t key) {
    if (tree == NULL) {
        return false;
    }
    const BstNode *node = tree->root;
    while (node != NULL) {
        if (key == node->key) {
            return true;
        }
        node = key < node->key ? node->left : node->right;
    }
    return false;
}

static void write_inorder(const BstNode *node, int64_t *out_keys, size_t *cursor) {
    if (node == NULL) {
        return;
    }
    write_inorder(node->left, out_keys, cursor);
    out_keys[(*cursor)++] = node->key;
    write_inorder(node->right, out_keys, cursor);
}

bool int_bst_inorder(const IntBst *tree, int64_t *out_keys, size_t capacity, size_t *out_count) {
    if (tree == NULL || out_count == NULL || capacity < tree->size) {
        return false;
    }
    size_t cursor = 0;
    write_inorder(tree->root, out_keys, &cursor);
    *out_count = cursor;
    return true;
}

int main(void) {
    IntBst *tree = int_bst_create();
    assert(tree != NULL);

    bool ins = false;
    int64_t initial[] = {50, 30, 70, 20, 40, 60, 80, 65, 68};
    for (size_t i = 0; i < sizeof(initial)/sizeof(initial[0]); ++i) {
        assert(int_bst_insert(tree, initial[i], &ins) && ins);
    }
    assert(tree->size == 9);

    bool rem = false;
    /* 1. 删除叶结点 20 */
    assert(int_bst_remove(tree, 20, &rem) && rem);
    assert(!int_bst_contains(tree, 20));
    assert(tree->size == 8);

    /* 2. 删除单孩子结点: 65 只有右孩子 68 */
    assert(int_bst_remove(tree, 65, &rem) && rem);
    assert(!int_bst_contains(tree, 65));
    assert(int_bst_contains(tree, 68));
    assert(tree->size == 7);

    /* 3. 删除双孩子根结点 50 */
    assert(int_bst_remove(tree, 50, &rem) && rem);
    assert(!int_bst_contains(tree, 50));
    assert(tree->size == 6);

    /* 4. 删除不存在的键 */
    assert(int_bst_remove(tree, 999, &rem) && !rem);
    assert(tree->size == 6);

    /* 验证中序遍历仍然严格有序 */
    int64_t out[16];
    size_t count = 0;
    assert(int_bst_inorder(tree, out, 16, &count));
    assert(count == 6);
    for (size_t i = 1; i < count; ++i) {
        assert(out[i - 1] < out[i]);
    }

    int_bst_destroy(tree);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：删除操作先耗费 $O(h)$ 定位目标结点；若为双孩子情形，寻找后继需耗费至多 $O(h)$，随后在右子树中删除后继同样耗费 $O(h)$。因此单次删除的总时间复杂度为严格的 $O(h)$。
- **空间复杂度**：递归栈深度受限于二叉搜索树的高度 $h$，空间复杂度为 $O(h)$。

## 延伸思考 / 模式总结

在非自平衡的二叉搜索树中，若长期反复采用“固定以右子树后继替换被删结点”的策略，会导致树形态产生向左倾斜的系统性漂移（Hibbard 倾斜现象）。平衡搜索树（如 AVL 树与红黑树）正是在本题删除算法的基础上，追加了回溯平衡旋转与颜色重修机制，从而彻底消除了树高恶化的风险。

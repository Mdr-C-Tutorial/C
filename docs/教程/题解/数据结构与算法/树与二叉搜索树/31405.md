# 31405 - 二叉树层次遍历与最大宽度

## 题目描述

实现二叉树的层次遍历（广度优先搜索，BFS），输出分层信息，并分别计算“单层最多结点数”与“带空隙的最大跨度宽度”。

```c
#include <stdbool.h>
#include <stddef.h>

typedef struct TreeNode TreeNode;

struct TreeNode {
    int val;
    TreeNode *left;
    TreeNode *right;
};

bool tree_level_order(
    const TreeNode *root,
    int *out_keys,
    size_t capacity,
    size_t *out_count,
    size_t *level_sizes,
    size_t max_levels,
    size_t *out_level_count
);

size_t tree_max_width(const TreeNode *root);
size_t tree_max_span_width(const TreeNode *root);
```

要求：

1. `tree_level_order` 采用 FIFO 队列按从上到下、从左到右顺序访问结点，输出一维线性序列以及每一层的实际结点数。
2. `tree_max_width` 返回任意单层包含的非空结点数的最大值。
3. `tree_max_span_width` 返回完全二叉树编号系统下的最大跨度宽度（即某一层中最左端非空结点到最右端非空结点之间的距离，中间即使包含空结点亦计入宽度）。
4. 队列实现需注意内存安全与容量检查；空树输入时返回 0 或空结果并安全退出。

## 解题思路

### 1. 广度优先分层模型

借助 FIFO 队列，在每次循环开始时，队列中结点的个数 `level_sz = tail - head` 恰好等于当前层的总结点数。通过一个内层循环执行 `level_sz` 次出队并将其非空子结点入队，即可清晰界定每一层的物理边界。

### 2. 带空隙跨度宽度的坐标系归一化

若按照完全二叉树编号，根结点下标为 $0$，其左孩子下标为 $2i$，右孩子下标为 $2i + 1$。
在树极度不平衡或较深时，指数级增长的下标可能导致 64 位整数溢出。为防止溢出，我们在处理每一层时，**以该层第一个出队的非空结点的下标作为偏移基准**，对该层所有结点的下标执行局部归一化：`norm_idx = item.index - first_idx`。
此时左孩子下标为 `2 * norm_idx`，右孩子下标为 `2 * norm_idx + 1`，跨度计算为 `last_idx - first_idx + 1`，彻底杜绝了深层路径上的整数溢出隐患。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

typedef struct TreeNode {
    int val;
    struct TreeNode *left;
    struct TreeNode *right;
} TreeNode;

static TreeNode *create_node(int val) {
    TreeNode *node = (TreeNode *)malloc(sizeof(TreeNode));
    assert(node != NULL);
    node->val = val;
    node->left = NULL;
    node->right = NULL;
    return node;
}

static void free_tree(TreeNode *root) {
    if (root == NULL) {
        return;
    }
    free_tree(root->left);
    free_tree(root->right);
    free(root);
}

typedef struct QueueItem {
    const TreeNode *node;
    uint64_t index;
} QueueItem;

typedef struct Queue {
    QueueItem *items;
    size_t head;
    size_t tail;
    size_t capacity;
} Queue;

static Queue *queue_create(size_t capacity) {
    Queue *q = (Queue *)malloc(sizeof(Queue));
    assert(q != NULL);
    q->items = (QueueItem *)malloc(sizeof(QueueItem) * capacity);
    assert(q->items != NULL);
    q->head = 0;
    q->tail = 0;
    q->capacity = capacity;
    return q;
}

static void queue_free(Queue *q) {
    if (q != NULL) {
        free(q->items);
        free(q);
    }
}

static bool queue_is_empty(const Queue *q) {
    return q->head == q->tail;
}

static void queue_push(Queue *q, const TreeNode *node, uint64_t index) {
    assert(q->tail < q->capacity);
    q->items[q->tail].node = node;
    q->items[q->tail].index = index;
    q->tail++;
}

static QueueItem queue_pop(Queue *q) {
    assert(q->head < q->tail);
    return q->items[q->head++];
}

bool tree_level_order(
    const TreeNode *root,
    int *out_keys,
    size_t capacity,
    size_t *out_count,
    size_t *level_sizes,
    size_t max_levels,
    size_t *out_level_count
) {
    if (out_count == NULL || out_level_count == NULL) {
        return false;
    }
    *out_count = 0;
    *out_level_count = 0;
    if (root == NULL) {
        return true;
    }

    Queue *q = queue_create(2048);
    queue_push(q, root, 0);

    size_t total_keys = 0;
    size_t level_idx = 0;

    while (!queue_is_empty(q)) {
        size_t level_sz = q->tail - q->head;
        if (level_idx < max_levels && level_sizes != NULL) {
            level_sizes[level_idx] = level_sz;
        }
        level_idx++;

        for (size_t i = 0; i < level_sz; ++i) {
            QueueItem item = queue_pop(q);
            if (total_keys < capacity && out_keys != NULL) {
                out_keys[total_keys] = item.node->val;
            }
            total_keys++;

            if (item.node->left != NULL) {
                queue_push(q, item.node->left, 0);
            }
            if (item.node->right != NULL) {
                queue_push(q, item.node->right, 0);
            }
        }
    }

    queue_free(q);

    if (total_keys > capacity) {
        return false;
    }

    *out_count = total_keys;
    *out_level_count = level_idx;
    return true;
}

size_t tree_max_width(const TreeNode *root) {
    if (root == NULL) {
        return 0;
    }
    Queue *q = queue_create(2048);
    queue_push(q, root, 0);

    size_t max_w = 0;
    while (!queue_is_empty(q)) {
        size_t level_sz = q->tail - q->head;
        if (level_sz > max_w) {
            max_w = level_sz;
        }
        for (size_t i = 0; i < level_sz; ++i) {
            QueueItem item = queue_pop(q);
            if (item.node->left != NULL) {
                queue_push(q, item.node->left, 0);
            }
            if (item.node->right != NULL) {
                queue_push(q, item.node->right, 0);
            }
        }
    }
    queue_free(q);
    return max_w;
}

size_t tree_max_span_width(const TreeNode *root) {
    if (root == NULL) {
        return 0;
    }
    Queue *q = queue_create(2048);
    queue_push(q, root, 0);

    uint64_t max_span = 0;

    while (!queue_is_empty(q)) {
        size_t level_sz = q->tail - q->head;
        uint64_t first_idx = q->items[q->head].index;
        uint64_t last_idx = q->items[q->tail - 1].index;
        uint64_t span = last_idx - first_idx + 1;
        if (span > max_span) {
            max_span = span;
        }

        for (size_t i = 0; i < level_sz; ++i) {
            QueueItem item = queue_pop(q);
            uint64_t normalized_idx = item.index - first_idx;
            if (item.node->left != NULL) {
                queue_push(q, item.node->left, normalized_idx * 2);
            }
            if (item.node->right != NULL) {
                queue_push(q, item.node->right, normalized_idx * 2 + 1);
            }
        }
    }
    queue_free(q);
    return (size_t)max_span;
}

int main(void) {
    /* 1. 测试空树 */
    assert(tree_max_width(NULL) == 0);
    assert(tree_max_span_width(NULL) == 0);

    /* 2. 测试典型二叉树:
             1
           /   \
          3     2
         / \     \
        5   3     9
    */
    TreeNode *root = create_node(1);
    root->left = create_node(3);
    root->right = create_node(2);
    root->left->left = create_node(5);
    root->left->right = create_node(3);
    root->right->right = create_node(9);

    int keys[16];
    size_t count = 0;
    size_t level_sizes[8];
    size_t level_count = 0;

    bool ok = tree_level_order(root, keys, 16, &count, level_sizes, 8, &level_count);
    assert(ok);
    assert(count == 6);
    assert(level_count == 3);
    assert(level_sizes[0] == 1);
    assert(level_sizes[1] == 2);
    assert(level_sizes[2] == 3);

    int expected[] = {1, 3, 2, 5, 3, 9};
    for (size_t i = 0; i < count; ++i) {
        assert(keys[i] == expected[i]);
    }

    assert(tree_max_width(root) == 3);
    /* 第 3 层：5 (下标 0), 3 (下标 1), 空 (下标 2), 9 (下标 3) -> 跨度宽度为 4 */
    assert(tree_max_span_width(root) == 4);

    free_tree(root);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：每个结点入队出队各一次，总时间复杂度为严格的 $O(n)$。
- **空间复杂度**：队列中存储的最大结点数对应二叉树某一层的最大宽度，在完全二叉树底层时最多达到 $\lceil n / 2 \rceil$，故额外空间复杂度为 $O(w) = O(n)$。

## 延伸思考 / 模式总结

层次遍历是图论广度优先搜索在无环连通分支上的直接映射。通过外层循环锁定当前层大小、内层循环消费当前层所有结点的范式，可以免去在队列项中额外记录深度属性，是一种极其简洁、高效的算法模式。

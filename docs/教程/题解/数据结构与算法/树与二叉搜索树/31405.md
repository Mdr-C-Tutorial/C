# 31405 - 二叉搜索树查找、插入与极值查找

## 题目描述

实现不透明的整数二叉搜索树（BST）抽象数据类型，支持结点的插入、成员查找、极值（最小值与最大值）查找以及中序升序遍历。

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct IntBst IntBst;

typedef enum {
    INT_BST_OK,
    INT_BST_INVALID,
    INT_BST_OUT_OF_MEMORY
} IntBstResult;

IntBst *int_bst_create(void);
void int_bst_destroy(IntBst *tree);
IntBstResult int_bst_insert(
    IntBst *tree,
    int64_t key,
    bool *out_inserted
);
bool int_bst_contains(const IntBst *tree, int64_t key);
bool int_bst_min(const IntBst *tree, int64_t *out_min);
bool int_bst_max(const IntBst *tree, int64_t *out_max);
size_t int_bst_size(const IntBst *tree);
bool int_bst_inorder(
    const IntBst *tree,
    int64_t *out_keys,
    size_t capacity,
    size_t *out_count
);
```

要求：

1. 每个结点的左子树所有键严格小于当前结点键，右子树所有键严格大于当前结点键。
2. 重复键插入时应成功返回并设置 `*out_inserted = false`，树中不增加重复结点。
3. 树拥有内部所有结点对象的生命周期；`int_bst_destroy(NULL)` 必须安全。
4. 插入形参无效、内存分配失败或大小计数溢出时返回相应错误状态，树和 `*out_inserted` 保持原有状态不变。
5. 中序遍历输出数组按严格递增顺序导出键；若容量不足或形参无效，不得修改输出对象。
6. 空树调用 `int_bst_min` 或 `int_bst_max` 返回 `false`，且不修改输出目标指针指向的值。
7. 单次查找、插入与极值查找时间复杂度为 $O(h)$，中序遍历时间复杂度为 $O(n)$。

## 解题思路

### 1. 二级指针与就地挂接

在 `int_bst_insert` 中，使用二级指针 `BstNode **link = &tree->root` 沿着搜索路径不断下探：

- 若 `key == (*link)->key`，说明键已存在，置 `*out_inserted = false`，操作直接完成。
- 若 `key < (*link)->key`，转向 `link = &(*link)->left`。
- 若 `key > (*link)->key`，转向 `link = &(*link)->right`。

当 `*link == NULL` 时，此二级指针正好代表待挂接父结点的左（或右）孩子指针地址！此时直接在此处分配新结点并令 `*link = node`，代码无需记录父结点指针，优雅消除了特判根结点为空的额外分支。

### 2. 极值搜索与中序单调性

- **最小值 (`int_bst_min`)**：从根结点出发，持续沿着 `left` 指针下探直到最左叶子，该结点的键即为整棵树的最小值。
- **最大值 (`int_bst_max`)**：从根结点出发，持续沿着 `right` 指针下探直到最右叶子，该结点的键即为整棵树的最大值。
- **中序遍历导出**：中序递归访问左子树、根结点、右子树。在真正写入前先校验 `capacity >= tree->size`，若缓冲区不足则提前失败，确保强异常安全。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct IntBst IntBst;

typedef enum {
    INT_BST_OK,
    INT_BST_INVALID,
    INT_BST_OUT_OF_MEMORY
} IntBstResult;

typedef struct BstNode BstNode;

struct BstNode {
    int64_t key;
    BstNode *left;
    BstNode *right;
};

struct IntBst {
    BstNode *root;
    size_t size;
};

IntBst *int_bst_create(void) {
    IntBst *tree = (IntBst *)malloc(sizeof *tree);
    if (tree == NULL) {
        return NULL;
    }
    tree->root = NULL;
    tree->size = 0;
    return tree;
}

static void destroy_nodes(BstNode *node) {
    if (node == NULL) {
        return;
    }
    destroy_nodes(node->left);
    destroy_nodes(node->right);
    free(node);
}

void int_bst_destroy(IntBst *tree) {
    if (tree == NULL) {
        return;
    }
    destroy_nodes(tree->root);
    free(tree);
}

IntBstResult int_bst_insert(
    IntBst *tree,
    int64_t key,
    bool *out_inserted
) {
    if (tree == NULL || out_inserted == NULL) {
        return INT_BST_INVALID;
    }

    BstNode **link = &tree->root;
    while (*link != NULL) {
        if (key == (*link)->key) {
            *out_inserted = false;
            return INT_BST_OK;
        }
        link = key < (*link)->key
            ? &(*link)->left
            : &(*link)->right;
    }

    if (tree->size == SIZE_MAX) {
        return INT_BST_INVALID;
    }
    BstNode *node = (BstNode *)malloc(sizeof *node);
    if (node == NULL) {
        return INT_BST_OUT_OF_MEMORY;
    }
    node->key = key;
    node->left = NULL;
    node->right = NULL;
    *link = node;
    ++tree->size;
    *out_inserted = true;
    return INT_BST_OK;
}

bool int_bst_contains(const IntBst *tree, int64_t key) {
    if (tree == NULL) {
        return false;
    }
    const BstNode *node = tree->root;
    while (node != NULL) {
        if (key == node->key) {
            return true;
        }
        node = key < node->key ? node->left : node->right;
    }
    return false;
}

bool int_bst_min(const IntBst *tree, int64_t *out_min) {
    if (tree == NULL || tree->root == NULL || out_min == NULL) {
        return false;
    }
    const BstNode *curr = tree->root;
    while (curr->left != NULL) {
        curr = curr->left;
    }
    *out_min = curr->key;
    return true;
}

bool int_bst_max(const IntBst *tree, int64_t *out_max) {
    if (tree == NULL || tree->root == NULL || out_max == NULL) {
        return false;
    }
    const BstNode *curr = tree->root;
    while (curr->right != NULL) {
        curr = curr->right;
    }
    *out_max = curr->key;
    return true;
}

size_t int_bst_size(const IntBst *tree) {
    return tree != NULL ? tree->size : 0;
}

static void write_inorder(
    const BstNode *node,
    int64_t *out_keys,
    size_t *cursor
) {
    if (node == NULL) {
        return;
    }
    write_inorder(node->left, out_keys, cursor);
    out_keys[(*cursor)++] = node->key;
    write_inorder(node->right, out_keys, cursor);
}

bool int_bst_inorder(
    const IntBst *tree,
    int64_t *out_keys,
    size_t capacity,
    size_t *out_count
) {
    if (tree == NULL || out_count == NULL ||
        capacity < tree->size ||
        (tree->size > 0 && out_keys == NULL)) {
        return false;
    }
    size_t cursor = 0;
    write_inorder(tree->root, out_keys, &cursor);
    *out_count = cursor;
    return true;
}

int main(void) {
    /* 空指针安全销毁 */
    int_bst_destroy(NULL);

    IntBst *tree = int_bst_create();
    assert(tree != NULL);
    assert(int_bst_size(tree) == 0);

    /* 空树极值测试 */
    int64_t min_val = 0, max_val = 0;
    assert(!int_bst_min(tree, &min_val));
    assert(!int_bst_max(tree, &max_val));

    /* 顺序插入多个键 */
    bool inserted = false;
    int64_t keys[] = {50, 30, 70, 20, 40, 60, 80};
    for (size_t i = 0; i < sizeof(keys) / sizeof(keys[0]); ++i) {
        assert(int_bst_insert(tree, keys[i], &inserted) == INT_BST_OK);
        assert(inserted);
    }
    assert(int_bst_size(tree) == 7);

    /* 重复插入测试 */
    assert(int_bst_insert(tree, 50, &inserted) == INT_BST_OK);
    assert(!inserted);
    assert(int_bst_size(tree) == 7);

    /* 成员查找测试 */
    assert(int_bst_contains(tree, 30));
    assert(int_bst_contains(tree, 80));
    assert(!int_bst_contains(tree, 999));

    /* 极值查询测试 */
    assert(int_bst_min(tree, &min_val) && min_val == 20);
    assert(int_bst_max(tree, &max_val) && max_val == 80);

    /* 中序升序遍历验证 */
    int64_t out[16];
    size_t count = 0;
    assert(int_bst_inorder(tree, out, 16, &count));
    assert(count == 7);
    int64_t expected[] = {20, 30, 40, 50, 60, 70, 80};
    for (size_t i = 0; i < 7; ++i) {
        assert(out[i] == expected[i]);
    }

    /* 容量不足防护检查 */
    assert(!int_bst_inorder(tree, out, 5, &count));

    int_bst_destroy(tree);
    return 0;
}
```

## 复杂度分析

- **时间复杂度**：查找、插入、极值查询均只沿一条从根向下的路径行走，时间复杂度为 $O(h)$；在理想平衡时为 $O(\log n)$，在退化为单链的最坏情况下为 $O(n)$。中序遍历访问整棵树的全部结点，时间复杂度为严格的 $\Theta(n)$。
- **空间复杂度**：树中每个结点分配一个 `BstNode` 结构体，存储空间复杂度为 $O(n)$；查找与插入操作均为迭代循环，额外运行空间复杂度为 $O(1)$。

## 延伸思考 / 模式总结

使用二级指针（Double Pointer）操控链式二叉树的递归链接，是 C 语言实现树与链表修改操作的最佳实践之一。它彻底统一了“修改根指针”与“修改子结点指针”的语义逻辑，避免了显式保存 `parent` 指针并在循环结束时额外判断左右方向的冗长代码。

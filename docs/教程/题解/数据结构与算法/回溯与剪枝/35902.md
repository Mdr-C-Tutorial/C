# 35902 - 含有重复元素的全排列去重

## 题目描述

给定一个包含 `count` 个整数的数组 `items`，其中**可能包含重复的数值**。使用回溯法枚举其全部**互不相同**的全排列，每找到一个合法排列，便调用一次函数指针 `visit` 进行处理。

函数签名如下：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef bool (*PermutationVisitor)(
    const int64_t *items,
    size_t count,
    void *context
);

bool enumerate_unique_permutations(
    int64_t *items,
    size_t count,
    PermutationVisitor visit,
    void *context
);
```

要求：
1. 参数无效时返回 `false`。若 `visit` 返回 `false`，则提前中止整个枚举过程并向调用方返回 `false`。
2. 无论搜索正常结束还是提前中止，函数返回前必须将原数组 `items` 完整还原为调用前的初始顺序。
3. 枚举的排列不能出现重复。例如输入 `[1, 1, 2]`，仅生成 3 种排列而不是 6 种。
4. 递归调用栈及额外空间复杂度必须为 $O(n)$。

## 解题思路

1. **预排序确立同质元素邻接性**：
   去重的核心是避免在同一个排列位置上重复尝试相同数值的元素。
   首先对输入数组按升序进行稳定排序，使所有相同的数值在内存中连续排列。

2. **布尔访问标记与同层去重剪枝**：
   维护大小为 $n$ 的布尔状态数组 `used`，`used[i] == true` 表示 `items[i]` 已在当前排列的前缀中被使用。
   在排列树的当前层枚举可用元素 $i$（$0 \le i < count$）：
   - 若 `used[i]` 已为真，跳过；
   - **核心去重条件**：若当前元素与前一个元素相等（`items[i] == items[i - 1]`），且前一个元素尚未被使用（`!used[i - 1]`），则必须**剪枝跳过**！
   
   **原理解析**：当 `!used[i - 1]` 为真时，说明在当前位置的同一层枚举中，上一次循环已经以 `items[i - 1]` 为开头完成了一整套排列搜索并刚刚回溯回来。此时再用相同数值的 `items[i]` 作为开头，必然会导致产生完全对称的冗余子树。因此限定只有当 `items[i - 1]` 在更深层被固定使用时，才允许使用 `items[i]`，从而强制规定相同元素的相对选取顺序。

3. **原数组顺序的保存与复原**：
   为满足“函数返回前必须完整恢复原始顺序”的要求，在排序前可先用额外内存备份原数组，或在算法结束时将原备份数据复制回原数组。

## C 语言实现

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

typedef bool (*PermutationVisitor)(
    const int64_t *items,
    size_t count,
    void *context
);

static int compare_i64(const void *a, const void *b) {
    int64_t arg1 = *(const int64_t *)a;
    int64_t arg2 = *(const int64_t *)b;
    if (arg1 < arg2) return -1;
    if (arg1 > arg2) return 1;
    return 0;
}

static bool permute_unique_dfs(
    const int64_t *sorted_items,
    size_t count,
    size_t depth,
    bool *used,
    int64_t *current_perm,
    PermutationVisitor visit,
    void *context
) {
    if (depth == count) {
        return visit(current_perm, count, context);
    }

    for (size_t i = 0; i < count; ++i) {
        if (used[i]) {
            continue;
        }

        // 同层去重剪枝
        if (i > 0 && sorted_items[i] == sorted_items[i - 1] && !used[i - 1]) {
            continue;
        }

        used[i] = true;
        current_perm[depth] = sorted_items[i];

        bool keep_going = permute_unique_dfs(
            sorted_items, count, depth + 1, used, current_perm, visit, context
        );

        used[i] = false;

        if (!keep_going) {
            return false;
        }
    }

    return true;
}

bool enumerate_unique_permutations(
    int64_t *items,
    size_t count,
    PermutationVisitor visit,
    void *context
) {
    if (visit == NULL || (count > 0 && items == NULL)) {
        return false;
    }
    if (count == 0) {
        return visit(items, 0, context);
    }

    // 备份原数组以备恢复
    int64_t *backup = malloc(count * sizeof *backup);
    int64_t *sorted = malloc(count * sizeof *sorted);
    bool *used = calloc(count, sizeof *used);
    int64_t *current_perm = malloc(count * sizeof *current_perm);

    if (backup == NULL || sorted == NULL || used == NULL || current_perm == NULL) {
        free(backup); free(sorted); free(used); free(current_perm);
        return false;
    }

    memcpy(backup, items, count * sizeof *items);
    memcpy(sorted, items, count * sizeof *items);
    qsort(sorted, count, sizeof(int64_t), compare_i64);

    bool result = permute_unique_dfs(
        sorted, count, 0, used, current_perm, visit, context
    );

    // 严格复原原数组数据
    memcpy(items, backup, count * sizeof *items);

    free(backup);
    free(sorted);
    free(used);
    free(current_perm);
    return result;
}

typedef struct {
    size_t count;
} CollectContext;

static bool count_visitor(const int64_t *items, size_t count, void *context) {
    (void)items;
    (void)count;
    CollectContext *ctx = (CollectContext *)context;
    ctx->count++;
    return true;
}

int main(void) {
    CollectContext ctx;

    // 参数校验
    assert(!enumerate_unique_permutations(NULL, 5, count_visitor, &ctx));
    assert(!enumerate_unique_permutations((int64_t[]){1}, 1, NULL, &ctx));

    // 含有重复元素测试: [1, 1, 2] -> 3 种唯一全排列: [1,1,2], [1,2,1], [2,1,1]
    int64_t a1[3] = {1, 1, 2};
    ctx.count = 0;
    assert(enumerate_unique_permutations(a1, 3, count_visitor, &ctx));
    assert(ctx.count == 3);
    // 确认原数组顺序保持原样
    assert(a1[0] == 1 && a1[1] == 1 && a1[2] == 2);

    // 全相同元素测试: [5, 5, 5, 5] -> 仅 1 种排列
    int64_t a2[4] = {5, 5, 5, 5};
    ctx.count = 0;
    assert(enumerate_unique_permutations(a2, 4, count_visitor, &ctx));
    assert(ctx.count == 1);

    // 无重复元素测试: [3, 1, 2] -> 3! = 6 种排列
    int64_t a3[3] = {3, 1, 2};
    ctx.count = 0;
    assert(enumerate_unique_permutations(a3, 3, count_visitor, &ctx));
    assert(ctx.count == 6);
    assert(a3[0] == 3 && a3[1] == 1 && a3[2] == 2);

    return 0;
}
```

## 复杂度分析

- **时间复杂度**：$O(n!)$。设有 $k$ 种不同元素，重数为 $n_1, n_2, \dots, n_k$，生成的总唯一排列数为多项式系数 $\frac{n!}{n_1! n_2! \dots n_k!}$。算法在每个叶子节点均产出唯一排列，无冗余搜索。
- **空间复杂度**：$O(n)$。包含辅助拷贝数组与递归调用栈，空间与输入规模成线性关系。

## 延伸思考 / 模式总结

带重元素的全排列去重遵循经典的“排序 + 同层去重”范式。通过规定相同元素在树的搜索过程中必须按照原始下标顺序依次选取，斩断了对称镜像分支，是解决一切组合重排列去重题目的通用模板。

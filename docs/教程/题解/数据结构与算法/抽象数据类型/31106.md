# 题解：31104

## 题目

实现支持在 $O(1)$ 时间复杂度内获取当前最小元素的整数最小栈抽象数据类型（MinStack ADT）：

```c
#include <stdbool.h>
#include <stddef.h>

typedef struct MinStack MinStack;

MinStack *min_stack_create(void);
void min_stack_destroy(MinStack *stack);
bool min_stack_push(MinStack *stack, int value);
bool min_stack_pop(MinStack *stack, int *out_value);
bool min_stack_top(const MinStack *stack, int *out_value);
bool min_stack_get_min(const MinStack *stack, int *out_min);
size_t min_stack_size(const MinStack *stack);
```

要求：

1. **不透明抽象封装**：实现细节对调用方完全不可见，调用方无法获知内部是采用辅助同步栈还是单数组复合条目。
2. **常数级查询契约**：`min_stack_get_min` 必须在严格的 $O(1)$ 时间复杂度内返回当前栈内已有元素的最小值，严禁全栈遍历。
3. **强异常安全保证**：`push` 扩容失败时，栈对象的状态必须原样保留，并返回 `false`。
4. **防御性失败约定**：当栈为空或传入参数无效（空指针）时，`pop`、`top`、`get_min` 必须返回 `false`，且严禁修改输出指针所指向的对象。
5. **完整生命周期**：`min_stack_destroy(NULL)` 必须安全；销毁非空栈时必须释放其持有的全部底层存储资源。

## 解析

普通栈若要查询最小值，必须将元素依次弹出或遍历底层数组，耗时为 $O(n)$。最小栈 ADT 的精髓在于“在元素压入的同时以空间换时间，同步记录当前状态的历史极值”。

1. **内部状态设计**：
   - 常见做法有两种：双栈法（一个数据栈配合一个最小值辅助栈）或单数组复合条目法。
   - 采用单数组复合条目法更为紧凑且内存局部性更佳：定义内部结构体条目 `typedef struct { int value; int min; } MinStackEntry;`。
   - 每个条目在存储当前元素值的同时，记录“自栈底到当前栈顶位置的所有元素中的最小值”。

2. **状态推演与不变式**：
   - 当栈为空时压入 $x$，该条目的 `min` 即为 $x$。
   - 当栈非空时压入 $x$，前一栈顶的最小值为 $m_{prev}$，新条目的极值为 $m_{new} = \min(x, m_{prev})$。
   - 弹出栈顶元素时，只需使元素计数器递减；由于历史极值与栈内元素一一绑定，随着栈顶弹出，前一状态的最小值自动重新成为全局当前最小值，无需执行任何重算。
   - 因此，`get_min` 仅需一条读取指令：`stack->items[stack->size - 1].min`，严格满足 $O(1)$ 契约。

3. **动态扩容与强异常安全**：
   - 当 `size == capacity` 时触发几何扩容。先将申请得到的新内存赋予临时指针，校验成功后再替换旧数组指针与容量；若 `realloc` 失败，直接返回 `false`，原有数据与状态完全不变。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct MinStack MinStack;

MinStack *min_stack_create(void);
void min_stack_destroy(MinStack *stack);
bool min_stack_push(MinStack *stack, int value);
bool min_stack_pop(MinStack *stack, int *out_value);
bool min_stack_top(const MinStack *stack, int *out_value);
bool min_stack_get_min(const MinStack *stack, int *out_min);
size_t min_stack_size(const MinStack *stack);

typedef struct {
    int value;
    int min;
} MinStackEntry;

struct MinStack {
    MinStackEntry *items;
    size_t size;
    size_t capacity;
};

MinStack *min_stack_create(void) {
    return calloc(1, sizeof(MinStack));
}

void min_stack_destroy(MinStack *stack) {
    if (stack == NULL) {
        return;
    }
    free(stack->items);
    free(stack);
}

static bool min_stack_grow(MinStack *stack) {
    size_t max_cap = SIZE_MAX / sizeof(MinStackEntry);
    if (stack->capacity >= max_cap) {
        return false;
    }

    size_t next = (stack->capacity == 0) ? 8 : stack->capacity * 2;
    if (next > max_cap || next < stack->capacity) {
        next = max_cap;
    }

    MinStackEntry *new_items = realloc(stack->items, next * sizeof(MinStackEntry));
    if (new_items == NULL) {
        return false;
    }

    stack->items = new_items;
    stack->capacity = next;
    return true;
}

bool min_stack_push(MinStack *stack, int value) {
    if (stack == NULL) {
        return false;
    }

    if (stack->size == stack->capacity) {
        if (!min_stack_grow(stack)) {
            return false;
        }
    }

    int current_min = value;
    if (stack->size > 0) {
        int prev_min = stack->items[stack->size - 1].min;
        if (prev_min < current_min) {
            current_min = prev_min;
        }
    }

    stack->items[stack->size].value = value;
    stack->items[stack->size].min = current_min;
    stack->size++;
    return true;
}

bool min_stack_pop(MinStack *stack, int *out_value) {
    if (stack == NULL || out_value == NULL || stack->size == 0) {
        return false;
    }

    stack->size--;
    *out_value = stack->items[stack->size].value;
    return true;
}

bool min_stack_top(const MinStack *stack, int *out_value) {
    if (stack == NULL || out_value == NULL || stack->size == 0) {
        return false;
    }

    *out_value = stack->items[stack->size - 1].value;
    return true;
}

bool min_stack_get_min(const MinStack *stack, int *out_min) {
    if (stack == NULL || out_min == NULL || stack->size == 0) {
        return false;
    }

    *out_min = stack->items[stack->size - 1].min;
    return true;
}

size_t min_stack_size(const MinStack *stack) {
    return (stack == NULL) ? 0 : stack->size;
}

int main(void) {
    min_stack_destroy(NULL);
    assert(min_stack_size(NULL) == 0);

    MinStack *s = min_stack_create();
    assert(s != NULL);

    int out = 0;
    assert(!min_stack_pop(s, &out));
    assert(!min_stack_top(s, &out));
    assert(!min_stack_get_min(s, &out));

    assert(min_stack_push(s, 5));
    assert(min_stack_get_min(s, &out) && out == 5);

    assert(min_stack_push(s, 3));
    assert(min_stack_get_min(s, &out) && out == 3);

    assert(min_stack_push(s, 7));
    assert(min_stack_get_min(s, &out) && out == 3);

    assert(min_stack_push(s, 2));
    assert(min_stack_get_min(s, &out) && out == 2);

    assert(min_stack_pop(s, &out) && out == 2);
    assert(min_stack_get_min(s, &out) && out == 3);

    assert(min_stack_pop(s, &out) && out == 7);
    assert(min_stack_get_min(s, &out) && out == 3);

    assert(min_stack_pop(s, &out) && out == 3);
    assert(min_stack_get_min(s, &out) && out == 5);

    assert(min_stack_pop(s, &out) && out == 5);
    assert(min_stack_size(s) == 0);
    assert(!min_stack_get_min(s, &out));

    min_stack_destroy(s);
    return 0;
}
```

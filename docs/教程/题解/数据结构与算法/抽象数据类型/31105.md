# 题解：31105

## 题目

为不透明字符串键值字典设计独立的只读游标（游标/迭代器 ADT），实现字典封装与遍历机制的彻底解耦：

```c
#include <stdbool.h>
#include <stddef.h>

typedef struct PropertyBag PropertyBag;
typedef struct PropertyCursor PropertyCursor;

PropertyBag *prop_bag_create(void);
void prop_bag_destroy(PropertyBag *bag);
bool prop_bag_put(PropertyBag *bag, const char *key, const char *value);
const char *prop_bag_get(const PropertyBag *bag, const char *key);
size_t prop_bag_size(const PropertyBag *bag);

PropertyCursor *prop_cursor_create(const PropertyBag *bag);
void prop_cursor_destroy(PropertyCursor *cursor);
bool prop_cursor_valid(const PropertyCursor *cursor);
void prop_cursor_next(PropertyCursor *cursor);
const char *prop_cursor_key(const PropertyCursor *cursor);
const char *prop_cursor_value(const PropertyCursor *cursor);
```

要求：

1. **所有权契约**：`PropertyBag` 拥有存入的所有键和值字符串副本（深拷贝）；已有键再次存入时更新其值。
2. **游标与集合分离**：游标对象 `PropertyCursor` 维护对字典的遍历位置；遍历协议不得暴露内部数组或结点的物理指针。
3. **安全遍历协议**：游标使用 `prop_cursor_valid` 判断是否有效，`prop_cursor_next` 单步步进，`prop_cursor_key` 与 `prop_cursor_value` 返回当前位置只读借用指针。
4. **迭代器失效防御（Fail-Fast）**：字典内部维护修改版本计数器（`version`）；若在游标存续期间字典发生修改，游标必须能检测到版本不一致并立即将 `valid` 判定为 `false`，防御脏遍历。
5. **生命周期隔离**：销毁游标仅回收游标自身占用的资源，绝不影响字典对象及内部数据。

## 解析

直接在容器 ADT 上暴露出底层的物理下标或直接暴露内部结点的 `next` 指针，会严重打破抽象层。现代系统级软件通常采用“游标/迭代器模式”（Iterator Pattern）提供抽象的线性扫描视角。

1. **状态解耦与生命周期控制**：
   - 容器 `PropertyBag` 与遍历器 `PropertyCursor` 属于两个独立的 ADT。
   - `PropertyBag` 负责条目的持久管理、内存深拷贝与扩容。
   - `PropertyCursor` 仅持有容器的只读指针 `const PropertyBag *bag`、当前推进下标 `index`，以及创建时捕获的版本戳 `expected_version`。
   - 游标销毁函数 `prop_cursor_destroy` 仅释放游标自身结构体内存，容器的所有权保持独立。

2. **并发修改与迭代器失效（Modification Versioning）**：
   - 遍历过程中若容器被修改（如插入新键或更新已有键），底层数组可能发生重新分配（`realloc`），导致原有内存地址失效或遍历次序错乱。
   - 为此，字典对象内嵌单调自增的 `uint64_t version`。每次成功执行 `prop_bag_put` 时令 `version++`。
   - 游标在调用 `prop_cursor_valid` 时首先比对 `cursor->bag->version == cursor->expected_version`；一旦发现版本号不一致，立刻判定为无效，从而在运行时截断任何未定义行为。

3. **深拷贝与内存安全**：
   - 字符串复制使用专门的 `duplicate_str` 函数分配 `strlen + 1` 字节。
   - 字符串覆盖更新时，先复制成功新字符串，再释放旧字符串，确保若分配失败原字典状态不被损坏。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

typedef struct PropertyBag PropertyBag;
typedef struct PropertyCursor PropertyCursor;

PropertyBag *prop_bag_create(void);
void prop_bag_destroy(PropertyBag *bag);
bool prop_bag_put(PropertyBag *bag, const char *key, const char *value);
const char *prop_bag_get(const PropertyBag *bag, const char *key);
size_t prop_bag_size(const PropertyBag *bag);

PropertyCursor *prop_cursor_create(const PropertyBag *bag);
void prop_cursor_destroy(PropertyCursor *cursor);
bool prop_cursor_valid(const PropertyCursor *cursor);
void prop_cursor_next(PropertyCursor *cursor);
const char *prop_cursor_key(const PropertyCursor *cursor);
const char *prop_cursor_value(const PropertyCursor *cursor);

typedef struct {
    char *key;
    char *val;
} PropEntry;

struct PropertyBag {
    PropEntry *entries;
    size_t size;
    size_t capacity;
    uint64_t version;
};

struct PropertyCursor {
    const PropertyBag *bag;
    size_t index;
    uint64_t expected_version;
};

static char *duplicate_str(const char *s) {
    if (s == NULL) {
        return NULL;
    }
    size_t len = strlen(s);
    char *copy = malloc(len + 1);
    if (copy == NULL) {
        return NULL;
    }
    memcpy(copy, s, len + 1);
    return copy;
}

PropertyBag *prop_bag_create(void) {
    return calloc(1, sizeof(PropertyBag));
}

void prop_bag_destroy(PropertyBag *bag) {
    if (bag == NULL) {
        return;
    }
    for (size_t i = 0; i < bag->size; ++i) {
        free(bag->entries[i].key);
        free(bag->entries[i].val);
    }
    free(bag->entries);
    free(bag);
}

bool prop_bag_put(PropertyBag *bag, const char *key, const char *value) {
    if (bag == NULL || key == NULL || value == NULL) {
        return false;
    }

    for (size_t i = 0; i < bag->size; ++i) {
        if (strcmp(bag->entries[i].key, key) == 0) {
            char *new_val = duplicate_str(value);
            if (new_val == NULL) {
                return false;
            }
            free(bag->entries[i].val);
            bag->entries[i].val = new_val;
            bag->version++;
            return true;
        }
    }

    if (bag->size == bag->capacity) {
        size_t next = (bag->capacity == 0) ? 4 : bag->capacity * 2;
        if (next > SIZE_MAX / sizeof(PropEntry)) {
            return false;
        }
        PropEntry *new_entries = realloc(bag->entries, next * sizeof(PropEntry));
        if (new_entries == NULL) {
            return false;
        }
        bag->entries = new_entries;
        bag->capacity = next;
    }

    char *copy_k = duplicate_str(key);
    if (copy_k == NULL) {
        return false;
    }
    char *copy_v = duplicate_str(value);
    if (copy_v == NULL) {
        free(copy_k);
        return false;
    }

    bag->entries[bag->size].key = copy_k;
    bag->entries[bag->size].val = copy_v;
    bag->size++;
    bag->version++;
    return true;
}

const char *prop_bag_get(const PropertyBag *bag, const char *key) {
    if (bag == NULL || key == NULL) {
        return NULL;
    }
    for (size_t i = 0; i < bag->size; ++i) {
        if (strcmp(bag->entries[i].key, key) == 0) {
            return bag->entries[i].val;
        }
    }
    return NULL;
}

size_t prop_bag_size(const PropertyBag *bag) {
    return (bag == NULL) ? 0 : bag->size;
}

PropertyCursor *prop_cursor_create(const PropertyBag *bag) {
    if (bag == NULL) {
        return NULL;
    }
    PropertyCursor *cursor = malloc(sizeof(PropertyCursor));
    if (cursor == NULL) {
        return NULL;
    }
    cursor->bag = bag;
    cursor->index = 0;
    cursor->expected_version = bag->version;
    return cursor;
}

void prop_cursor_destroy(PropertyCursor *cursor) {
    free(cursor);
}

bool prop_cursor_valid(const PropertyCursor *cursor) {
    if (cursor == NULL || cursor->bag == NULL) {
        return false;
    }
    if (cursor->bag->version != cursor->expected_version) {
        return false;
    }
    return cursor->index < cursor->bag->size;
}

void prop_cursor_next(PropertyCursor *cursor) {
    if (cursor == NULL || !prop_cursor_valid(cursor)) {
        return;
    }
    cursor->index++;
}

const char *prop_cursor_key(const PropertyCursor *cursor) {
    if (!prop_cursor_valid(cursor)) {
        return NULL;
    }
    return cursor->bag->entries[cursor->index].key;
}

const char *prop_cursor_value(const PropertyCursor *cursor) {
    if (!prop_cursor_valid(cursor)) {
        return NULL;
    }
    return cursor->bag->entries[cursor->index].val;
}

int main(void) {
    PropertyBag *bag = prop_bag_create();
    assert(bag != NULL);

    assert(prop_bag_put(bag, "server", "nginx"));
    assert(prop_bag_put(bag, "port", "8080"));
    assert(prop_bag_put(bag, "env", "prod"));
    assert(prop_bag_size(bag) == 3);

    assert(strcmp(prop_bag_get(bag, "port"), "8080") == 0);
    assert(prop_bag_get(bag, "non_exist") == NULL);

    assert(prop_bag_put(bag, "port", "443"));
    assert(strcmp(prop_bag_get(bag, "port"), "443") == 0);
    assert(prop_bag_size(bag) == 3);

    PropertyCursor *cur = prop_cursor_create(bag);
    assert(cur != NULL);

    size_t count = 0;
    while (prop_cursor_valid(cur)) {
        const char *k = prop_cursor_key(cur);
        const char *v = prop_cursor_value(cur);
        assert(k != NULL && v != NULL);
        count++;
        prop_cursor_next(cur);
    }
    assert(count == 3);
    assert(!prop_cursor_valid(cur));
    prop_cursor_destroy(cur);

    PropertyCursor *cur2 = prop_cursor_create(bag);
    assert(prop_cursor_valid(cur2));
    assert(prop_bag_put(bag, "timeout", "30"));
    assert(!prop_cursor_valid(cur2));
    prop_cursor_destroy(cur2);

    prop_bag_destroy(bag);
    return 0;
}
```

# 题解：31101

## 题目

设计并实现不可变有理数抽象数据类型（Rational ADT），支持值语义的代数运算与前置条件/代数不变式契约约束：

```c
#include <stdbool.h>
#include <stdint.h>

typedef struct {
    int64_t num;
    int64_t den;
} Rational;

bool rational_create(int64_t num, int64_t den, Rational *out);
bool rational_add(Rational a, Rational b, Rational *out);
bool rational_sub(Rational a, Rational b, Rational *out);
bool rational_mul(Rational a, Rational b, Rational *out);
int rational_compare(Rational a, Rational b);
```

要求：

1. **代数不变式**：分母严格大于 0（$den > 0$）；分子与分母互质（$\gcd(|num|, den) == 1$）；数值 0 的唯一规范表示为 $0/1$。
2. **构造契约**：分母为 0 或输出指针为 `NULL` 时返回 `false`，不得修改 `*out`；若分母为负，须将符号转移至分子。
3. **溢出安全性**：加、减、乘运算若发生 64 位有符号整数溢出，必须返回 `false`，不得引发未定义行为，且不得修改输出对象。
4. **全序比较**：`rational_compare` 比较两个有理数的大小：若 $a < b$ 返回负数，若 $a == b$ 返回 0，若 $a > b$ 返回正数；比较过程严防算术溢出。
5. **复杂度**：除辗转相除法开销外，所有运算均为 $O(1)$ 时间复杂度。

## 解析

抽象数据类型（ADT）的核心不仅在于提供一组操作接口，更在于维系其公开的行为契约与内部不变式（Invariants）。对于具备值语义的有理数类型而言，不变式即“任何合法的有理数对象在生命周期内都必须处于分母为正且互质的最简分数形态”。

1. **构造与最简规约**：
   - 欧几里得辗转相除法（Euclidean Algorithm）可在对数时间内求得最大公约数 $\gcd(u, v)$。
   - 针对 64 位极值边界，有符号整数在 $num = -2^{63}$（`INT64_MIN`）时，直接取相反数 `-num` 将导致有符号整数溢出（未定义行为）。必须使用无符号类型 `uint64_t` 进行安全绝对值转换：`safe_abs(x) = (x >= 0) ? (uint64_t)x : (uint64_t)(-(x + 1)) + 1ULL`。
   - 符号统筹规则：最终分子的符号为原分子与分母符号的异或结果，分母强制规约到正半轴。

2. **交叉计算溢出防御**：
   - 有理数加减法公式为 $\frac{n_1}{d_1} \pm \frac{n_2}{d_2} = \frac{n_1 (d_2 / g) \pm n_2 (d_1 / g)}{d_1 (d_2 / g)}$，其中 $g = \gcd(d_1, d_2)$。在通分之前先除以最大公约数，可大幅降低中间乘积溢出的风险。
   - 乘法运算中，利用交叉约分：计算 $g_1 = \gcd(|n_1|, d_2)$ 与 $g_2 = \gcd(|n_2|, d_1)$，先将各项分子分母约简后再执行乘法，最大限度扩展能够精确计算的数值范围。
   - 所有中间乘法与加减法均采用内建溢出检测函数 `__builtin_mul_overflow`、`__builtin_add_overflow` 与 `__builtin_sub_overflow`（或对应的标准边界检测），确保在溢出发生时安全降级返回 `false`。

3. **防溢出大小比较（连分数比较法）**：
   - 比较 $\frac{n_1}{d_1}$ 与 $\frac{n_2}{d_2}$ 若直接使用交叉相乘 $n_1 \cdot d_2$ 与 $n_2 \cdot d_1$，当分母很大时极易发生 64 位整数溢出，导致比较结果颠倒。
   - 正确解法为提取整数商与余数：先比较整数商 $\lfloor n_1 / d_1 \rfloor$ 与 $\lfloor n_2 / d_2 \rfloor$；若整数商相同，再通过连分数展开法则比较小数余数部分 $\frac{r_1}{d_1}$ 与 $\frac{r_2}{d_2}$（倒数后反转符号迭代），全程仅使用除法与取模，彻底杜绝乘法溢出。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stdint.h>

typedef struct {
    int64_t num;
    int64_t den;
} Rational;

static uint64_t safe_abs_i64(int64_t x) {
    if (x >= 0) {
        return (uint64_t)x;
    }
    return (uint64_t)(-(x + 1)) + 1ULL;
}

static uint64_t gcd_u64(uint64_t a, uint64_t b) {
    while (b != 0) {
        uint64_t r = a % b;
        a = b;
        b = r;
    }
    return a;
}

static bool checked_mul_i64(int64_t a, int64_t b, int64_t *out) {
    if (__builtin_mul_overflow(a, b, out)) {
        return false;
    }
    return true;
}

static bool checked_add_i64(int64_t a, int64_t b, int64_t *out) {
    if (__builtin_add_overflow(a, b, out)) {
        return false;
    }
    return true;
}

static bool checked_sub_i64(int64_t a, int64_t b, int64_t *out) {
    if (__builtin_sub_overflow(a, b, out)) {
        return false;
    }
    return true;
}

bool rational_create(int64_t num, int64_t den, Rational *out) {
    if (den == 0 || out == NULL) {
        return false;
    }
    if (den == INT64_MIN && num == INT64_MIN) {
        out->num = 1;
        out->den = 1;
        return true;
    }

    int sign = 1;
    if (den < 0) {
        sign = -sign;
    }
    if (num < 0) {
        sign = -sign;
    }

    uint64_t unum = safe_abs_i64(num);
    uint64_t uden = safe_abs_i64(den);

    if (unum == 0) {
        out->num = 0;
        out->den = 1;
        return true;
    }

    uint64_t g = gcd_u64(unum, uden);
    unum /= g;
    uden /= g;

    if (uden > (uint64_t)INT64_MAX) {
        return false;
    }
    if (sign > 0) {
        if (unum > (uint64_t)INT64_MAX) {
            return false;
        }
        out->num = (int64_t)unum;
    } else {
        if (unum > (uint64_t)INT64_MAX + 1ULL) {
            return false;
        }
        out->num = (unum == (uint64_t)INT64_MAX + 1ULL) ? INT64_MIN : -(int64_t)unum;
    }
    out->den = (int64_t)uden;
    return true;
}

bool rational_add(Rational a, Rational b, Rational *out) {
    if (out == NULL || a.den <= 0 || b.den <= 0) {
        return false;
    }

    uint64_t g = gcd_u64((uint64_t)a.den, (uint64_t)b.den);
    int64_t b_div = (int64_t)((uint64_t)b.den / g);
    int64_t a_div = (int64_t)((uint64_t)a.den / g);

    int64_t t1 = 0;
    int64_t t2 = 0;
    if (!checked_mul_i64(a.num, b_div, &t1) ||
        !checked_mul_i64(b.num, a_div, &t2)) {
        return false;
    }

    int64_t num_sum = 0;
    if (!checked_add_i64(t1, t2, &num_sum)) {
        return false;
    }

    int64_t den_prod = 0;
    if (!checked_mul_i64(a.den, b_div, &den_prod)) {
        return false;
    }

    return rational_create(num_sum, den_prod, out);
}

bool rational_sub(Rational a, Rational b, Rational *out) {
    if (out == NULL || a.den <= 0 || b.den <= 0) {
        return false;
    }

    uint64_t g = gcd_u64((uint64_t)a.den, (uint64_t)b.den);
    int64_t b_div = (int64_t)((uint64_t)b.den / g);
    int64_t a_div = (int64_t)((uint64_t)a.den / g);

    int64_t t1 = 0;
    int64_t t2 = 0;
    if (!checked_mul_i64(a.num, b_div, &t1) ||
        !checked_mul_i64(b.num, a_div, &t2)) {
        return false;
    }

    int64_t num_diff = 0;
    if (!checked_sub_i64(t1, t2, &num_diff)) {
        return false;
    }

    int64_t den_prod = 0;
    if (!checked_mul_i64(a.den, b_div, &den_prod)) {
        return false;
    }

    return rational_create(num_diff, den_prod, out);
}

bool rational_mul(Rational a, Rational b, Rational *out) {
    if (out == NULL || a.den <= 0 || b.den <= 0) {
        return false;
    }

    uint64_t g1 = gcd_u64(safe_abs_i64(a.num), (uint64_t)b.den);
    uint64_t g2 = gcd_u64(safe_abs_i64(b.num), (uint64_t)a.den);

    int64_t a_num_red = a.num / (int64_t)g1;
    int64_t b_den_red = b.den / (int64_t)g1;
    int64_t b_num_red = b.num / (int64_t)g2;
    int64_t a_den_red = a.den / (int64_t)g2;

    int64_t num_prod = 0;
    int64_t den_prod = 0;
    if (!checked_mul_i64(a_num_red, b_num_red, &num_prod) ||
        !checked_mul_i64(a_den_red, b_den_red, &den_prod)) {
        return false;
    }

    return rational_create(num_prod, den_prod, out);
}

int rational_compare(Rational a, Rational b) {
    if (a.num == b.num && a.den == b.den) {
        return 0;
    }

    int64_t q_a = a.num / a.den;
    int64_t r_a = a.num % a.den;
    if (r_a < 0) {
        q_a -= 1;
        r_a += a.den;
    }

    int64_t q_b = b.num / b.den;
    int64_t r_b = b.num % b.den;
    if (r_b < 0) {
        q_b -= 1;
        r_b += b.den;
    }

    if (q_a != q_b) {
        return q_a < q_b ? -1 : 1;
    }

    Rational rem_a = {r_a, a.den};
    Rational rem_b = {r_b, b.den};
    while (rem_a.num != 0 && rem_b.num != 0) {
        int64_t div_a = rem_a.den / rem_a.num;
        int64_t next_r_a = rem_a.den % rem_a.num;

        int64_t div_b = rem_b.den / rem_b.num;
        int64_t next_r_b = rem_b.den % rem_b.num;

        if (div_a != div_b) {
            return div_a > div_b ? -1 : 1;
        }

        rem_a.den = rem_a.num;
        rem_a.num = next_r_a;
        rem_b.den = rem_b.num;
        rem_b.num = next_r_b;
    }

    if (rem_a.num == 0 && rem_b.num == 0) {
        return 0;
    }
    return rem_a.num == 0 ? -1 : 1;
}

int main(void) {
    Rational r1;
    assert(rational_create(2, 4, &r1));
    assert(r1.num == 1 && r1.den == 2);

    Rational r2;
    assert(rational_create(-6, -9, &r2));
    assert(r2.num == 2 && r2.den == 3);

    Rational r3;
    assert(rational_create(4, -8, &r3));
    assert(r3.num == -1 && r3.den == 2);

    Rational r_zero;
    assert(rational_create(0, 5, &r_zero));
    assert(r_zero.num == 0 && r_zero.den == 1);

    Rational invalid;
    assert(!rational_create(1, 0, &invalid));

    Rational sum;
    assert(rational_add(r1, r2, &sum));
    assert(sum.num == 7 && sum.den == 6);

    Rational diff;
    assert(rational_sub(r1, r2, &diff));
    assert(diff.num == -1 && diff.den == 6);

    Rational prod;
    assert(rational_mul(r1, r2, &prod));
    assert(prod.num == 1 && prod.den == 3);

    assert(rational_compare(r1, r2) < 0);
    assert(rational_compare(r2, r1) > 0);
    assert(rational_compare(r1, r1) == 0);

    Rational r_neg = {-1, 3};
    Rational r_pos = {1, 3};
    assert(rational_compare(r_neg, r_pos) < 0);

    return 0;
}
```

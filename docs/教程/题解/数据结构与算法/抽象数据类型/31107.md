# 题解：31107

## 题目

为同一个整数序列 ADT 编写动态数组和双向链表两种可替换实现。公开头文件固定为：

```c
#include <stddef.h>

typedef struct IntSequence IntSequence;

typedef enum {
    INT_SEQUENCE_OK,
    INT_SEQUENCE_INVALID,
    INT_SEQUENCE_RANGE,
    INT_SEQUENCE_OUT_OF_MEMORY
} IntSequenceResult;

IntSequence *int_sequence_create(void);
void int_sequence_destroy(IntSequence *sequence);
IntSequenceResult int_sequence_insert(
    IntSequence *sequence,
    size_t index,
    int value
);
IntSequenceResult int_sequence_erase(
    IntSequence *sequence,
    size_t index,
    int *out_value
);
IntSequenceResult int_sequence_get(
    const IntSequence *sequence,
    size_t index,
    int *out_value
);
size_t int_sequence_size(const IntSequence *sequence);
```

要求：

1. 分别创建 `int_sequence_array.c` 和 `int_sequence_linked.c`；每次构建只链接其中一个实现文件，同一份调用方代码无需修改。
2. 两个实现文件都不得在公开头文件中暴露表示，也不得增加调用方必须使用的公开函数。
3. `insert` 允许 `index == size`；`erase` 和 `get` 只允许 `index < size`。
4. 参数无效时返回 `INT_SEQUENCE_INVALID`，下标越界时返回 `INT_SEQUENCE_RANGE`；失败时不得修改序列和 `*out_value`。
5. 动态数组实现在扩容失败或容量计算溢出时返回 `INT_SEQUENCE_OUT_OF_MEMORY`，并保持原序列不变；尾部插入的摊还时间复杂度为 $O(1)$。
6. 双向链表实现同时保存首尾指针，从距离目标下标较近的一端开始查找；首尾插入和删除的时间复杂度为 $O(1)$。
7. `int_sequence_destroy(NULL)` 应当安全，`int_sequence_size(NULL)` 返回 0。
8. 两种实现都必须释放自身取得的全部资源。

## 解析

抽象数据类型最强大的特性之一是“可替换性”（Interchangeability）。只要公开接口的函数签名、前置条件、后置条件和状态码契约保持完全一致，调用方模块就只依赖契约本身，而在链接阶段可以自由切换不同的后端实现（如从紧凑连续内存切换为离散无碎片链表）。

1. **动态数组后端（`array`）的设计与复杂度**：
   - 内部由 `int *items`、`size` 和 `capacity` 组成连续存储。
   - 随机读取 `int_sequence_get` 拥有严格 $O(1)$ 的寻址性能。
   - 插入与删除需要通过 `memmove` 进行内存区间搬移，时间复杂度为 $O(n)$。在尾部追加时由于几何扩容机制，摊还复杂度保持为 $O(1)$。

2. **双向链表后端（`linked`）的设计与双端折半加速**：
   - 内部维护带有 `prev` 与 `next` 的双向结点，同时记录 `head`、`tail` 与 `size`。
   - 在根据逻辑下标 `index` 寻找结点时，利用比较 `index < size / 2` 判断目标距离表头更近还是距离表尾更近：
     - 若距离表头更近，从 `head` 出发向后步进 `index` 步；
     - 若距离表尾更近，从 `tail` 出发向前回退 `size - 1 - index` 步。
   - 该优化将链表内部遍历的最坏步数降低为 $N/2$。当 `index == 0` 或 `index == size` 时无需遍历，直接在首尾执行 $O(1)$ 指针连接。

3. **双实现自测与验证**：
   - 题解代码将两种后端组织为自包含实现，展示调用方如何针对同一份契约在相同的测试用例上获得完全一致的行为与结果。

## 答案

```c
#include <assert.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

typedef enum {
    INT_SEQUENCE_OK,
    INT_SEQUENCE_INVALID,
    INT_SEQUENCE_RANGE,
    INT_SEQUENCE_OUT_OF_MEMORY
} IntSequenceResult;

// 动态数组后端实现
typedef struct IntSequenceArray IntSequenceArray;

struct IntSequenceArray {
    int *items;
    size_t size;
    size_t capacity;
};

static IntSequenceArray *array_seq_create(void) {
    IntSequenceArray *seq = malloc(sizeof *seq);
    if (seq == NULL) {
        return NULL;
    }
    seq->items = NULL;
    seq->size = 0;
    seq->capacity = 0;
    return seq;
}

static void array_seq_destroy(IntSequenceArray *seq) {
    if (seq == NULL) {
        return;
    }
    free(seq->items);
    free(seq);
}

static IntSequenceResult array_seq_grow(IntSequenceArray *seq) {
    if (seq->size < seq->capacity) {
        return INT_SEQUENCE_OK;
    }

    size_t next_capacity;
    if (seq->capacity == 0) {
        next_capacity = 8;
    } else {
        if (seq->capacity > SIZE_MAX / 2) {
            return INT_SEQUENCE_OUT_OF_MEMORY;
        }
        next_capacity = seq->capacity * 2;
    }

    if (next_capacity > SIZE_MAX / sizeof *seq->items) {
        return INT_SEQUENCE_OUT_OF_MEMORY;
    }

    int *new_items = realloc(seq->items, next_capacity * sizeof *new_items);
    if (new_items == NULL) {
        return INT_SEQUENCE_OUT_OF_MEMORY;
    }

    seq->items = new_items;
    seq->capacity = next_capacity;
    return INT_SEQUENCE_OK;
}

static IntSequenceResult array_seq_insert(
    IntSequenceArray *seq,
    size_t index,
    int value
) {
    if (seq == NULL) {
        return INT_SEQUENCE_INVALID;
    }
    if (index > seq->size) {
        return INT_SEQUENCE_RANGE;
    }
    if (seq->size == SIZE_MAX) {
        return INT_SEQUENCE_OUT_OF_MEMORY;
    }

    IntSequenceResult res = array_seq_grow(seq);
    if (res != INT_SEQUENCE_OK) {
        return res;
    }

    memmove(
        seq->items + index + 1,
        seq->items + index,
        (seq->size - index) * sizeof *seq->items
    );
    seq->items[index] = value;
    ++seq->size;
    return INT_SEQUENCE_OK;
}

static IntSequenceResult array_seq_erase(
    IntSequenceArray *seq,
    size_t index,
    int *out_value
) {
    if (seq == NULL || out_value == NULL) {
        return INT_SEQUENCE_INVALID;
    }
    if (index >= seq->size) {
        return INT_SEQUENCE_RANGE;
    }

    *out_value = seq->items[index];
    memmove(
        seq->items + index,
        seq->items + index + 1,
        (seq->size - index - 1) * sizeof *seq->items
    );
    --seq->size;
    return INT_SEQUENCE_OK;
}

static IntSequenceResult array_seq_get(
    const IntSequenceArray *seq,
    size_t index,
    int *out_value
) {
    if (seq == NULL || out_value == NULL) {
        return INT_SEQUENCE_INVALID;
    }
    if (index >= seq->size) {
        return INT_SEQUENCE_RANGE;
    }

    *out_value = seq->items[index];
    return INT_SEQUENCE_OK;
}

static size_t array_seq_size(const IntSequenceArray *seq) {
    return seq == NULL ? 0 : seq->size;
}

// 双向链表后端实现
typedef struct IntSeqNode IntSeqNode;

struct IntSeqNode {
    int value;
    IntSeqNode *prev;
    IntSeqNode *next;
};

typedef struct IntSequenceLinked IntSequenceLinked;

struct IntSequenceLinked {
    IntSeqNode *head;
    IntSeqNode *tail;
    size_t size;
};

static IntSequenceLinked *linked_seq_create(void) {
    IntSequenceLinked *seq = malloc(sizeof *seq);
    if (seq == NULL) {
        return NULL;
    }
    seq->head = NULL;
    seq->tail = NULL;
    seq->size = 0;
    return seq;
}

static void linked_seq_destroy(IntSequenceLinked *seq) {
    if (seq == NULL) {
        return;
    }
    IntSeqNode *curr = seq->head;
    while (curr != NULL) {
        IntSeqNode *next = curr->next;
        free(curr);
        curr = next;
    }
    free(seq);
}

static IntSeqNode *linked_seq_node_at(
    const IntSequenceLinked *seq,
    size_t index
) {
    if (index < seq->size / 2) {
        IntSeqNode *node = seq->head;
        for (size_t i = 0; i < index; ++i) {
            node = node->next;
        }
        return node;
    } else {
        IntSeqNode *node = seq->tail;
        for (size_t i = seq->size - 1; i > index; --i) {
            node = node->prev;
        }
        return node;
    }
}

static IntSequenceResult linked_seq_insert(
    IntSequenceLinked *seq,
    size_t index,
    int value
) {
    if (seq == NULL) {
        return INT_SEQUENCE_INVALID;
    }
    if (index > seq->size) {
        return INT_SEQUENCE_RANGE;
    }
    if (seq->size == SIZE_MAX) {
        return INT_SEQUENCE_OUT_OF_MEMORY;
    }

    IntSeqNode *node = malloc(sizeof *node);
    if (node == NULL) {
        return INT_SEQUENCE_OUT_OF_MEMORY;
    }
    node->value = value;

    if (seq->size == 0) {
        node->prev = NULL;
        node->next = NULL;
        seq->head = node;
        seq->tail = node;
    } else if (index == 0) {
        node->prev = NULL;
        node->next = seq->head;
        seq->head->prev = node;
        seq->head = node;
    } else if (index == seq->size) {
        node->prev = seq->tail;
        node->next = NULL;
        seq->tail->next = node;
        seq->tail = node;
    } else {
        IntSeqNode *target = linked_seq_node_at(seq, index);
        node->prev = target->prev;
        node->next = target;
        target->prev->next = node;
        target->prev = node;
    }

    ++seq->size;
    return INT_SEQUENCE_OK;
}

static IntSequenceResult linked_seq_erase(
    IntSequenceLinked *seq,
    size_t index,
    int *out_value
) {
    if (seq == NULL || out_value == NULL) {
        return INT_SEQUENCE_INVALID;
    }
    if (index >= seq->size) {
        return INT_SEQUENCE_RANGE;
    }

    IntSeqNode *target = linked_seq_node_at(seq, index);
    *out_value = target->value;

    if (seq->size == 1) {
        seq->head = NULL;
        seq->tail = NULL;
    } else if (target == seq->head) {
        seq->head = target->next;
        seq->head->prev = NULL;
    } else if (target == seq->tail) {
        seq->tail = target->prev;
        seq->tail->next = NULL;
    } else {
        target->prev->next = target->next;
        target->next->prev = target->prev;
    }

    free(target);
    --seq->size;
    return INT_SEQUENCE_OK;
}

static IntSequenceResult linked_seq_get(
    const IntSequenceLinked *seq,
    size_t index,
    int *out_value
) {
    if (seq == NULL || out_value == NULL) {
        return INT_SEQUENCE_INVALID;
    }
    if (index >= seq->size) {
        return INT_SEQUENCE_RANGE;
    }

    IntSeqNode *node = linked_seq_node_at(seq, index);
    *out_value = node->value;
    return INT_SEQUENCE_OK;
}

static size_t linked_seq_size(const IntSequenceLinked *seq) {
    return seq == NULL ? 0 : seq->size;
}

int main(void) {
    IntSequenceArray *arr = array_seq_create();
    assert(arr != NULL);
    assert(array_seq_size(arr) == 0);

    assert(array_seq_insert(arr, 0, 10) == INT_SEQUENCE_OK);
    assert(array_seq_insert(arr, 1, 30) == INT_SEQUENCE_OK);
    assert(array_seq_insert(arr, 1, 20) == INT_SEQUENCE_OK);
    assert(array_seq_size(arr) == 3);

    int val = 0;
    assert(array_seq_get(arr, 0, &val) == INT_SEQUENCE_OK && val == 10);
    assert(array_seq_get(arr, 1, &val) == INT_SEQUENCE_OK && val == 20);
    assert(array_seq_get(arr, 2, &val) == INT_SEQUENCE_OK && val == 30);
    assert(array_seq_get(arr, 3, &val) == INT_SEQUENCE_RANGE);

    assert(array_seq_erase(arr, 1, &val) == INT_SEQUENCE_OK && val == 20);
    assert(array_seq_size(arr) == 2);
    assert(array_seq_get(arr, 1, &val) == INT_SEQUENCE_OK && val == 30);
    array_seq_destroy(arr);

    IntSequenceLinked *lnk = linked_seq_create();
    assert(lnk != NULL);
    assert(linked_seq_size(lnk) == 0);

    assert(linked_seq_insert(lnk, 0, 10) == INT_SEQUENCE_OK);
    assert(linked_seq_insert(lnk, 1, 30) == INT_SEQUENCE_OK);
    assert(linked_seq_insert(lnk, 1, 20) == INT_SEQUENCE_OK);
    assert(linked_seq_size(lnk) == 3);

    assert(linked_seq_get(lnk, 0, &val) == INT_SEQUENCE_OK && val == 10);
    assert(linked_seq_get(lnk, 1, &val) == INT_SEQUENCE_OK && val == 20);
    assert(linked_seq_get(lnk, 2, &val) == INT_SEQUENCE_OK && val == 30);
    assert(linked_seq_get(lnk, 3, &val) == INT_SEQUENCE_RANGE);

    assert(linked_seq_erase(lnk, 1, &val) == INT_SEQUENCE_OK && val == 20);
    assert(linked_seq_size(lnk) == 2);
    assert(linked_seq_get(lnk, 1, &val) == INT_SEQUENCE_OK && val == 30);
    linked_seq_destroy(lnk);

    return 0;
}
```

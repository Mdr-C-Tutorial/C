# 题解：31108

## 题目

在 C 语言中基于虚函数表（Virtual Method Table, VTable）模拟面向对象多态分派机制，设计并实现通用多态字节流抽象数据类型（ByteStream ADT）：

```c
#include <stdbool.h>
#include <stddef.h>

typedef struct ByteStream ByteStream;

typedef struct {
    size_t (*read)(ByteStream *stream, void *buffer, size_t count);
    size_t (*write)(ByteStream *stream, const void *buffer, size_t count);
    bool (*flush)(ByteStream *stream);
    void (*destroy)(ByteStream *stream);
} ByteStreamOps;

struct ByteStream {
    const ByteStreamOps *ops;
};

size_t stream_read(ByteStream *stream, void *buffer, size_t count);
size_t stream_write(ByteStream *stream, const void *buffer, size_t count);
bool stream_flush(ByteStream *stream);
void stream_destroy(ByteStream *stream);

ByteStream *mem_stream_create(void *buffer, size_t capacity);
ByteStream *dyn_stream_create(size_t initial_capacity);
const void *dyn_stream_data(const ByteStream *stream, size_t *out_len);
```

要求：

1. **统一多态分派**：调用方仅持有 `ByteStream *` 指针，所有 I/O 操作通过 `stream_read` 等公开包装函数统一调用，包装函数内部通过 `ops` 虚表路由至具体流实现。
2. **结构体单继承内存对齐**：派生类结构体必须将 `ByteStream base;` 作为其首个成员，确保派生类对象指针与基类指针可安全互转。
3. **定长内存流（`MemStream`）**：封装外部传入的固定大小内存块，维护独立的读写偏移；写入不能超出物理容量上限。
4. **动态自扩容流（`DynStream`）**：拥有自己的堆分配连续存储，写入超出当前容量时按几何倍增自动扩容，并支持通过 `dyn_stream_data` 获取底层有效缓冲视图。
5. **虚析构契约**：调用 `stream_destroy` 时必须通过虚表分派执行具体流特有的析构逻辑，彻底释放派生类持有的所有资源，杜绝内存泄漏。

## 解析

C 语言虽然没有面向对象语言中的 `class` 与 `virtual` 关键字，但现代工业级系统架构（例如 Linux 内核的虚拟文件系统 VFS `struct file_operations`、GLib 的 `GObject`、SQLite 驱动层）正是通过结构体函数指针表构建了极高抽象层次的多态 ADT。

1. **虚函数表（VTable）对象模型**：
   - 基类 `ByteStream` 仅包含一个指向虚函数表常量的指针：`const ByteStreamOps *ops;`。
   - 所有具体的派生类实现（如 `MemStream` 和 `DynStream`）都将 `ByteStream base;` 放置在结构体内存布局的起始位置（偏移量为 0）。
   - C 语言标准保证结构体的首个成员与结构体本身的起始地址相同。因此，将派生类指针转换为基类指针、或者将 `ByteStream *stream` 强制转换为具体的 `MemStream *` 或 `DynStream *` 都是合法且类型对齐安全的。

2. **多态分派与生命周期**：
   - 虚表中的每个函数指针代表一个抽象方法：`read`、`write`、`flush`、`destroy`。
   - 虚函数表本身为静态只读全局常量（如 `static const ByteStreamOps g_mem_stream_ops`），所有同类对象共享同一张虚表，不需要为每个对象重复开辟函数指针内存。
   - 虚析构：`stream_destroy` 不直接 `free(stream)`，而是调用 `stream->ops->destroy(stream)`。对于 `MemStream`，外部注入的数组不归流管辖，仅释放封装结构体；对于 `DynStream`，析构函数负责释放动态缓冲数组 `self->data` 和结构体自身。

3. **动态扩容流的不变式与失败安全**：
   - `DynStream` 在扩容时先用局部临时指针接收 `realloc` 结果，确认成功后再提交至 `self->data` 与 `self->capacity`，维持强失败安全性。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

typedef struct ByteStream ByteStream;

typedef struct {
    size_t (*read)(ByteStream *stream, void *buffer, size_t count);
    size_t (*write)(ByteStream *stream, const void *buffer, size_t count);
    bool (*flush)(ByteStream *stream);
    void (*destroy)(ByteStream *stream);
} ByteStreamOps;

struct ByteStream {
    const ByteStreamOps *ops;
};

size_t stream_read(ByteStream *stream, void *buffer, size_t count) {
    if (stream == NULL || stream->ops == NULL || stream->ops->read == NULL) {
        return 0;
    }
    return stream->ops->read(stream, buffer, count);
}

size_t stream_write(ByteStream *stream, const void *buffer, size_t count) {
    if (stream == NULL || stream->ops == NULL || stream->ops->write == NULL) {
        return 0;
    }
    return stream->ops->write(stream, buffer, count);
}

bool stream_flush(ByteStream *stream) {
    if (stream == NULL || stream->ops == NULL || stream->ops->flush == NULL) {
        return false;
    }
    return stream->ops->flush(stream);
}

void stream_destroy(ByteStream *stream) {
    if (stream == NULL) {
        return;
    }
    if (stream->ops != NULL && stream->ops->destroy != NULL) {
        stream->ops->destroy(stream);
    }
}

// 具体实现 1：定长内存流 (MemStream)
typedef struct {
    ByteStream base;
    uint8_t *buffer;
    size_t capacity;
    size_t read_pos;
    size_t write_pos;
} MemStream;

static size_t mem_stream_read_impl(ByteStream *stream, void *buf, size_t count) {
    MemStream *self = (MemStream *)stream;
    if (buf == NULL || self->read_pos >= self->write_pos) {
        return 0;
    }
    size_t available = self->write_pos - self->read_pos;
    size_t to_read = (count < available) ? count : available;
    memcpy(buf, self->buffer + self->read_pos, to_read);
    self->read_pos += to_read;
    return to_read;
}

static size_t mem_stream_write_impl(ByteStream *stream, const void *buf, size_t count) {
    MemStream *self = (MemStream *)stream;
    if (buf == NULL || self->write_pos >= self->capacity) {
        return 0;
    }
    size_t available = self->capacity - self->write_pos;
    size_t to_write = (count < available) ? count : available;
    memcpy(self->buffer + self->write_pos, buf, to_write);
    self->write_pos += to_write;
    return to_write;
}

static bool mem_stream_flush_impl(ByteStream *stream) {
    (void)stream;
    return true;
}

static void mem_stream_destroy_impl(ByteStream *stream) {
    MemStream *self = (MemStream *)stream;
    free(self);
}

static const ByteStreamOps g_mem_stream_ops = {
    .read = mem_stream_read_impl,
    .write = mem_stream_write_impl,
    .flush = mem_stream_flush_impl,
    .destroy = mem_stream_destroy_impl
};

ByteStream *mem_stream_create(void *buffer, size_t capacity) {
    if (buffer == NULL && capacity > 0) {
        return NULL;
    }
    MemStream *self = malloc(sizeof(MemStream));
    if (self == NULL) {
        return NULL;
    }
    self->base.ops = &g_mem_stream_ops;
    self->buffer = (uint8_t *)buffer;
    self->capacity = capacity;
    self->read_pos = 0;
    self->write_pos = 0;
    return &self->base;
}

// 具体实现 2：动态扩容字节流 (DynStream)
typedef struct {
    ByteStream base;
    uint8_t *data;
    size_t length;
    size_t capacity;
    size_t read_pos;
} DynStream;

static size_t dyn_stream_read_impl(ByteStream *stream, void *buf, size_t count) {
    DynStream *self = (DynStream *)stream;
    if (buf == NULL || self->read_pos >= self->length) {
        return 0;
    }
    size_t available = self->length - self->read_pos;
    size_t to_read = (count < available) ? count : available;
    memcpy(buf, self->data + self->read_pos, to_read);
    self->read_pos += to_read;
    return to_read;
}

static size_t dyn_stream_write_impl(ByteStream *stream, const void *buf, size_t count) {
    DynStream *self = (DynStream *)stream;
    if (buf == NULL || count == 0) {
        return 0;
    }
    if (count > SIZE_MAX - self->length) {
        return 0;
    }

    size_t required = self->length + count;
    if (required > self->capacity) {
        size_t next = (self->capacity == 0) ? 16 : self->capacity * 2;
        while (next < required) {
            if (next > SIZE_MAX / 2) {
                next = SIZE_MAX;
                break;
            }
            next *= 2;
        }
        uint8_t *new_data = realloc(self->data, next);
        if (new_data == NULL) {
            return 0;
        }
        self->data = new_data;
        self->capacity = next;
    }

    memcpy(self->data + self->length, buf, count);
    self->length += count;
    return count;
}

static bool dyn_stream_flush_impl(ByteStream *stream) {
    (void)stream;
    return true;
}

static void dyn_stream_destroy_impl(ByteStream *stream) {
    DynStream *self = (DynStream *)stream;
    free(self->data);
    free(self);
}

static const ByteStreamOps g_dyn_stream_ops = {
    .read = dyn_stream_read_impl,
    .write = dyn_stream_write_impl,
    .flush = dyn_stream_flush_impl,
    .destroy = dyn_stream_destroy_impl
};

ByteStream *dyn_stream_create(size_t initial_capacity) {
    DynStream *self = malloc(sizeof(DynStream));
    if (self == NULL) {
        return NULL;
    }
    self->base.ops = &g_dyn_stream_ops;
    self->data = (initial_capacity > 0) ? malloc(initial_capacity) : NULL;
    if (initial_capacity > 0 && self->data == NULL) {
        free(self);
        return NULL;
    }
    self->length = 0;
    self->capacity = initial_capacity;
    self->read_pos = 0;
    return &self->base;
}

const void *dyn_stream_data(const ByteStream *stream, size_t *out_len) {
    if (stream == NULL || stream->ops != &g_dyn_stream_ops) {
        if (out_len != NULL) {
            *out_len = 0;
        }
        return NULL;
    }
    const DynStream *self = (const DynStream *)stream;
    if (out_len != NULL) {
        *out_len = self->length;
    }
    return self->data;
}

int main(void) {
    stream_destroy(NULL);

    uint8_t raw_buf[16];
    ByteStream *m_stream = mem_stream_create(raw_buf, sizeof(raw_buf));
    assert(m_stream != NULL);

    assert(stream_write(m_stream, "Hello", 5) == 5);
    assert(stream_write(m_stream, " World", 6) == 6);
    assert(stream_flush(m_stream));

    char read_buf[32] = {0};
    assert(stream_read(m_stream, read_buf, 5) == 5);
    read_buf[5] = '\0';
    assert(strcmp(read_buf, "Hello") == 0);

    assert(stream_read(m_stream, read_buf + 5, 10) == 6);
    read_buf[11] = '\0';
    assert(strcmp(read_buf, "Hello World") == 0);

    stream_destroy(m_stream);

    ByteStream *d_stream = dyn_stream_create(4);
    assert(d_stream != NULL);

    const char *msg = "Polymorphic Abstract Data Type in C";
    size_t msg_len = strlen(msg);
    assert(stream_write(d_stream, msg, msg_len) == msg_len);

    size_t actual_len = 0;
    const char *payload = (const char *)dyn_stream_data(d_stream, &actual_len);
    assert(payload != NULL && actual_len == msg_len);
    assert(memcmp(payload, msg, msg_len) == 0);

    char dyn_read[64] = {0};
    assert(stream_read(d_stream, dyn_read, msg_len) == msg_len);
    assert(memcmp(dyn_read, msg, msg_len) == 0);

    stream_destroy(d_stream);
    return 0;
}
```

# 题解：31204

## 题目

实现整数单链表：

```c
#include <stdbool.h>
#include <stddef.h>

typedef struct IntChain IntChain;

IntChain *int_chain_create(void);
void int_chain_destroy(IntChain *chain);
bool int_chain_push_front(IntChain *chain, int value);
bool int_chain_push_back(IntChain *chain, int value);
bool int_chain_pop_front(IntChain *chain, int *out_value);
bool int_chain_remove_first(IntChain *chain, int value);
size_t int_chain_size(const IntChain *chain);
```

要求：

1. `int_chain_create` 返回的链表由调用方拥有；链表拥有其全部结点。
2. 头部插入、尾部插入和头部删除的时间复杂度均为 $O(1)$，删除首个指定值的结点为 $O(n)$。
3. 分配失败或大小无法表示时，插入函数返回 `false`，链表保持不变。
4. 空链表或参数无效时，`int_chain_pop_front` 返回 `false`，不得修改输出对象。
5. 删除最后一个结点后，头指针和尾指针都必须恢复为空指针。
6. `int_chain_destroy(NULL)` 应当安全，并且不得泄漏结点。

## 解析

单链表通过指针将离散的物理存储单元串联为逻辑线性序列。为了使尾端追加摆脱 $O(n)$ 的从头遍历，链表头结构体中同时缓存了头指针 `head` 与尾指针 `tail`。

1. **首尾指针不变式**：
   - 空链表状态：`head == NULL`，`tail == NULL`，`size == 0`。
   - 非空链表状态：`head != NULL`，`tail != NULL`，且末结点的后继指针满足 `tail->next == NULL`。
   - 插入操作在为新结点分配内存成功后才接入链表，保证在堆耗尽时链表拓扑关系完好无损。
   - 尾部插入直接连接 `tail->next = node` 并推进 `tail = node`，时间复杂度为严格的 $O(1)$。

2. **边界删除与尾指针同步维护**：
   - 在头部弹出 `pop_front` 或按值删除 `remove_first` 时，若被删除的恰好是链表唯一的结点（即删除后 `head == NULL`），必须显式将 `tail` 置为 `NULL`，防止悬挂指针破坏空表不变式。
   - 在按值删除非首结点时，若目标结点为尾结点（`current == chain->tail`），必须将 `tail` 更新为前驱指针 `previous`。

3. **资源归属与彻底销毁**：
   - 外部调用方持有链表控制块指针 `IntChain *`，而内部结点的内存地址绝不泄漏给调用方。
   - 销毁链表时，沿着 `next` 指向逐个释放每个结点，最后释放主结构体自身，确保 0 内存泄漏。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct IntChain IntChain;

IntChain *int_chain_create(void);
void int_chain_destroy(IntChain *chain);
bool int_chain_push_front(IntChain *chain, int value);
bool int_chain_push_back(IntChain *chain, int value);
bool int_chain_pop_front(IntChain *chain, int *out_value);
bool int_chain_remove_first(IntChain *chain, int value);
size_t int_chain_size(const IntChain *chain);

typedef struct IntNode {
    int value;
    struct IntNode *next;
} IntNode;

struct IntChain {
    IntNode *head;
    IntNode *tail;
    size_t size;
};

IntChain *int_chain_create(void) {
    IntChain *chain = malloc(sizeof *chain);
    if (chain == NULL) {
        return NULL;
    }

    chain->head = NULL;
    chain->tail = NULL;
    chain->size = 0;
    return chain;
}

void int_chain_destroy(IntChain *chain) {
    if (chain == NULL) {
        return;
    }

    IntNode *node = chain->head;
    while (node != NULL) {
        IntNode *next = node->next;
        free(node);
        node = next;
    }
    free(chain);
}

static IntNode *int_node_create(int value) {
    IntNode *node = malloc(sizeof *node);
    if (node == NULL) {
        return NULL;
    }

    node->value = value;
    node->next = NULL;
    return node;
}

bool int_chain_push_front(IntChain *chain, int value) {
    if (chain == NULL || chain->size == SIZE_MAX) {
        return false;
    }

    IntNode *node = int_node_create(value);
    if (node == NULL) {
        return false;
    }

    node->next = chain->head;
    chain->head = node;
    if (chain->tail == NULL) {
        chain->tail = node;
    }
    ++chain->size;
    return true;
}

bool int_chain_push_back(IntChain *chain, int value) {
    if (chain == NULL || chain->size == SIZE_MAX) {
        return false;
    }

    IntNode *node = int_node_create(value);
    if (node == NULL) {
        return false;
    }

    if (chain->tail == NULL) {
        chain->head = node;
        chain->tail = node;
    } else {
        chain->tail->next = node;
        chain->tail = node;
    }
    ++chain->size;
    return true;
}

bool int_chain_pop_front(IntChain *chain, int *out_value) {
    if (chain == NULL || out_value == NULL || chain->head == NULL) {
        return false;
    }

    IntNode *node = chain->head;
    *out_value = node->value;
    chain->head = node->next;
    if (chain->head == NULL) {
        chain->tail = NULL;
    }
    free(node);
    --chain->size;
    return true;
}

bool int_chain_remove_first(IntChain *chain, int value) {
    if (chain == NULL || chain->head == NULL) {
        return false;
    }

    IntNode *previous = NULL;
    IntNode *current = chain->head;

    while (current != NULL && current->value != value) {
        previous = current;
        current = current->next;
    }

    if (current == NULL) {
        return false;
    }

    if (previous == NULL) {
        chain->head = current->next;
    } else {
        previous->next = current->next;
    }

    if (current == chain->tail) {
        chain->tail = previous;
    }

    free(current);
    --chain->size;
    return true;
}

size_t int_chain_size(const IntChain *chain) {
    return chain == NULL ? 0 : chain->size;
}

int main(void) {
    int_chain_destroy(NULL);
    assert(int_chain_size(NULL) == 0);

    IntChain *c = int_chain_create();
    assert(c != NULL);
    assert(int_chain_size(c) == 0);

    int val = 0;
    assert(!int_chain_pop_front(c, &val));

    assert(int_chain_push_back(c, 20));
    assert(int_chain_push_front(c, 10));
    assert(int_chain_push_back(c, 30));
    assert(int_chain_size(c) == 3);

    assert(int_chain_pop_front(c, &val) && val == 10);
    assert(int_chain_size(c) == 2);

    assert(int_chain_remove_first(c, 30));
    assert(int_chain_size(c) == 1);

    assert(int_chain_remove_first(c, 20));
    assert(int_chain_size(c) == 0);
    assert(c->head == NULL && c->tail == NULL);

    int_chain_destroy(c);
    return 0;
}
```

# 题解：31206

## 题目

实现整数双端队列：

```c
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>

typedef struct IntDeque IntDeque;

IntDeque *int_deque_create(void);
void int_deque_destroy(IntDeque *deque);
bool int_deque_push_front(IntDeque *deque, int64_t value);
bool int_deque_push_back(IntDeque *deque, int64_t value);
bool int_deque_pop_front(IntDeque *deque, int64_t *out_value);
bool int_deque_pop_back(IntDeque *deque, int64_t *out_value);
size_t int_deque_size(const IntDeque *deque);
```

要求：

1. 使用环形存储，同时支持两端插入和删除；从空队列弹出或参数无效时返回 `false`，不得修改输出对象。
2. 扩容时保持从首端到尾端的逻辑顺序；分配失败或大小计算溢出时操作失败，队列保持不变。
3. `int_deque_destroy(NULL)` 安全，`int_deque_size(NULL)` 返回 0。
4. 两端操作的摊还时间复杂度为 $O(1)$；说明首下标、元素数量和容量之间的不变式。

## 解析

双端队列（Deque）允许在序列的首尾两端以 $O(1)$ 的时间复杂度进行插入和删除。若采用普通平移数组，在首端插入或删除将耗费 $O(n)$ 时间。环形缓冲区通过首下标 `head` 的前后环绕跳变，消除了数据搬移开销。

1. **环形物理映射与不变式**：
   - 内部结构包含底层数组指针 `int64_t *items`、容量 `capacity`、首元素物理下标 `head` 与当前元素数量 `size`。
   - 结构不变式约束：
     ```text
     size <= capacity
     capacity == 0 当且仅当 items == NULL
     非空容量下必须满足 head < capacity
     ```
   - 逻辑偏移量 `offset`（$0 \le offset < size$）对应的物理下标计算：
     为防止无符号整数加法溢出，定义安全距离 $distance = capacity - head$。若 $offset \ge distance$，物理下标为 $offset - distance$；否则物理下标为 $head + offset$。

2. **首尾操作逻辑**：
   - 头部插入：`head` 向左环绕回退一步：`head = (head == 0) ? (capacity - 1) : (head - 1)`，写入新元素，`size++`。
   - 尾部插入：直接写入逻辑末端对应的物理槽位 `deque_at(deque, size)`，`size++`。
   - 头部删除：提取 `items[head]`，`head = (head + 1 == capacity) ? 0 : (head + 1)`，`size--`。
   - 尾部删除：提取逻辑末尾元素 `items[deque_at(deque, size - 1)]`，`size--`。

3. **顺序保真扩容**：
   - 当队列已满（`size == capacity`）时触发几何扩容。
   - 分配新的双倍连续存储，按逻辑顺序 $0 \dots size-1$ 将旧数组中的元素依次复制至新数组的前端区间 $[0, size-1]$。
   - 复制完成后释放旧数组并将 `head` 规约重置为 0。若内存分配失败，原队列保持完全不变。两端操作的摊还时间复杂度为严格的 $O(1)$。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct IntDeque IntDeque;

struct IntDeque {
    int64_t *items;
    size_t capacity;
    size_t head;
    size_t size;
};

static size_t deque_at(
    const IntDeque *deque,
    size_t offset
) {
    size_t distance = deque->capacity - deque->head;
    if (offset >= distance) {
        return offset - distance;
    }
    return deque->head + offset;
}

static bool deque_grow(IntDeque *deque) {
    if (deque->capacity > SIZE_MAX / sizeof *deque->items) {
        return false;
    }

    size_t maximum = SIZE_MAX / sizeof *deque->items;
    size_t next = deque->capacity == 0 ? 8 : deque->capacity;
    if (next > maximum) {
        return false;
    }
    while (next <= deque->size) {
        if (next > maximum / 2) {
            next = maximum;
            break;
        }
        next *= 2;
    }
    if (next <= deque->size || next == 0) {
        return false;
    }

    int64_t *replacement = malloc(next * sizeof *replacement);
    if (replacement == NULL) {
        return false;
    }
    for (size_t i = 0; i < deque->size; ++i) {
        replacement[i] = deque->items[deque_at(deque, i)];
    }
    free(deque->items);
    deque->items = replacement;
    deque->capacity = next;
    deque->head = 0;
    return true;
}

IntDeque *int_deque_create(void) {
    IntDeque *deque = malloc(sizeof *deque);
    if (deque == NULL) {
        return NULL;
    }
    deque->items = NULL;
    deque->capacity = 0;
    deque->head = 0;
    deque->size = 0;
    return deque;
}

void int_deque_destroy(IntDeque *deque) {
    if (deque == NULL) {
        return;
    }
    free(deque->items);
    free(deque);
}

static bool deque_prepare_push(IntDeque *deque) {
    if (deque == NULL) {
        return false;
    }
    if (deque->size < deque->capacity) {
        return true;
    }
    return deque_grow(deque);
}

bool int_deque_push_front(IntDeque *deque, int64_t value) {
    if (!deque_prepare_push(deque)) {
        return false;
    }
    deque->head = deque->head == 0
        ? deque->capacity - 1
        : deque->head - 1;
    deque->items[deque->head] = value;
    ++deque->size;
    return true;
}

bool int_deque_push_back(IntDeque *deque, int64_t value) {
    if (!deque_prepare_push(deque)) {
        return false;
    }
    deque->items[deque_at(deque, deque->size)] = value;
    ++deque->size;
    return true;
}

bool int_deque_pop_front(IntDeque *deque, int64_t *out_value) {
    if (deque == NULL || out_value == NULL || deque->size == 0) {
        return false;
    }
    *out_value = deque->items[deque->head];
    deque->head = deque->head + 1 == deque->capacity
        ? 0
        : deque->head + 1;
    --deque->size;
    if (deque->size == 0) {
        deque->head = 0;
    }
    return true;
}

bool int_deque_pop_back(IntDeque *deque, int64_t *out_value) {
    if (deque == NULL || out_value == NULL || deque->size == 0) {
        return false;
    }
    size_t index = deque_at(deque, deque->size - 1);
    *out_value = deque->items[index];
    --deque->size;
    if (deque->size == 0) {
        deque->head = 0;
    }
    return true;
}

size_t int_deque_size(const IntDeque *deque) {
    return deque == NULL ? 0 : deque->size;
}

int main(void) {
    int_deque_destroy(NULL);
    assert(int_deque_size(NULL) == 0);

    IntDeque *dq = int_deque_create();
    assert(dq != NULL);
    assert(int_deque_size(dq) == 0);

    int64_t val = 0;
    assert(!int_deque_pop_front(dq, &val));
    assert(!int_deque_pop_back(dq, &val));

    assert(int_deque_push_back(dq, 100));
    assert(int_deque_push_front(dq, 50));
    assert(int_deque_push_back(dq, 200));
    assert(int_deque_push_front(dq, 25));
    assert(int_deque_size(dq) == 4);

    assert(int_deque_pop_front(dq, &val) && val == 25);
    assert(int_deque_pop_back(dq, &val) && val == 200);
    assert(int_deque_size(dq) == 2);

    for (int i = 0; i < 30; ++i) {
        assert(int_deque_push_back(dq, i));
    }
    assert(int_deque_size(dq) == 32);

    assert(int_deque_pop_front(dq, &val) && val == 50);
    assert(int_deque_pop_front(dq, &val) && val == 100);
    for (int i = 0; i < 30; ++i) {
        assert(int_deque_pop_front(dq, &val) && val == i);
    }
    assert(int_deque_size(dq) == 0);

    int_deque_destroy(dq);
    return 0;
}
```

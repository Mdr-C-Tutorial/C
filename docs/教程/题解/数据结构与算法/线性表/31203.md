# 题解：31203

## 题目

实现整数顺序表的区间插入与删除：

```c
#include <stdbool.h>
#include <stddef.h>

typedef struct {
    int *items;
    size_t size;
    size_t capacity;
} IntList;

void int_list_init(IntList *list);
void int_list_destroy(IntList *list);
bool int_list_insert_range(
    IntList *list,
    size_t index,
    const int *source,
    size_t count
);
bool int_list_erase_range(
    IntList *list,
    size_t index,
    size_t count
);
```

要求：

1. 始终保持 `size <= capacity`，并保证 `capacity == 0` 当且仅当 `items == NULL`。
2. 插入位置可以等于 `size`。`count` 为 0 时插入成功，此时 `source` 可以为 `NULL`。
3. 当 `count > 0` 时，`source` 必须有效，且不得与顺序表拥有的存储区域重叠。
4. 插入时使用几何增长扩容；参数无效、大小计算溢出或分配失败时返回 `false`，并保持原顺序表不变。
5. 删除区间必须完全位于已有元素范围内。删除成功后保持剩余元素的相对顺序，不要求缩容。
6. 区间搬移必须使用适合重叠存储区域的标准库函数。

## 解析

动态顺序表由连续内存块构成。在支持单点插入删除的基础上扩展出“区间插入”与“区间删除”，既能减少多次单点插入导致的重复扩容开销，又能避免逐个平移带来的多次内存搬移。

1. **表示不变式约束**：
   - 顺序表对象始终维持结构守恒：
     ```text
     size <= capacity
     capacity == 0 当且仅当 items == NULL
     capacity <= SIZE_MAX / sizeof(int)
     ```
   - 初始化时设定三者归零；销毁时释放物理存储并调用初始化重置结构。

2. **区间移动与重叠区域安全**：
   - 在执行区间插入时，需要将原数组中从逻辑位置 `index` 开始的后缀整体向右腾挪 `count` 个槽位。由于源内存区间与目标内存区间高度重叠，必须使用具备重叠安全保障的标准库函数 `memmove`，严禁使用未定义行为的 `memcpy`。
   - 腾出连续槽位后，来源数据 `source` 契约明确与顺序表自身存储互不重叠，此时调用 `memcpy` 将外部元素拷贝填入。
   - 区间删除时，将下标 `index + count` 处的后缀整体向左搬移覆盖至 `index` 位置，同样使用 `memmove`。

3. **强异常安全保证**：
   - 在修改任何现有数据之前，先进行参数合法性检查（`index <= list->size`、溢出校验 `count <= SIZE_MAX - list->size`）。
   - 几何扩容通过临时指针接收 `realloc` 结果，若分配失败则原顺序表与已有元素完全保持原样，直接返回 `false`。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int *items;
    size_t size;
    size_t capacity;
} IntList;

static bool int_list_valid(const IntList *list) {
    if (list == NULL || list->size > list->capacity) {
        return false;
    }
    if ((list->capacity == 0) != (list->items == NULL)) {
        return false;
    }
    return list->capacity <= SIZE_MAX / sizeof *list->items;
}

void int_list_init(IntList *list) {
    if (list == NULL) {
        return;
    }
    list->items = NULL;
    list->size = 0;
    list->capacity = 0;
}

void int_list_destroy(IntList *list) {
    if (list == NULL) {
        return;
    }
    free(list->items);
    int_list_init(list);
}

static bool int_list_reserve(
    IntList *list,
    size_t minimum
) {
    size_t maximum = SIZE_MAX / sizeof *list->items;
    if (minimum > maximum) {
        return false;
    }
    if (minimum <= list->capacity) {
        return true;
    }

    size_t next = list->capacity == 0
        ? (maximum < 8 ? maximum : 8)
        : list->capacity;

    while (next < minimum) {
        if (next > maximum / 2) {
            next = maximum;
        } else {
            next *= 2;
        }
    }

    int *replacement = realloc(
        list->items,
        next * sizeof *replacement
    );
    if (replacement == NULL) {
        return false;
    }

    list->items = replacement;
    list->capacity = next;
    return true;
}

bool int_list_insert_range(
    IntList *list,
    size_t index,
    const int *source,
    size_t count
) {
    if (!int_list_valid(list) || index > list->size) {
        return false;
    }
    if (count == 0) {
        return true;
    }
    if (source == NULL || count > SIZE_MAX - list->size) {
        return false;
    }

    size_t new_size = list->size + count;
    if (!int_list_reserve(list, new_size)) {
        return false;
    }

    size_t tail_count = list->size - index;
    memmove(
        list->items + index + count,
        list->items + index,
        tail_count * sizeof *list->items
    );
    memcpy(
        list->items + index,
        source,
        count * sizeof *list->items
    );
    list->size = new_size;
    return true;
}

bool int_list_erase_range(
    IntList *list,
    size_t index,
    size_t count
) {
    if (!int_list_valid(list) || index > list->size) {
        return false;
    }
    if (count > list->size - index) {
        return false;
    }
    if (count == 0) {
        return true;
    }

    size_t after = index + count;
    size_t tail_count = list->size - after;
    memmove(
        list->items + index,
        list->items + after,
        tail_count * sizeof *list->items
    );
    list->size -= count;
    return true;
}

int main(void) {
    IntList list;
    int_list_init(&list);
    assert(list.size == 0 && list.capacity == 0 && list.items == NULL);

    int data1[] = {10, 20, 30};
    assert(int_list_insert_range(&list, 0, data1, 3));
    assert(list.size == 3);
    assert(list.items[0] == 10 && list.items[1] == 20 && list.items[2] == 30);

    int data2[] = {15, 18};
    assert(int_list_insert_range(&list, 1, data2, 2));
    assert(list.size == 5);
    assert(list.items[0] == 10 && list.items[1] == 15 && list.items[2] == 18 &&
           list.items[3] == 20 && list.items[4] == 30);

    assert(int_list_erase_range(&list, 1, 2));
    assert(list.size == 3);
    assert(list.items[0] == 10 && list.items[1] == 20 && list.items[2] == 30);

    assert(int_list_insert_range(&list, 2, NULL, 0));
    assert(list.size == 3);

    assert(!int_list_erase_range(&list, 2, 5));
    assert(list.size == 3);

    int_list_destroy(&list);
    assert(list.items == NULL && list.size == 0 && list.capacity == 0);
    return 0;
}
```

# 题解：31208

## 题目

结合双向链表与线性查找，设计并实现具有固定容量上限的最近最少使用缓存（LRU Cache ADT）：

```c
#include <stdbool.h>
#include <stddef.h>

typedef struct LruCache LruCache;

LruCache *lru_create(size_t capacity);
void lru_destroy(LruCache *cache);
bool lru_get(LruCache *cache, int key, int *out_val);
bool lru_put(LruCache *cache, int key, int val);
size_t lru_size(const LruCache *cache);
```

要求：

1. **时序维护不变式**：双向链表严格维护访问时序——最新被访问（读取或写入）的结点始终位于链表头部，最久未被访问的结点始终位于链表尾部。
2. **读取与提升契约**：`lru_get` 查询键：命中时将该结点在常数时间内摘除并移到表头，通过 `*out_val` 返回数值并返回 `true`；未命中返回 `false`，不得修改输出指针指向的对象。
3. **写入与淘汰机制**：`lru_put` 写入键值：若键已存在，更新其数值并瞬移至表头；若键不存在且容量未达上限，创建新结点插入表头；若已达容量上限，淘汰并释放表尾最久未用的结点，然后将新结点插入表头。
4. **异常与生命周期安全**：`capacity == 0` 时创建失败返回 `NULL`；分配失败时保持原有缓存状态不变；`lru_destroy` 彻底释放所有结点与结构体内存。

## 解析

缓存置换算法的核心目标是在有限的存储空间内尽可能提高命中率。LRU（Least Recently Used）基于时间局部性原理，假定最近刚刚被访问的数据在不久的将来最有可能被再次访问。

1. **双向链表在 LRU 中的角色**：
   - 普通数组若要将一个元素提到最前面，必须将前面所有元素后移，时间开销为 $O(n)$。
   - 双向链表拥有前后双向指针，只要持有目标结点的指针，就可以在严格 $O(1)$ 时间内完成结点的“原地摘除”（Detach）并重新“挂载至表头”（Attach Head）。
   - 链表尾部直接指向最久未访问的淘汰候选者（Victim），在容量满时可在 $O(1)$ 时间内精准定位并执行驱逐。

2. **结点摘除与头插逻辑**：
   - **摘除操作**：
     ```c
     if (node->prev) node->prev->next = node->next; else cache->head = node->next;
     if (node->next) node->next->prev = node->prev; else cache->tail = node->prev;
     ```
   - **挂载至头部**：
     ```c
     node->prev = NULL;
     node->next = cache->head;
     if (cache->head) cache->head->prev = node; else cache->tail = node;
     cache->head = node;
     ```

3. **淘汰与更新协同**：
   - 当插入新键导致容量溢出时，优先摘除 `cache->tail` 对应的结点并调用 `free` 回收，使得活跃结点数量恰好保持在 `capacity` 之内。
   - 所有写入操作先校验分配成功，再接入链表，保证强异常安全性。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdlib.h>

typedef struct LruCache LruCache;

LruCache *lru_create(size_t capacity);
void lru_destroy(LruCache *cache);
bool lru_get(LruCache *cache, int key, int *out_val);
bool lru_put(LruCache *cache, int key, int val);
size_t lru_size(const LruCache *cache);

typedef struct LruNode {
    int key;
    int val;
    struct LruNode *prev;
    struct LruNode *next;
} LruNode;

struct LruCache {
    size_t capacity;
    size_t size;
    LruNode *head;
    LruNode *tail;
};

LruCache *lru_create(size_t capacity) {
    if (capacity == 0) {
        return NULL;
    }
    LruCache *cache = malloc(sizeof(LruCache));
    if (cache == NULL) {
        return NULL;
    }
    cache->capacity = capacity;
    cache->size = 0;
    cache->head = NULL;
    cache->tail = NULL;
    return cache;
}

void lru_destroy(LruCache *cache) {
    if (cache == NULL) {
        return;
    }
    LruNode *curr = cache->head;
    while (curr != NULL) {
        LruNode *next = curr->next;
        free(curr);
        curr = next;
    }
    free(cache);
}

static void lru_detach(LruCache *cache, LruNode *node) {
    if (node->prev != NULL) {
        node->prev->next = node->next;
    } else {
        cache->head = node->next;
    }

    if (node->next != NULL) {
        node->next->prev = node->prev;
    } else {
        cache->tail = node->prev;
    }

    node->prev = NULL;
    node->next = NULL;
}

static void lru_attach_head(LruCache *cache, LruNode *node) {
    node->prev = NULL;
    node->next = cache->head;
    if (cache->head != NULL) {
        cache->head->prev = node;
    } else {
        cache->tail = node;
    }
    cache->head = node;
}

bool lru_get(LruCache *cache, int key, int *out_val) {
    if (cache == NULL || out_val == NULL) {
        return false;
    }

    LruNode *curr = cache->head;
    while (curr != NULL) {
        if (curr->key == key) {
            *out_val = curr->val;
            lru_detach(cache, curr);
            lru_attach_head(cache, curr);
            return true;
        }
        curr = curr->next;
    }
    return false;
}

bool lru_put(LruCache *cache, int key, int val) {
    if (cache == NULL) {
        return false;
    }

    LruNode *curr = cache->head;
    while (curr != NULL) {
        if (curr->key == key) {
            curr->val = val;
            lru_detach(cache, curr);
            lru_attach_head(cache, curr);
            return true;
        }
        curr = curr->next;
    }

    if (cache->size == cache->capacity) {
        LruNode *victim = cache->tail;
        lru_detach(cache, victim);
        free(victim);
        cache->size--;
    }

    LruNode *new_node = malloc(sizeof(LruNode));
    if (new_node == NULL) {
        return false;
    }
    new_node->key = key;
    new_node->val = val;
    new_node->prev = NULL;
    new_node->next = NULL;

    lru_attach_head(cache, new_node);
    cache->size++;
    return true;
}

size_t lru_size(const LruCache *cache) {
    return (cache == NULL) ? 0 : cache->size;
}

int main(void) {
    lru_destroy(NULL);
    assert(lru_size(NULL) == 0);

    LruCache *c = lru_create(3);
    assert(c != NULL);

    int val = 0;
    assert(!lru_get(c, 1, &val));

    assert(lru_put(c, 1, 100));
    assert(lru_put(c, 2, 200));
    assert(lru_put(c, 3, 300));
    assert(lru_size(c) == 3);

    assert(lru_get(c, 1, &val) && val == 100);

    assert(lru_put(c, 4, 400));
    assert(!lru_get(c, 2, &val));
    assert(lru_get(c, 3, &val) && val == 300);
    assert(lru_get(c, 1, &val) && val == 100);
    assert(lru_get(c, 4, &val) && val == 400);

    assert(lru_put(c, 1, 999));
    assert(lru_get(c, 1, &val) && val == 999);

    lru_destroy(c);
    return 0;
}
```

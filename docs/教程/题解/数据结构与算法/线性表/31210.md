# 题解：31210

## 题目

结合连续数组的高速缓存局部性与链表的高效插入优势，设计并实现块状展开链表抽象数据类型（Unrolled Linked List ADT）：

```c
#include <stdbool.h>
#include <stddef.h>

#define UNROLLED_NODE_CAPACITY 8

typedef struct UnrolledList UnrolledList;

UnrolledList *unrolled_create(void);
void unrolled_destroy(UnrolledList *list);
bool unrolled_insert(UnrolledList *list, size_t index, int value);
bool unrolled_erase(UnrolledList *list, size_t index, int *out_value);
bool unrolled_get(const UnrolledList *list, size_t index, int *out_value);
size_t unrolled_size(const UnrolledList *list);
```

要求：

1. **块状结点存储**：每个链表结点内部包含一个容量为固定常数 `UNROLLED_NODE_CAPACITY`（定义为 8）的紧凑数组以及当前有效元素计数器。
2. **跨块跳跃与物理寻址**：按逻辑下标 `index` 查找目标时，利用各结点的元素计数快速跨块跳跃，大幅降低单链表遍历的指针解引用次数。
3. **结点分裂机制（Node Splitting）**：在指定位置插入元素时，若定位到的目标块已满（`count == UNROLLED_NODE_CAPACITY`），必须将该结点平分为两个各包含一半元素的结点，再在对应半块中插入新元素，维持块容量不变式。
4. **空块回收机制**：删除元素时，在块内执行紧凑平移；若某结点中的有效元素数量减少至 0，必须将该空结点从链表中摘除并调用 `free` 回收其内存。
5. **强异常安全与边界防御**：内存分配失败时不修改原链表并返回 `false`；越界访问或参数无效时返回 `false` 且不修改输出指针指向的对象；`unrolled_destroy` 彻底回收全部块结点。

## 解析

展开链表（Unrolled Linked List）是一种高阶混合线性表，它在顺序表与单链表之间取得了极佳的工程权衡：
- 普通单链表每个结点仅包含一个数据元素，指针额外开销占比极大（如 64 位下指针占 8 字节），且结点分散在堆中各处，CPU 缓存命中率极差；
- 普通顺序表在中间插入删除时，需要将后续成千上万个元素整体平移，开销巨大；
- 展开链表将多个元素打包在一个块结点内，兼具单链表低成本插入/删除拓扑优势与数组局部连续高速读写的特性。

1. **逻辑下标跨块定位**：
   - 设全局逻辑下标为 `index`。令剩余偏移 `offset = index`，当前扫描结点为 `curr = head`。
   - 若 `offset >= curr->count`，说明目标位于后续块中，令 `offset -= curr->count`，并推进 `curr = curr->next`。
   - 该跳跃过程以块为单位前进，将平均指针寻址次数降低为原来的 $1 / B$（其中 $B = UNROLLED\_NODE\_CAPACITY$）。

2. **结点满分裂（Split）算法**：
   - 当向一个已经拥有 8 个元素的满块中插入第 9 个元素时，若不进行分裂将破坏局部数组的物理边界。
   - 分裂步骤：
     1. 分配一个全新的展开结点 `new_node`；
     2. 将当前结点后半部分的 4 个元素使用 `memcpy` 复制到 `new_node` 中；
     3. 更新元素计数：`curr->count = 4`，`new_node->count = 4`；
     4. 调整链表指针：`new_node->next = curr->next; curr->next = new_node;`；
     5. 判定插入点：若原待插入偏移 `offset <= 4`，在前半块 `curr` 中插入；否则在后半块 `new_node` 的 `offset - 4` 处插入。
   - 分裂保证了每个结点的元素数量始终在安全范围内，且单次插入的移动元素个数至多为 $B$，时间复杂度上限被严格限制。

3. **空块修剪与全生命周期清理**：
   - 当持续删除导致某个块结点的有效元素计数 `count == 0` 时，保留空结点不仅浪费内存，还会拖慢后续遍历。
   - 通过单链表常规的双指针（`prev` 与 `curr`）在常数时间内将空块从链表中摘除并安全释放。
   - 销毁整个展开链表时，遍历单链表逐个释放每个块结点及其内部数组。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdlib.h>
#include <string.h>

#define UNROLLED_NODE_CAPACITY 8

typedef struct UnrolledList UnrolledList;

UnrolledList *unrolled_create(void);
void unrolled_destroy(UnrolledList *list);
bool unrolled_insert(UnrolledList *list, size_t index, int value);
bool unrolled_erase(UnrolledList *list, size_t index, int *out_value);
bool unrolled_get(const UnrolledList *list, size_t index, int *out_value);
size_t unrolled_size(const UnrolledList *list);

typedef struct UnrolledNode UnrolledNode;

struct UnrolledNode {
    int elements[UNROLLED_NODE_CAPACITY];
    size_t count;
    UnrolledNode *next;
};

struct UnrolledList {
    UnrolledNode *head;
    size_t total_size;
};

UnrolledList *unrolled_create(void) {
    UnrolledList *list = malloc(sizeof(UnrolledList));
    if (list == NULL) {
        return NULL;
    }
    list->head = NULL;
    list->total_size = 0;
    return list;
}

void unrolled_destroy(UnrolledList *list) {
    if (list == NULL) {
        return;
    }
    UnrolledNode *curr = list->head;
    while (curr != NULL) {
        UnrolledNode *next = curr->next;
        free(curr);
        curr = next;
    }
    free(list);
}

static UnrolledNode *create_unrolled_node(void) {
    UnrolledNode *node = malloc(sizeof(UnrolledNode));
    if (node == NULL) {
        return NULL;
    }
    node->count = 0;
    node->next = NULL;
    return node;
}

bool unrolled_insert(UnrolledList *list, size_t index, int value) {
    if (list == NULL || index > list->total_size) {
        return false;
    }

    if (list->head == NULL) {
        UnrolledNode *node = create_unrolled_node();
        if (node == NULL) {
            return false;
        }
        node->elements[0] = value;
        node->count = 1;
        list->head = node;
        list->total_size = 1;
        return true;
    }

    UnrolledNode *curr = list->head;
    size_t offset = index;

    while (curr->next != NULL && offset > curr->count) {
        offset -= curr->count;
        curr = curr->next;
    }

    if (offset == curr->count && curr->count == UNROLLED_NODE_CAPACITY && curr->next != NULL) {
        curr = curr->next;
        offset = 0;
    }

    if (curr->count < UNROLLED_NODE_CAPACITY) {
        memmove(&curr->elements[offset + 1],
                &curr->elements[offset],
                (curr->count - offset) * sizeof(int));
        curr->elements[offset] = value;
        curr->count++;
        list->total_size++;
        return true;
    }

    UnrolledNode *new_node = create_unrolled_node();
    if (new_node == NULL) {
        return false;
    }

    size_t split_idx = UNROLLED_NODE_CAPACITY / 2;
    size_t move_count = UNROLLED_NODE_CAPACITY - split_idx;
    memcpy(new_node->elements, &curr->elements[split_idx], move_count * sizeof(int));
    new_node->count = move_count;
    curr->count = split_idx;

    new_node->next = curr->next;
    curr->next = new_node;

    if (offset <= split_idx) {
        memmove(&curr->elements[offset + 1],
                &curr->elements[offset],
                (curr->count - offset) * sizeof(int));
        curr->elements[offset] = value;
        curr->count++;
    } else {
        size_t new_offset = offset - split_idx;
        memmove(&new_node->elements[new_offset + 1],
                &new_node->elements[new_offset],
                (new_node->count - new_offset) * sizeof(int));
        new_node->elements[new_offset] = value;
        new_node->count++;
    }

    list->total_size++;
    return true;
}

bool unrolled_erase(UnrolledList *list, size_t index, int *out_value) {
    if (list == NULL || out_value == NULL || index >= list->total_size) {
        return false;
    }

    UnrolledNode *curr = list->head;
    UnrolledNode *prev = NULL;
    size_t offset = index;

    while (curr != NULL && offset >= curr->count) {
        offset -= curr->count;
        prev = curr;
        curr = curr->next;
    }

    if (curr == NULL) {
        return false;
    }

    *out_value = curr->elements[offset];
    memmove(&curr->elements[offset],
            &curr->elements[offset + 1],
            (curr->count - offset - 1) * sizeof(int));
    curr->count--;
    list->total_size--;

    if (curr->count == 0) {
        if (prev == NULL) {
            list->head = curr->next;
        } else {
            prev->next = curr->next;
        }
        free(curr);
    }

    return true;
}

bool unrolled_get(const UnrolledList *list, size_t index, int *out_value) {
    if (list == NULL || out_value == NULL || index >= list->total_size) {
        return false;
    }

    const UnrolledNode *curr = list->head;
    size_t offset = index;

    while (curr != NULL && offset >= curr->count) {
        offset -= curr->count;
        curr = curr->next;
    }

    if (curr == NULL) {
        return false;
    }

    *out_value = curr->elements[offset];
    return true;
}

size_t unrolled_size(const UnrolledList *list) {
    return (list == NULL) ? 0 : list->total_size;
}

int main(void) {
    unrolled_destroy(NULL);
    assert(unrolled_size(NULL) == 0);

    UnrolledList *list = unrolled_create();
    assert(list != NULL);
    assert(unrolled_size(list) == 0);

    int val = 0;
    assert(!unrolled_get(list, 0, &val));
    assert(!unrolled_erase(list, 0, &val));

    for (int i = 0; i < 30; ++i) {
        assert(unrolled_insert(list, (size_t)i, i * 10));
    }
    assert(unrolled_size(list) == 30);

    for (size_t i = 0; i < 30; ++i) {
        assert(unrolled_get(list, i, &val) && val == (int)i * 10);
    }

    assert(unrolled_insert(list, 15, 999));
    assert(unrolled_size(list) == 31);
    assert(unrolled_get(list, 15, &val) && val == 999);
    assert(unrolled_get(list, 16, &val) && val == 150);

    assert(unrolled_erase(list, 15, &val) && val == 999);
    assert(unrolled_size(list) == 30);
    assert(unrolled_get(list, 15, &val) && val == 150);

    for (int i = 0; i < 30; ++i) {
        assert(unrolled_erase(list, 0, &val) && val == i * 10);
    }
    assert(unrolled_size(list) == 0);
    assert(list->head == NULL);

    unrolled_destroy(list);
    return 0;
}
```

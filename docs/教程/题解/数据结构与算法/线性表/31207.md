# 题解：31207

## 题目

实现基于独立哨兵结点的双向循环链表抽象数据类型（Sentinel Doubly Linked Circular List），支持常数时间任意端操作与链表拼接：

```c
#include <stdbool.h>
#include <stddef.h>

typedef struct DList DList;

DList *dlist_create(void);
void dlist_destroy(DList *list);
bool dlist_push_back(DList *list, int val);
bool dlist_push_front(DList *list, int val);
bool dlist_pop_back(DList *list, int *out_val);
bool dlist_pop_front(DList *list, int *out_val);
bool dlist_splice_back(DList *target, DList *source);
size_t dlist_size(const DList *list);
```

要求：

1. **哨兵自闭环不变式**：链表内部包含一个固定的哨兵头结点（sentinel）。链表为空时，哨兵结点的 `next` 与 `prev` 均自环指向自身；链表非空时，`sentinel.next` 指向首数据结点，`sentinel.prev` 指向末数据结点。
2. **零空指针特判分支**：利用双向循环对称性，插入与删除底层逻辑严禁包含任何针对应对 `head == NULL` 或 `tail == NULL` 的特判分支。
3. **整表常数级剪切追加（Splice）**：`dlist_splice_back` 将 `source` 中的全部结点在严格 $O(1)$ 时间内按原序完整剪切并追加到 `target` 表尾，拼接完成后 `source` 自动恢复为空表。
4. **内存安全契约**：销毁链表时仅释放有效数据结点与主结构体，哨兵结点随主结构体自动回收，保证 0 内存泄漏。

## 解析

在 Linux 内核（`struct list_head`）和工业级运行时底层，双向循环带哨兵链表是应用最为广泛的链式容器。普通双向链表在增删结点时，必须针对“空表插入”、“首结点删除”、“尾结点删除”编写大量易错的 `if-else` 分支。引入哨兵结点能够让所有有效数据结点的插入与删除在结构逻辑上完全等价。

1. **统一的结点插入与断开原子操作**：
   - **在任意两结点之间插入新结点**：
     ```c
     node->prev = prev;
     node->next = next;
     prev->next = node;
     next->prev = node;
     ```
     - 在表尾插入新值，只需调用 `insert_between(list->sentinel.prev, &list->sentinel, val)`；
     - 在表头插入新值，只需调用 `insert_between(&list->sentinel, list->sentinel.next, val)`。
   - **摘除任意现有结点**：
     ```c
     node->prev->next = node->next;
     node->next->prev = node->prev;
     free(node);
     ```
     - 弹出表头只需摘除 `list->sentinel.next`；弹出表尾只需摘除 `list->sentinel.prev`。

2. **$O(1)$ 整表剪切拼接（Splice）原理**：
   - 设 `target` 原末结点为 `tgt_last = target->sentinel.prev`；
   - 设 `source` 首尾有效结点分别为 `src_first = source->sentinel.next` 与 `src_last = source->sentinel.prev`。
   - 若 `source` 非空，只需调整 4 个指针的指向：
     1. 连接 `target` 末尾与 `source` 头部：`tgt_last->next = src_first; src_first->prev = tgt_last;`
     2. 连接 `source` 末尾与 `target` 哨兵：`src_last->next = &target->sentinel; target->sentinel.prev = src_last;`
   - 最后重置 `source` 哨兵自闭环：`source->sentinel.next = source->sentinel.prev = &source->sentinel;`。
   - 整个拼接仅涉及 6 次指针赋值与计数相加，与 `source` 的结点数量完全无关，时间复杂度为严格的常数 $O(1)$。

## 答案

```c
#include <assert.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdlib.h>

typedef struct DList DList;

typedef struct DNode {
    int value;
    struct DNode *prev;
    struct DNode *next;
} DNode;

struct DList {
    DNode sentinel;
    size_t size;
};

DList *dlist_create(void) {
    DList *list = malloc(sizeof(DList));
    if (list == NULL) {
        return NULL;
    }
    list->sentinel.prev = &list->sentinel;
    list->sentinel.next = &list->sentinel;
    list->size = 0;
    return list;
}

void dlist_destroy(DList *list) {
    if (list == NULL) {
        return;
    }
    DNode *curr = list->sentinel.next;
    while (curr != &list->sentinel) {
        DNode *next = curr->next;
        free(curr);
        curr = next;
    }
    free(list);
}

static bool dlist_insert_between(DNode *prev, DNode *next, int val) {
    DNode *node = malloc(sizeof(DNode));
    if (node == NULL) {
        return false;
    }
    node->value = val;
    node->prev = prev;
    node->next = next;
    prev->next = node;
    next->prev = node;
    return true;
}

static void dlist_remove_node(DNode *node) {
    node->prev->next = node->next;
    node->next->prev = node->prev;
    free(node);
}

bool dlist_push_back(DList *list, int val) {
    if (list == NULL || list->size == SIZE_MAX) {
        return false;
    }
    if (!dlist_insert_between(list->sentinel.prev, &list->sentinel, val)) {
        return false;
    }
    list->size++;
    return true;
}

bool dlist_push_front(DList *list, int val) {
    if (list == NULL || list->size == SIZE_MAX) {
        return false;
    }
    if (!dlist_insert_between(&list->sentinel, list->sentinel.next, val)) {
        return false;
    }
    list->size++;
    return true;
}

bool dlist_pop_back(DList *list, int *out_val) {
    if (list == NULL || out_val == NULL || list->size == 0) {
        return false;
    }
    DNode *target = list->sentinel.prev;
    *out_val = target->value;
    dlist_remove_node(target);
    list->size--;
    return true;
}

bool dlist_pop_front(DList *list, int *out_val) {
    if (list == NULL || out_val == NULL || list->size == 0) {
        return false;
    }
    DNode *target = list->sentinel.next;
    *out_val = target->value;
    dlist_remove_node(target);
    list->size--;
    return true;
}

bool dlist_splice_back(DList *target, DList *source) {
    if (target == NULL || source == NULL || target == source) {
        return false;
    }
    if (source->size == 0) {
        return true;
    }
    if (target->size > SIZE_MAX - source->size) {
        return false;
    }

    DNode *src_first = source->sentinel.next;
    DNode *src_last = source->sentinel.prev;
    DNode *tgt_last = target->sentinel.prev;

    tgt_last->next = src_first;
    src_first->prev = tgt_last;

    src_last->next = &target->sentinel;
    target->sentinel.prev = src_last;

    target->size += source->size;

    source->sentinel.next = &source->sentinel;
    source->sentinel.prev = &source->sentinel;
    source->size = 0;
    return true;
}

size_t dlist_size(const DList *list) {
    return (list == NULL) ? 0 : list->size;
}

int main(void) {
    dlist_destroy(NULL);
    assert(dlist_size(NULL) == 0);

    DList *l1 = dlist_create();
    DList *l2 = dlist_create();
    assert(l1 != NULL && l2 != NULL);

    int val = 0;
    assert(!dlist_pop_front(l1, &val));
    assert(!dlist_pop_back(l1, &val));

    assert(dlist_push_back(l1, 10));
    assert(dlist_push_back(l1, 20));

    assert(dlist_push_back(l2, 30));
    assert(dlist_push_back(l2, 40));
    assert(dlist_push_back(l2, 50));

    assert(dlist_splice_back(l1, l2));
    assert(dlist_size(l1) == 5);
    assert(dlist_size(l2) == 0);

    assert(dlist_pop_front(l1, &val) && val == 10);
    assert(dlist_pop_front(l1, &val) && val == 20);
    assert(dlist_pop_front(l1, &val) && val == 30);
    assert(dlist_pop_back(l1, &val) && val == 50);
    assert(dlist_pop_back(l1, &val) && val == 40);
    assert(dlist_size(l1) == 0);

    dlist_destroy(l1);
    dlist_destroy(l2);
    return 0;
}
```
